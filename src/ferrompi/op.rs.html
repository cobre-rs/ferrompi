1//! User-defined MPI reduction operations via [`MPI_Op_create`].
2//!
3//! This module provides [`UserOp<T>`], a safe wrapper around a user-supplied
4//! Rust closure that MPI invokes during reduction collectives
5//! (`MPI_Reduce`, `MPI_Allreduce`, `MPI_Scan`, etc.).
6//!
7//! ## `compile_fail` doctest — `Send + Sync` bound
8//!
9//! ```compile_fail
10//! use ferrompi::UserOp;
11//! let local = std::rc::Rc::new(0i32);
12//! let op = UserOp::new(move |_a: &[f64], _b: &mut [f64]| {
13//!     let _ = local.clone();
14//! });
15//! ```
16

30/// Reduction operations
31///
32/// The `Replace` and `NoOp` variants are only available with the `rma` feature.
33///
34/// # Feature-gated variants

41    doc = "// This must not compile without --features rma."

47    feature = "rma",
48    doc = "// With --features rma, ReduceOp::Replace is available."
49)]
50#[cfg_attr(feature = "rma", doc = "let _ = ferrompi::ReduceOp::Replace;")]
51#[cfg_attr(feature = "rma", doc = "```")]
52#[derive(Debug, Clone, Copy, PartialEq, Eq)]
53#[repr(i32)]
54pub enum ReduceOp {
55    /// Sum of values
56    Sum = 0,
57    /// Maximum value
58    Max = 1,
59    /// Minimum value
60    Min = 2,
61    /// Product of values
62    Prod = 3,
63    /// Bitwise OR (`MPI_BOR`). Valid only for integer types; MPI returns
64    /// `MPI_ERR_OP` when used with floating-point types.
65    BitwiseOr = 4,
66    /// Bitwise AND (`MPI_BAND`). Valid only for integer types; MPI returns
67    /// `MPI_ERR_OP` when used with floating-point types.
68    BitwiseAnd = 5,
69    /// Bitwise XOR (`MPI_BXOR`). Valid only for integer types; MPI returns
70    /// `MPI_ERR_OP` when used with floating-point types.
71    BitwiseXor = 6,
72    /// Logical OR (`MPI_LOR`). Interprets nonzero as `true`. Valid for
73    /// integer types.
74    LogicalOr = 7,
75    /// Logical AND (`MPI_LAND`). Interprets nonzero as `true`. Valid for
76    /// integer types.
77    LogicalAnd = 8,
78    /// Logical XOR (`MPI_LXOR`). Interprets nonzero as `true`. Valid for
79    /// integer types.
80    LogicalXor = 9,
81    /// Maximum value with location (`MPI_MAXLOC`). Returns the maximum value
82    /// and the rank (index) where it occurred. Only valid with
83    /// [`MpiIndexedDatatype`](crate::MpiIndexedDatatype) via
84    /// [`Communicator::allreduce_indexed`](crate::Communicator::allreduce_indexed).
85    MaxLoc = 10,
86    /// Minimum value with location (`MPI_MINLOC`). Returns the minimum value
87    /// and the rank (index) where it occurred. Only valid with
88    /// [`MpiIndexedDatatype`](crate::MpiIndexedDatatype) via
89    /// [`Communicator::allreduce_indexed`](crate::Communicator::allreduce_indexed).
90    MinLoc = 11,
91    /// Replace the target buffer with the source value (`MPI_REPLACE`).
92    ///
93    /// Only valid for `MPI_Accumulate`-family operations. Passing to
94    /// `allreduce`, `reduce`, `scan`, etc. returns `MPI_ERR_OP` from MPI.
95    ///
96    /// This variant is only present when the `rma` feature is enabled.
97    #[cfg(feature = "rma")]
98    Replace = 12,
99    /// No-op: leaves the target buffer unchanged (`MPI_NO_OP`).
100    ///
101    /// Only valid for `MPI_Accumulate`-family operations. Passing to
102    /// `allreduce`, `reduce`, `scan`, etc. returns `MPI_ERR_OP` from MPI.
103    ///
104    /// # Compile-time availability
105    ///
106    /// This variant is only present when the `rma` feature is enabled.
107    #[cfg(feature = "rma")]
108    NoOp = 13,
109}

112// Slot count — must match MAX_OPS in csrc/ferrompi.c
113// ============================================================================
114const MAX_OPS: usize = 16;
115
116/// Type-erased closure stored in each op slot: it receives MPI's raw
117/// `invec`/`inoutvec` pointers and element count.
118type ByteClosure = Box<dyn Fn(*const c_void, *mut c_void, usize) + Send + Sync + 'static>;
119
120/// Per-slot registry of thin pointers to boxed closures.
121///
122/// Each slot holds `Box::into_raw(Box::new(byte_closure))` — a thin
123/// `*mut ByteClosure` — or null.  Protocol:
124/// - `UserOp::new_impl` publishes a slot with `Ordering::Release` before
125///   calling `MPI_Op_create`.
126/// - `rust_user_op_invoke` (the trampoline) loads a slot with
127///   `Ordering::Acquire`.
128/// - `ferrompi_op_drop_closure` swaps a slot to null only after
129///   `MPI_Op_free` returns, so no trampoline call can observe a freed
130///   closure.
131static REGISTRY: [AtomicPtr<ByteClosure>; MAX_OPS] =
132    [const { AtomicPtr::new(std::ptr::null_mut()) }; MAX_OPS];
133
134// ============================================================================
135// Extern "C" callbacks exposed to the C layer
136// ============================================================================
137
138/// Called by each C trampoline `ferrompi_user_op_trampoline_N`.
139///
140/// The C trampoline passes only its slot number and the raw buffer pointers
141/// it received from MPI.  This function loads the slot's closure and invokes
142/// it, wrapped in `catch_unwind`.
143///
144/// # Safety
145///
146/// * `slot` is in range `0..MAX_OPS` and was published by `UserOp::new_impl`
147///   before `MPI_Op_create` was called for it.
148/// * `invec` and `inoutvec` are the pointers MPI passed to the user
149///   function, each to `len` elements of the datatype the op was applied
150///   with.
151/// * `len` is non-negative.
152///
153/// Called from C, so the ABI must be exactly `extern "C"`.
154#[unsafe(no_mangle)]
155pub unsafe extern "C" fn rust_user_op_invoke(
156    slot: std::ffi::c_int,
157    invec: *const c_void,
158    inoutvec: *mut c_void,
159    len: std::ffi::c_int,
160) {
161    let ptr = REGISTRY[slot as usize].load(Ordering::Acquire);
162    if ptr.is_null() {
163        // A null slot here is an invariant violation: the publish-before-
164        // create / drop-after-free protocol on REGISTRY guarantees a live
165        // closure for every slot MPI can still invoke.  Treat it the same as
166        // the panic path below — abort rather than deref a null pointer.
167        std::process::abort();
168    }
169    // SAFETY: ptr is non-null, published by `new_impl` with `Ordering::Release`
170    // before `MPI_Op_create`, and this `Ordering::Acquire` load of the same
171    // atomic synchronizes-with that store.  `ferrompi_op_drop_closure` nulls
172    // the slot only after `MPI_Op_free` returns, so no drop can race with
173    // this borrow.
174    let closure: &ByteClosure = unsafe { &*ptr };
175
176    // Wrap the closure call in catch_unwind (ADR-0005 Decision 6).
177    // A panic across the FFI boundary is UB; abort on Err.
178    let result = std::panic::catch_unwind(std::panic::AssertUnwindSafe(|| {
179        closure(invec, inoutvec, len as usize);
180    }));
181    if result.is_err() {
182        // A panic inside a user-defined reduction closure is a fatal
183        // programming error.  Abort immediately to prevent silent data
184        // corruption in the collective result (ADR-0005 Decision 6).
185        std::process::abort();
186    }
187}

189/// Called by `ferrompi_op_free` (in C) after `MPI_Op_free` returns — including
190/// when the `ferrompi_finalize` sweep invokes `ferrompi_op_free` on a still-used
191/// slot — and by `UserOp::new_impl`'s rollback path when `MPI_Op_create` fails.
192///
193/// Swaps the slot to null and, if it held a pointer, drops the boxed
194/// closure.  Idempotent: a second call on the same slot is a no-op — this is
195/// the one drop routine both the C free path and the Rust rollback path
196/// share.
197///
198/// # Safety
199///
200/// * `slot` must be in range `0..MAX_OPS`.
201/// * If `REGISTRY[slot]` is non-null, it must point to a `ByteClosure`
202///   produced by `Box::into_raw` in `UserOp::new_impl` that has not yet been
203///   dropped, and no trampoline call for this slot may be in flight (i.e.
204///   `MPI_Op_create` for this slot never succeeded, or `MPI_Op_free` for it
205///   has already returned).
206#[unsafe(no_mangle)]
207pub unsafe extern "C" fn ferrompi_op_drop_closure(slot: i32) {
208    if slot < 0 || slot as usize >= MAX_OPS {
209        return;
210    }
211    let ptr = REGISTRY[slot as usize].swap(std::ptr::null_mut(), Ordering::Acquire);
212    if ptr.is_null() {
213        return;
214    }
215    // SAFETY: ptr was produced by Box::into_raw in new_impl and has not been
216    // dropped; the swap above claimed sole ownership by nulling the slot,
217    // and the caller guarantees no trampoline call for this slot is in
218    // flight.
219    drop(unsafe { Box::from_raw(ptr) });
220}

226/// A user-defined MPI reduction operation backed by a Rust closure.
227///
228/// `T` must implement [`MpiDatatype`] — i.e., it must be one of the primitive
229/// types recognised by ferrompi (`f32`, `f64`, `i32`, `i64`, `u8`, `u32`,
230/// `u64`).
231///
232/// The closure receives `invec` as a shared slice and `inoutvec` as a mutable
233/// slice of the same length.  It must accumulate `invec[i]` into `inoutvec[i]`
234/// for each index `i` — the standard MPI semantics for user reduction
235/// functions.
236///
237/// # Thread-safety
238///
239/// The closure is called from whichever thread MPI uses internally for the
240/// reduction.  Under `MPI_THREAD_MULTIPLE` the same op may be invoked
241/// concurrently from multiple threads; the closure must be safe for concurrent
242/// invocation, enforced by the `Sync` bound. The closure is moved into a
243/// static slot table accessible from any thread (MPI may call it from an
244/// internal thread pool), enforced by the `Send` bound. The closure is held
245/// until `MPI_Op_free` returns, which may be much later than the call site,
246/// enforced by the `'static` bound.
247///
248/// # Panic behaviour
249///
250/// A panic inside the closure **aborts the process**.
251///
252/// Panicking across a C FFI boundary is undefined behaviour; there is no
253/// mechanism for MPI to propagate or observe a Rust panic.  The trampoline
254/// wraps every closure call in [`std::panic::catch_unwind`]: on `Err` it calls
255/// [`std::process::abort`] before the panic can reach the C frame.  Treat a
256/// panic inside a reduction closure as a fatal programming error.
257///
258/// # Slot-table limit
259///
260/// At most 16 `UserOp` instances may be live concurrently per process.
261/// Attempting to create a seventeenth returns [`Error::Mpi`] with class `Other`.
262///
263/// If a `UserOp` outlives the `Mpi` handle, finalizing MPI frees its op and
264/// drops its closure, and dropping the `UserOp` afterwards does nothing.
265///
266/// # Examples
267///
268/// ```no_run
269/// use ferrompi::{Mpi, UserOp};
270///
271/// let mpi = Mpi::init().unwrap();
272/// let world = mpi.world();
273///
274/// let op: UserOp<f64> = UserOp::new(|invec: &[f64], inoutvec: &mut [f64]| {
275///     for (x, y) in invec.iter().zip(inoutvec.iter_mut()) {
276///         *y = x.max(*y);
277///     }
278/// }).unwrap();
279///
280/// let send = vec![world.rank() as f64 + 1.5_f64];
281/// let mut recv = vec![0.0_f64];
282/// world.allreduce_with_op(&send, &mut recv, &op).unwrap();
283/// ```
284pub struct UserOp<T: MpiDatatype> {
285    pub(crate) handle: i32,
286    _marker: PhantomData<T>,
287}
288
289impl<T: MpiDatatype> UserOp<T> {
290    /// Create a commutative user-defined reduction op.
291    ///
292    /// MPI is permitted to reorder operands for optimisation purposes.  Use
293    /// this constructor for operations that satisfy `f(a, b) == f(b, a)` —
294    /// element-wise sums, maxima, minima, etc.
295    ///
296    /// # Errors
297    ///
298    /// Returns `Err` if the op-slot table is full (16 concurrent `UserOp`s)
299    /// or if `MPI_Op_create` fails.
300    pub fn new<F>(f: F) -> Result<Self>
301    where
302        F: Fn(&[T], &mut [T]) + Send + Sync + 'static,
303    {
304        Self::new_impl(f, 1)
305    }
306
307    /// Create a non-commutative user-defined reduction op.
308    ///
309    /// MPI will not reorder operands.  Use this constructor for operations
310    /// where order matters — matrix multiplication, string concatenation, etc.
311    ///
312    /// # Errors
313    ///
314    /// Returns `Err` if the op-slot table is full or if `MPI_Op_create` fails.
315    pub fn new_noncommutative<F>(f: F) -> Result<Self>
316    where
317        F: Fn(&[T], &mut [T]) + Send + Sync + 'static,
318    {
319        Self::new_impl(f, 0)
320    }
321
322    fn new_impl<F>(f: F, commute: i32) -> Result<Self>
323    where
324        F: Fn(&[T], &mut [T]) + Send + Sync + 'static,
325    {
326        // Step 1: allocate a slot in the C-side op table.
327        let mut slot: i32 = -1;
328        // SAFETY: slot is a local out-parameter written by ferrompi_op_alloc_slot
329        // before this function reads it below (guarded by the `ret != 0` check).
330        let ret = unsafe { ffi::ferrompi_op_alloc_slot(&mut slot) };
331        if ret != 0 {
332            return Err(Error::Mpi {
333                class: MpiErrorClass::Other,
334                code: ret,
335                message: "op-slot table is full (MAX_OPS=16 concurrent UserOps)".to_string(),
336                operation: Some("op_create"),
337            });
338        }
339        let idx = slot as usize;
340        debug_assert!(idx < MAX_OPS, "slot out of range");
341
342        // Step 2: wrap the typed closure in the type-erased adapter.
343        let byte_closure: ByteClosure = typed_adapter::<T, F>(f);
344
345        // Step 3: publish the boxed closure to the registry.  A thin
346        // `*mut ByteClosure` — no fat-pointer decomposition needed.  This
347        // Release store, paired with the trampoline's Acquire load of the
348        // same atomic, synchronizes-with that load directly (no external
349        // happens-before edge required).
350        let ptr: *mut ByteClosure = Box::into_raw(Box::new(byte_closure));
351        REGISTRY[idx].store(ptr, Ordering::Release);
352
353        // Step 4: call MPI_Op_create via the C shim.
354        let mut handle: i32 = -1;
355        // SAFETY: handle is a local out-parameter written by
356        // ferrompi_op_create_user before this function reads it below; slot is
357        // the value ferrompi_op_alloc_slot returned above.
358        let ret = unsafe { ffi::ferrompi_op_create_user(slot, commute, &mut handle) };
359        if ret != 0 {
360            // Rollback: MPI_Op_create failed so no MPI_Op was registered and
361            // no trampoline call for this slot can be in flight.  Use the
362            // same drop routine the C free path uses (ferrompi_op_drop_closure
363            // is idempotent and bounds-checked), then release the slot
364            // without calling MPI_Op_free — the slot never held a live
365            // MPI_Op, and MPI_Op_free on MPI_OP_NULL is implementation-defined.
366            // SAFETY: slot is in range (checked by ferrompi_op_alloc_slot
367            // above); the closure at REGISTRY[idx] was published above and no
368            // trampoline call for this slot has occurred, since
369            // ferrompi_op_create_user just returned failure.
370            unsafe { ferrompi_op_drop_closure(slot) };
371            // SAFETY: slot is the value ferrompi_op_alloc_slot returned above;
372            // the closure stored in it has just been dropped, so no dangling
373            // pointer remains for a later trampoline call to observe.
374            unsafe { ffi::ferrompi_op_free_slot_only(slot) };
375            return Err(Error::from_code_with_op(ret, "op_create"));
376        }
377        debug_assert_eq!(handle, slot);
378
379        Ok(UserOp {
380            handle,
381            _marker: PhantomData,
382        })
383    }
384}

386/// Wraps a typed reduction closure in the type-erased form `REGISTRY` stores.
387fn typed_adapter<T: MpiDatatype, F>(f: F) -> ByteClosure
388where
389    F: Fn(&[T], &mut [T]) + Send + Sync + 'static,
390{
391    Box::new(
392        move |invec: *const c_void, inoutvec: *mut c_void, len: usize| {
393            if len == 0 {
394                // MPI may pass null buffers for an empty reduction; an empty
395                // slice needs no pointer.
396                f(&[], &mut []);
397                return;
398            }
399            // SAFETY: invec/inoutvec point to `len` elements of the datatype the
400            // op was applied with, which allreduce_with_op always passes as T's
401            // own; MPI's buffers are aligned for that datatype; the two buffers
402            // are distinct; T: MpiDatatype is Copy with a stable layout.
403            let (invec, inoutvec) = unsafe {
404                (
405                    std::slice::from_raw_parts(invec.cast::<T>(), len),
406                    std::slice::from_raw_parts_mut(inoutvec.cast::<T>(), len),
407                )
408            };
409            f(invec, inoutvec);
410        },
411    )
412}

414impl<T: MpiDatatype> Drop for UserOp<T> {
415    fn drop(&mut self) {
416        if !rt::drop_guard("UserOp") {
417            return;
418        }
419        // Drop ordering (ADR-0005 Decision 3):
420        //   1. ferrompi_op_free → MPI_Op_free (MPI will not invoke the
421        //      trampoline after this returns).
422        //   2. ferrompi_op_free → ferrompi_op_drop_closure (Rust callback,
423        //      drops the Box).
424        //   3. ferrompi_op_free → free_op_slot (reclaims the C-side slot).
425        //
426        // The handle is valid for the lifetime of this UserOp; it is freed
427        // exactly once here.
428        let ret = unsafe {
429            // SAFETY: self.handle was allocated by UserOp::new_impl and has
430            // not been freed.  Drop is called exactly once.
431            ffi::ferrompi_op_free(self.handle)
432        };
433        // Log but do not panic in Drop.
434        if ret != 0 {
435            eprintln!("ferrompi: UserOp::drop — ferrompi_op_free returned error code {ret}");
436        }
437    }
438}
