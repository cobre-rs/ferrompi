11/// Element count at or below which request-handle scratch buffers live on the
12/// stack. Draining a handful-to-few-dozen in-flight requests on the completion
13/// path (halo exchange, ping-pong, progress loops) then incurs no allocator
14/// traffic; larger batches fall back to the heap. Used by both `Request` and
15/// `PersistentRequest`. Mirrors `FERROMPI_REQ_STACK` in `csrc/ferrompi.c`.
16const HANDLE_STACK_CAP: usize = 64;
17
18/// Run `f` with the handles of `items` copied into a stack buffer when the
19/// batch is small, falling back to a heap `Vec` only for large batches, and
20/// a same-length zeroed `done` buffer for `f` to report completions into.
21/// After `f` returns, calls `on_done` on every item whose `done` byte is
22/// non-zero, whatever `f`'s result. Removes the per-call `Vec<i64>`
23/// allocation on the completion path. Shared by `Request` and
24/// `PersistentRequest`.
25#[inline]
26pub(crate) fn with_handles<E, R>(
27    items: &mut [E],
28    handle: impl Fn(&E) -> i64,
29    f: impl FnOnce(&[i64], &mut [u8]) -> R,
30    on_done: impl Fn(&mut E),
31) -> R {
32    let len = items.len();
33    if len <= HANDLE_STACK_CAP {
34        let mut hbuf = [0i64; HANDLE_STACK_CAP];
35        let mut dbuf = [0u8; HANDLE_STACK_CAP];
36        for (slot, item) in hbuf[..len].iter_mut().zip(items.iter()) {
37            *slot = handle(item);
38        }
39        let result = f(&hbuf[..len], &mut dbuf[..len]);
40        for (item, &done) in items.iter_mut().zip(&dbuf[..len]) {
41            if done != 0 {
42                on_done(item);
43            }
44        }
45        result
46    } else {
47        let hvec: Vec<i64> = items.iter().map(&handle).collect();
48        let mut dvec = vec![0u8; len];
49        let result = f(&hvec, &mut dvec);
50        for (item, &done) in items.iter_mut().zip(&dvec) {
51            if done != 0 {
52                on_done(item);
53            }
54        }
55        result
56    }
57}

59/// The handle a batch call passes for `r`: `-1` (the C layer's null
60/// handle) once `r` has completed.
61fn handle_or_skip(r: &Request<'_>) -> i64 {
62    if r.completed {
63        -1
64    } else {
65        r.handle
66    }
67}

75/// Run `f` with zeroed `i32` index and status scratch buffers, both of length
76/// `len` and stack-allocated when small. Used for the `*some` outputs: the
77/// shim writes `statuses[k]` for each reported index.
78#[inline]
79fn with_index_buf<R>(len: usize, f: impl FnOnce(&mut [i32], &mut [ffi::FerrompiStatus]) -> R) -> R {
80    if len <= HANDLE_STACK_CAP {
81        let mut buf = [0i32; HANDLE_STACK_CAP];
82        let mut statuses = [ffi::FerrompiStatus::default(); HANDLE_STACK_CAP];
83        f(&mut buf[..len], &mut statuses[..len])
84    } else {
85        let mut buf = vec![0i32; len];
86        let mut statuses = vec![ffi::FerrompiStatus::default(); len];
87        f(&mut buf, &mut statuses)
88    }
89}

91/// The `(index, Status)` pairs a `*some` shim reported. Empty unless `ret` is
92/// success and `outcount` is positive.
93fn some_results(
94    ret: i32,
95    outcount: i64,
96    indices: &[i32],
97    statuses: &[ffi::FerrompiStatus],
98) -> Vec<(usize, Status)> {
99    if ret != 0 || outcount <= 0 {
100        return Vec::new();
101    }
102    let n = outcount as usize;
103    indices[..n]
104        .iter()
105        .zip(&statuses[..n])
106        .map(|(&index, &raw)| (index as usize, Status::from_ffi(raw)))
107        .collect()
108}

110/// Check a batch wait/test return code, appending the failing request's
111/// slice index to an `Error::Mpi` message. `failed` is the caller-slice
112/// index of the failing request as the C shim wrote it (`-1` when it
113/// found none). Shared by `Request::wait_all`, `Request::wait_any`,
114/// `Request::test_any` and `PersistentRequest::wait_all`.
115pub(crate) fn check_batch(ret: i32, operation: &'static str, failed: i64) -> Result<()> {
116    match Error::check_with_op(ret, operation) {
117        Err(Error::Mpi {
118            class,
119            code,
120            message,
121            operation,
122        }) if failed >= 0 => Err(Error::Mpi {
123            class,
124            code,
125            message: format!("{message} (request {failed})"),
126            operation,
127        }),
128        other => other,
129    }
130}

141/// A handle to a nonblocking MPI operation.
142///
143/// A request is created inside a [`scope`](crate::scope), or with the `rma`
144/// feature inside a passive epoch (`PassiveEpoch::rput`, `rget` and
145/// `raccumulate`), and borrows its buffers until the scope returns (the epoch,
146/// until its unlock), even after the request has completed. It is completed by
147/// [`wait`](Request::wait), [`test`](Request::test) or a batch call such as
148/// [`wait_all`](Request::wait_all) or, at the latest, by its scope or epoch,
149/// which waits for every request still pending before it returns or unwinds
150/// (the epoch, before it unlocks). Dropping or forgetting a request does not
151/// complete it early, and the scope or epoch still waits.
152///
153/// A request is not `Send`, so it cannot be moved to another thread:
154///
155/// ```compile_fail,E0277
156/// # use ferrompi::{Mpi, ThreadLevel};
157/// # fn main() -> ferrompi::Result<()> {
158/// # let mpi = Mpi::init_thread(ThreadLevel::Multiple)?;
159/// # let world = mpi.world();
160/// ferrompi::scope(|s| {
161///     let req = world.ibarrier(s)?;
162///     std::thread::scope(|t| {
163///         t.spawn(move || req.wait());
164///     });
165///     Ok(())
166/// })?;
167/// # Ok(())
168/// # }
169/// ```
170///
171/// A thread that needs a request opens its own scope and creates and waits it
172/// there:
173///
174/// ```no_run
175/// # use ferrompi::{Mpi, ThreadLevel};
176/// # fn main() -> ferrompi::Result<()> {
177/// # let mpi = Mpi::init_thread(ThreadLevel::Multiple)?;
178/// # let world = mpi.world();
179/// std::thread::scope(|t| {
180///     t.spawn(|| {
181///         ferrompi::scope(|s| {
182///             let req = world.ibarrier(s)?;
183///             req.wait()?;
184///             Ok(())
185///         })
186///     });
187/// });
188/// # Ok(())
189/// # }
190/// ```
191///
192/// # Example
193///
194/// ```no_run
195/// use ferrompi::{Mpi, ReduceOp};
196///
197/// let mpi = Mpi::init().unwrap();
198/// let world = mpi.world();
199///
200/// let send = vec![world.rank() as f64; 10];
201/// let mut recv = vec![0.0; 10];
202///
203/// ferrompi::scope(|s| {
204///     // Start nonblocking all-reduce
205///     let request = world.iallreduce(s, &send, &mut recv, ReduceOp::Sum)?;
206///
207///     // Do other work while communication proceeds...
208///
209///     // Wait for completion
210///     request.wait()?;
211///     Ok(())
212/// })
213/// .unwrap();
214///
215/// // Now recv contains the result
216/// println!("Sum: {:?}", recv);
217/// ```
218pub struct Request<'s> {
219    handle: i64,
220    completed: bool,
221    kind: RequestKind,
222    registry: &'s Registry,
223    // PERF: `usize`, not `u32`. With a `u32` the request is 24 bytes, its last
224    // word (slot, `completed`, `kind`) is stored in pieces and loaded whole, and
225    // ffi_overhead's `isend+irecv+wait` reads 7 ns slower.
226    slot: usize,
227}

258    /// Wait for this operation to complete.
259    ///
260    /// Blocks until the operation is finished. After this returns successfully,
261    /// MPI no longer uses the request's buffers; a request keeps them borrowed
262    /// until its scope returns (an epoch's, until its unlock).
263    ///
264    /// For a receive the returned [`Status`] holds the matched source, tag and
265    /// element count. For a send, collective or RMA request the `Status` is the
266    /// empty status, and so is a cancelled receive or a request that was
267    /// already completed (a [`test`](Request::test) that failed after MPI
268    /// completed it, say); only `error` has meaning, and this single-request
269    /// call leaves it `None`.
270    ///
271    /// A request cannot leave the thread that created it, so the thread-level
272    /// check cannot refuse its wait from safe code. Once `Mpi` is finalized the
273    /// call returns `Err(`[`Error::Finalized`]`)`.
274    ///
275    /// In a debug build at `Serialized`, a wait that overlaps another
276    /// thread's MPI call can neither run nor hand the request back, so it
277    /// prints a message and aborts the process.
278    #[inline]
279    pub fn wait(self) -> Result<Status> {
280        if self.completed {
281            return Ok(Status::EMPTY);
282        }
283        let mut status = ffi::FerrompiStatus::default();
284        self.wait_raw(&mut status)?;
285        Ok(Status::from_ffi(status))
286    }
287
288    /// The body of [`wait`](Request::wait) for a request that is not yet
289    /// completed, out of line so that the inline wrapper decodes `status` only
290    /// when its caller uses the result.
291    #[inline(never)]
292    fn wait_raw(self, status: &mut ffi::FerrompiStatus) -> Result<()> {
293        Error::check_with_op(rt::check_completion(), "wait")?;
294        // SAFETY: self.handle is a valid MPI request handle registered in the
295        // C-side request table by the nonblocking constructor that produced
296        // this Request; self.completed was false on entry (checked by wait),
297        // so MPI_Wait has not already consumed this handle. status is a valid
298        // out-parameter.
299        let ret = unsafe { ffi::ferrompi_wait(self.handle, status) };
300        #[cfg(debug_assertions)]
301        if ret == FERROMPI_ERR_THREAD_LEVEL {
302            // rt::check_completion() above already returns this same sentinel
303            // for a call from a non-init thread below Serialized, and
304            // propagates it via `?` before reaching here; so this can only be
305            // the Serialized overlap check rejecting the call before MPI saw
306            // it. `self` cannot be handed back, and the final wait of the
307            // scope or epoch that owns it would overlap the other thread's MPI
308            // call.
309            let _ = std::io::stderr().write_all(
310                b"ferrompi: Request::wait overlapped another thread's MPI call at ThreadLevel::Serialized\n",
311            );
312            std::process::abort();
313        }
314        // A refused call leaves the request pending, so its slot stays live
315        // for the final wait of the scope or epoch that owns it.
316        if ret != FERROMPI_ERR_FINALIZED && ret != FERROMPI_ERR_THREAD_LEVEL {
317            self.release();
318        }
319        Error::check_with_op(ret, "wait")
320    }

322    /// Test if this operation has completed without blocking.
323    ///
324    /// Returns `Some(`[`Status`]`)` if the operation is complete, `None`
325    /// otherwise. For a receive the `Status` holds the matched source, tag and
326    /// element count. For a send, collective or RMA request the `Status` is the
327    /// empty status, and so is a cancelled receive or a request that was
328    /// already completed; only `error` has meaning, and these single-request
329    /// calls leave it `None`.
330    ///
331    /// # Note
332    ///
333    /// If this returns `Some`, the request is consumed and you should not call
334    /// `wait()` or `test()` again. A test that fails with an error still marks
335    /// the request completed when MPI completed it (e.g. a truncated
336    /// receive), so a later `wait` or the scope does not complete it a second
337    /// time.
338    #[inline]
339    pub fn test(&mut self) -> Result<Option<Status>> {
340        if self.completed {
341            return Ok(Some(Status::EMPTY));
342        }
343        let mut status = ffi::FerrompiStatus::default();
344        Ok(self
345            .test_raw(&mut status)?
346            .then(|| Status::from_ffi(status)))
347    }

349    /// The body of [`test`](Request::test) for a request that is not yet
350    /// completed, out of line like [`wait_raw`](Request::wait_raw). Returns
351    /// whether MPI completed it.
352    #[inline(never)]
353    fn test_raw(&mut self, status: &mut ffi::FerrompiStatus) -> Result<bool> {
354        let mut flag: i32 = 0;
355        // SAFETY: self.handle is a valid MPI request handle registered in the
356        // C-side request table; self.completed was false on entry (checked by
357        // test), so MPI_Test has not already consumed this handle. flag is a
358        // local out-parameter written before this function reads it below;
359        // status is a valid out-parameter.
360        let ret = unsafe { ffi::ferrompi_test(self.handle, &mut flag, status) };
361        // Set completed from flag BEFORE the `?` below: ferrompi_test frees
362        // the slot and reports flag=1 whenever MPI completed the request,
363        // even when it returns an error (e.g. MPI_ERR_TRUNCATE), so a later
364        // `wait`, or the final wait of the scope or epoch that owns the
365        // request, must not wait on that now-freed handle.
366        if flag != 0 {
367            self.completed = true;
368            self.release();
369        }
370        Error::check_with_op(ret, "test")?;
371        Ok(flag != 0)
372    }

374    /// Wait for any one request in a collection to complete.
375    ///
376    /// Blocks until at least one not-yet-completed request completes and
377    /// returns its index and [`Status`]. Completed entries are skipped;
378    /// `wait_any` returns `Ok(None)` once every entry in the slice is completed
379    /// (or the slice was empty), which is what lets the standard MPI Waitany
380    /// loop idiom — calling `wait_any` repeatedly on the same slice without
381    /// removing completed entries — terminate.
382    ///
383    /// For a receive the `Status` holds the matched source, tag and element
384    /// count. For a send, collective or RMA request the `Status` is the empty
385    /// status, and so is a cancelled receive; only `error` has meaning, and
386    /// this call leaves it `None`.
387    ///
388    /// The completed `Request` is marked completed in place. Removing it
389    /// from the vector is optional, not required for correctness.
390    ///
391    /// On a failed request, the returned error carries that request's own
392    /// class and code. When the MPI library reports which request failed
393    /// (MPICH and Open MPI do), the message also ends with `(request N)`,
394    /// `N` being its index in `requests`.
395    pub fn wait_any(requests: &mut [Request<'_>]) -> Result<Option<(usize, Status)>> {
396        if requests.is_empty() {
397            return Ok(None);
398        }
399        let mut index: i32 = 0;
400        let mut status = ffi::FerrompiStatus::default();
401        // SAFETY: with_handles provides a valid, contiguous [i64] of the request
402        // handles and a same-length [u8] done buffer, both sized to the count we
403        // pass; index and status are valid stack-allocated output parameters.
404        let ret = with_handles(
405            requests,
406            handle_or_skip,
407            |handles, done| unsafe {
408                ffi::ferrompi_waitany(
409                    handles.len() as i64,
410                    handles.as_ptr(),
411                    &mut index,
412                    done.as_mut_ptr(),
413                    &mut status,
414                )
415            },
416            mark_completed,
417        );
418        check_batch(ret, "waitany", index as i64)?;
419        if index < 0 {
420            return Ok(None);
421        }
422        Ok(Some((index as usize, Status::from_ffi(status))))
423    }

425    /// Wait until at least one request in a collection completes.
426    ///
427    /// Returns each completed request's index in `requests` and [`Status`], in
428    /// completion order. A request that failed has `error: Some(class)`, and
429    /// the others in the same call still complete and are reported. `Err` is
430    /// returned only when the call itself fails. Returns `Ok(vec![])` when no
431    /// requests were active (all null, all already completed, or a mix of the
432    /// two).
433    ///
434    /// For a receive the `Status` holds the matched source, tag and element
435    /// count. For an entry with `error: Some(_)` they are as MPI reported them,
436    /// defined when the receive matched (a `Truncate`, say); a failed receive
437    /// whose source MPI left at `MPI_ANY_SOURCE` reads `Source::ProcNull`. For a
438    /// send, collective or RMA request it is the empty status, and so is a
439    /// cancelled receive; only `error` has meaning.
440    ///
441    /// Completed entries are skipped. The completed `Request`s are marked
442    /// completed in place. Removing them from the vector is optional, not
443    /// required for correctness.
444    pub fn wait_some(requests: &mut [Request<'_>]) -> Result<Vec<(usize, Status)>> {
445        if requests.is_empty() {
446            return Ok(vec![]);
447        }
448        let len = requests.len();
449        let mut outcount: i64 = 0;
450        let (ret, completed) = with_handles(
451            requests,
452            handle_or_skip,
453            |handles, done| {
454                with_index_buf(len, |indices, statuses| {
455                    // SAFETY: with_handles / with_index_buf supply valid,
456                    // appropriately-sized [i64] handle, [u8] done, [i32] index
457                    // and status buffers whose lengths match `count`; outcount
458                    // is a valid stack-allocated output parameter.
459                    let ret = unsafe {
460                        ffi::ferrompi_waitsome(
461                            handles.len() as i64,
462                            handles.as_ptr(),
463                            &mut outcount,
464                            indices.as_mut_ptr(),
465                            done.as_mut_ptr(),
466                            statuses.as_mut_ptr(),
467                        )
468                    };
469                    // outcount == -1 means all null or a rejected result.
470                    (ret, some_results(ret, outcount, indices, statuses))
471                })
472            },
473            mark_completed,
474        );
475        Error::check_with_op(ret, "waitsome")?;
476        Ok(completed)
477    }

499        if requests.is_empty() {
500            return Ok(None);
501        }
502        let mut index: i32 = 0;
503        let mut flag: i32 = 0;
504        let mut status = ffi::FerrompiStatus::default();
505        // SAFETY: with_handles provides a valid, contiguous [i64] of the request
506        // handles and a same-length [u8] done buffer, both sized to the count we
507        // pass; index, flag and status are valid stack-allocated output

589    /// Non-destructive query: check whether this request has completed
590    /// without consuming it. Unlike [`test`](Request::test), this does NOT
591    /// free the request on completion; it only probes.
592    ///
593    /// Returns `Ok(true)` if the MPI runtime reports the request is complete,
594    /// `Ok(false)` otherwise. Does NOT mutate `completed` — this is a probe,
595    /// not a commit.
596    pub fn get_status(&self) -> Result<bool> {
597        if self.completed {
598            return Ok(true);
599        }
600        let mut flag: i32 = 0;
601        // SAFETY: self.handle is a valid request handle issued by the C shim.
602        // flag is a valid stack-allocated i32 output parameter.
603        let ret = unsafe { ffi::ferrompi_request_get_status(self.handle, &mut flag) };
604        Error::check_with_op(ret, "request_get_status")?;
605        Ok(flag != 0)
606    }

608    /// Request cancellation of a pending point-to-point operation.
609    ///
610    /// # Errors
611    ///
612    /// Returns [`Error::NotSupported`] without calling into MPI when `self`
613    /// is a nonblocking-collective or RMA request: the MPI standard defines
614    /// `MPI_Cancel` only for point-to-point requests, and both MPICH and
615    /// Open MPI reject it for other request kinds. The request is left
616    /// pending; follow up with [`wait`](Request::wait) as usual.
617    ///
618    /// # Portability
619    ///
620    /// Per the MPI 4.0 standard, `MPI_Cancel` is effectively deprecated for
621    /// send requests. Open MPI refuses to cancel sends; MPICH may report
622    /// success but not actually cancel the send. Cancellation reliably works
623    /// only for receives.
624    ///
625    /// # Usage
626    ///
627    /// `cancel` does NOT complete the request. The caller must follow up with
628    /// [`wait`](Request::wait) to reclaim the handle:
629    ///
630    /// ```no_run
631    /// # use ferrompi::{Mpi, Result};
632    /// # fn main() -> Result<()> {
633    /// # let mpi = Mpi::init()?;
634    /// # let world = mpi.world();
635    /// # let mut buf = vec![0u8; 10];
636    /// ferrompi::scope(|s| {
637    ///     let mut req = world.irecv(s, &mut buf, 0, 0)?;
638    ///     req.cancel()?;
639    ///     req.wait()?;
640    ///     Ok(())
641    /// })?;
642    /// # Ok(()) }
643    /// ```
644    pub fn cancel(&mut self) -> Result<()> {
645        match self.kind {
646            RequestKind::PointToPoint => {}
647            RequestKind::Collective => {
648                return Err(Error::NotSupported(
649                    "cancel of a nonblocking collective request".into(),
650                ));
651            }
652            #[cfg(feature = "rma")]
653            RequestKind::Rma => {
654                return Err(Error::NotSupported("cancel of an RMA request".into()));
655            }
656        }
657        if self.completed {
658            return Ok(());
659        }
660        // SAFETY: self.handle is a valid request handle issued by the C shim.
661        let ret = unsafe { ffi::ferrompi_cancel(self.handle) };
662        Error::check_with_op(ret, "cancel")
663    }

665    /// Wait for all requests in a slice to complete.
666    ///
667    /// This is more efficient than waiting for each request individually.
668    ///
669    /// Takes the requests by `&mut [Request]` (rather than consuming a
670    /// `Vec<Request>`) so a caller can reuse one backing buffer across a drain
671    /// loop, mirroring
672    /// [`PersistentRequest::wait_all`](crate::PersistentRequest::wait_all).
673    /// Whatever the result, every request MPI completed — one that completed
674    /// with an error included — is marked completed in place; the others
675    /// stay pending. The same policy applies to `PersistentRequest::wait_all`.
676    ///
677    /// On a failed request, the returned error carries that request's own
678    /// class and code, and its message ends with `(request N)`, `N` being
679    /// its index in `requests`.
680    pub fn wait_all(requests: &mut [Request<'_>]) -> Result<()> {
681        if requests.is_empty() {
682            return Ok(());
683        }
684
685        let mut failed: i64 = -1;
686        // SAFETY: with_handles provides a valid, contiguous [i64] of the request
687        // handles and a same-length [u8] done buffer, both sized to the count we
688        // pass; failed is a valid stack-allocated i64 output parameter.
689        let ret = with_handles(
690            requests,
691            handle_or_skip,
692            |handles, done| unsafe {
693                ffi::ferrompi_waitall(
694                    handles.len() as i64,
695                    handles.as_ptr(),
696                    done.as_mut_ptr(),
697                    &mut failed,
698                )
699            },
700            mark_completed,
701        );
702
703        check_batch(ret, "waitall", failed)
704    }
705}

878    /// Called by the one test that initializes MPI. A handle kept past its
879    /// request's completion is rejected by the C layer, and does not act on the
880    /// request that reused the slot.
881    pub(crate) fn stale_handle_is_rejected(world: &Communicator) {
882        let me = world.rank();
883        let mut a_buf = [0u8; 4];
884        let mut b_buf = [0u8; 4];
885        crate::scope(|s| {
886            let a = world.irecv(s, &mut a_buf, me, 31)?;
887            let stale = a.handle;
888            world.send(&[31u8; 4], me, 31)?;
889            a.wait()?;
890
891            let b = world.irecv(s, &mut b_buf, me, 32)?;
892            // Sent before either assertion below can fail: the scope waits for
893            // `b` while it unwinds, and `b` completes only if its message exists.
894            world.send(&[32u8; 4], me, 32)?;
895            assert!(
896                (b.handle & 0xffff_ffff) == (stale & 0xffff_ffff) && b.handle != stale,
897                "slot not reused with a new generation: {stale:#x} then {:#x}",
898                b.handle
899            );
900            let mut flag: i32 = 0;
901            // SAFETY: `flag` is a valid out-parameter and a null status is
902            // MPI_STATUS_IGNORE; the stale handle must be refused by the lookup.
903            let ret = unsafe { crate::ffi::ferrompi_test(stale, &mut flag, std::ptr::null_mut()) };
904            assert!(
905                matches!(
906                    Error::from_code(ret),
907                    Error::Mpi {
908                        class: MpiErrorClass::Request,
909                        ..
910                    }
911                ),
912                "test(stale) returned {ret}"
913            );
914            b.wait()?;
915            Ok(())
916        })
917        .expect("scope failed");
918        assert_eq!(b_buf, [32u8; 4]);
919    }

921    /// Called by the one test that initializes MPI. An `rput` request the
922    /// closure never waits is completed by its epoch, whatever the closure does:
923    /// every epoch then gets the first epoch's table slot again.
924    #[cfg(feature = "rma")]
925    pub(crate) fn epoch_completes_pending_rput(world: &Communicator) {
926        use crate::{LockType, Win};
927        use std::panic::{catch_unwind, set_hook, take_hook, AssertUnwindSafe};
928
929        const ROUNDS: usize = 300;
930        let me = world.rank();
931        let mut win = Win::<i32>::allocate(world, 4).expect("Win::allocate failed");
932        let buf = [10i32, 20, 30, 40];
933        for exit in 0..3 {
934            let mut slots = Vec::with_capacity(ROUNDS);
935            let mut ended_as_asked = true;
936            // Silent only while the closures run (exit 2 panics once per round).
937            // Nothing is asserted until the default hook is back, so a failed
938            // assertion prints its message.
939            set_hook(Box::new(|_| {}));
940            for _ in 0..ROUNDS {
941                let ended = catch_unwind(AssertUnwindSafe(|| {
942                    win.lock(LockType::Exclusive, me, |e| {
943                        let req = e.rput(&buf, me, 0)?;
944                        slots.push(req.handle as u32);
945                        match exit {
946                            0 => Ok(()),
947                            // Rank 99 is outside the window's communicator.
948                            1 => e.rput(&buf, 99, 0).map(|_| ()),
949                            _ => panic!("closure panics with a request pending"),
950                        }
951                    })
952                }));
953                ended_as_asked &= matches!(
954                    (exit, ended),
955                    (0, Ok(Ok(()))) | (1, Ok(Err(_))) | (2, Err(_))
956                );
957            }
958            drop(take_hook());
959            assert!(ended_as_asked, "exit {exit}: an epoch did not end as asked");
960            assert!(
961                slots.len() == ROUNDS && slots.iter().all(|&slot| slot == slots[0]),
962                "exit {exit}: slots {:?}",
963                &slots[..slots.len().min(8)]
964            );
965        }
966    }
