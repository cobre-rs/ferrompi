1//! MPI datatype trait and type tag mapping.
2//!
3//! This module provides three sealed traits:
4//!
5//! - [`MpiDatatype`]: maps primitive Rust types to MPI datatype tags for all
6//!   communication operations (broadcast, send, recv, allreduce with scalar ops).
7//! - [`MpiIndexedDatatype`]: marks the six MPI paired value+index structs that are
8//!   only valid with [`MPI_MAXLOC`/`MPI_MINLOC`](crate::ReduceOp::MaxLoc) reductions.
9//! - [`BytePermutable`]: marks types safe for byte-level bitwise reductions via
10//!   `MPI_BYTE` (`BitwiseOr`, `BitwiseAnd`, `BitwiseXor`).
11//!
12//! # Primitive Types (`MpiDatatype`)
13//!
14//! | Rust Type | MPI Equivalent     | Tag Value |
15//! |-----------|-------------------|-----------|
16//! | `f32`     | `MPI_FLOAT`       | 0         |
17//! | `f64`     | `MPI_DOUBLE`      | 1         |
18//! | `i32`     | `MPI_INT32_T`     | 2         |
19//! | `i64`     | `MPI_INT64_T`     | 3         |
20//! | `u8`      | `MPI_UINT8_T`     | 4         |
21//! | `u32`     | `MPI_UINT32_T`    | 5         |
22//! | `u64`     | `MPI_UINT64_T`    | 6         |
23//!
24//! # Paired Value+Index Types (`MpiIndexedDatatype`)
25//!
26//! | Rust Struct      | MPI Equivalent        | Tag Value |
27//! |------------------|-----------------------|-----------|
28//! | [`FloatInt`]     | `MPI_FLOAT_INT`       | 7         |
29//! | [`DoubleInt`]    | `MPI_DOUBLE_INT`      | 8         |
30//! | [`LongInt`] (Linux x86_64/aarch64/ppc64le only) | `MPI_LONG_INT` | 9 |
31//! | [`Int2`]         | `MPI_2INT`            | 10        |
32//! | [`ShortInt`]     | `MPI_SHORT_INT`       | 11        |
33//! | [`LongDoubleInt`] (Linux x86_64/aarch64/ppc64le only) | `MPI_LONG_DOUBLE_INT` | 12 |
34//!
35//! # Byte-Permutable Types (`BytePermutable`)
36//!
37//! | Rust Type       | MPI Equivalent | Tag Value |
38//! |-----------------|----------------|-----------|
39//! | u8-like bytes   | `MPI_BYTE`     | 13        |

69/// Tag values matching C-side `FERROMPI_*` defines.
70///
71/// These discriminants must stay in sync with the `#define FERROMPI_*` values
72/// in `csrc/ferrompi.h`.
73#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
74#[repr(i32)]
75pub enum DatatypeTag {
76    /// 32-bit floating point (`MPI_FLOAT`)
77    F32 = 0,
78    /// 64-bit floating point (`MPI_DOUBLE`)
79    F64 = 1,
80    /// 32-bit signed integer (`MPI_INT32_T`)
81    I32 = 2,
82    /// 64-bit signed integer (`MPI_INT64_T`)
83    I64 = 3,
84    /// 8-bit unsigned integer (`MPI_UINT8_T`)
85    U8 = 4,
86    /// 32-bit unsigned integer (`MPI_UINT32_T`)
87    U32 = 5,
88    /// 64-bit unsigned integer (`MPI_UINT64_T`)
89    U64 = 6,
90    /// Paired `{ f32 value; i32 index }` for `MPI_FLOAT_INT` (MAXLOC/MINLOC)
91    FloatInt = 7,
92    /// Paired `{ f64 value; i32 index }` for `MPI_DOUBLE_INT` (MAXLOC/MINLOC)
93    DoubleInt = 8,
94    /// Paired `{ i64 value; i32 index }` for `MPI_LONG_INT` (MAXLOC/MINLOC).
95    /// Note: on most 64-bit platforms `long` is 8 bytes, matching `i64`.
96    /// The matching `LongInt` struct exists only on Linux x86_64, aarch64
97    /// and little-endian powerpc64.
98    LongInt = 9,
99    /// Paired `{ i32 value; i32 index }` for `MPI_2INT` (MAXLOC/MINLOC)
100    Int2 = 10,
101    /// Paired `{ i16 value; i32 index }` for `MPI_SHORT_INT` (MAXLOC/MINLOC)
102    ShortInt = 11,
103    /// Paired `{ f128-equivalent value; i32 index }` for `MPI_LONG_DOUBLE_INT`
104    /// (MAXLOC/MINLOC). Uses `[u8; 16]` on x86_64 where `long double` is 80-bit
105    /// extended precision stored in 16 bytes.
106    /// The matching `LongDoubleInt` struct exists only on Linux x86_64,
107    /// aarch64 and little-endian powerpc64.
108    LongDoubleInt = 12,
109    /// Opaque 1-byte unit (`MPI_BYTE`) for type-erased bitwise reductions.
110    ///
111    /// Used exclusively with [`BytePermutable`]-bounded types and
112    /// [`Communicator::allreduce_bytes`](crate::Communicator::allreduce_bytes). The count passed to MPI is the
113    /// total byte count (`element_count * size_of::<T>()`), so MPI treats
114    /// the buffer as a flat array of bytes.
115    Byte = 13,
116}

118/// Trait for types that can be used in MPI communication operations.
119///
120/// This is a **sealed trait** — it cannot be implemented outside this crate.
121/// Supported types: [`f32`], [`f64`], [`i32`], [`i64`], [`u8`], [`u32`], [`u64`].
122///
123/// # Example
124///
125/// ```no_run
126/// use ferrompi::{Mpi, MpiDatatype};
127///
128/// let mpi = Mpi::init().unwrap();
129/// let world = mpi.world();
130///
131/// // Works with f64
132/// let mut data_f64 = vec![1.0f64; 10];
133/// world.broadcast(&mut data_f64, 0).unwrap();
134///
135/// // Works with i32
136/// let mut data_i32 = vec![42i32; 10];
137/// world.broadcast(&mut data_i32, 0).unwrap();
138/// ```
139pub trait MpiDatatype: sealed::Sealed + Copy + Send + 'static {
140    /// The datatype tag used for FFI dispatch to the C layer.
141    const TAG: DatatypeTag;
142}

144/// Bound for the element type of the custom-datatype point-to-point methods
145/// ([`Communicator::send_custom`](crate::Communicator::send_custom),
146/// [`recv_custom`](crate::Communicator::recv_custom),
147/// [`isend_custom`](crate::Communicator::isend_custom) and
148/// [`irecv_custom`](crate::Communicator::irecv_custom)).
149///
150/// # Safety
151///
152/// Implementors guarantee that every bit pattern of `size_of::<Self>()`
153/// bytes is a valid `Self`. The type must therefore have no references, no
154/// raw pointers that MPI's write could invalidate, no `Box`, `Vec`, `bool`,
155/// `char`, enum, or `NonZero*` fields. Padding bytes are allowed. The type
156/// should be `#[repr(C)]` so its layout matches the datatype built for it.
157///
158/// This trait is blanket-implemented for every [`MpiDatatype`] and for
159/// fixed-size arrays `[T; N]` of any `PlainData` type.
160///
161/// # Example
162///
163/// ```compile_fail
164/// # use ferrompi::{CustomDatatype, DatatypeTag, Mpi};
165/// # let _mpi = Mpi::init().unwrap();
166/// # let world = _mpi.world();
167/// let dt8 = CustomDatatype::contiguous(8, DatatypeTag::U8).unwrap();
168/// let mut boxes: [Box<u64>; 1] = [Box::new(5)];
169/// world.recv_custom(&mut boxes, &dt8, 0, 2).unwrap();
170/// ```
171///
172/// ```no_run
173/// # use ferrompi::{CustomDatatype, DatatypeTag, Mpi, PlainData};
174/// #[repr(C)]
175/// #[derive(Clone, Copy)]
176/// struct Pair { v: f64, i: i32 }
177/// // SAFETY: Pair is #[repr(C)] of an f64 and an i32, so any bit pattern is valid.
178/// unsafe impl PlainData for Pair {}
179/// # let _mpi = Mpi::init().unwrap();
180/// # let world = _mpi.world();
181/// let dt8 = CustomDatatype::contiguous(8, DatatypeTag::U8).unwrap();
182/// let mut boxes: [Pair; 1] = [Pair { v: 0.0, i: 0 }];
183/// world.recv_custom(&mut boxes, &dt8, 0, 2).unwrap();
184/// ```
185pub unsafe trait PlainData: Copy + 'static {}
186// SAFETY: every MpiDatatype is a primitive integer or float, and any bit pattern of its size is a valid value.
187unsafe impl<T: MpiDatatype> PlainData for T {}
188// SAFETY: an array of plain data is plain data — its bytes are exactly N back-to-back elements, each valid for any bit pattern.
189unsafe impl<T: PlainData, const N: usize> PlainData for [T; N] {}

191/// Describes a typed slice for MPI FFI as `(pointer, count, tag)`.
192///
193/// The pointer is valid for `s.len()` elements of `T` for as long as the
194/// borrow of `s` lives (for [`buf_mut`], it is also valid for writes and
195/// unaliased). The count is exact: a slice length never exceeds
196/// `isize::MAX <= i64::MAX`. The tag is the MPI datatype that the sealed
197/// `MpiDatatype` impl assigns to `T`, so MPI's element size and layout equal
198/// `T`'s. The pointer is never null, even for an empty slice, so a NULL
199/// passed elsewhere as an in-place marker stays unambiguous.
200pub(crate) fn buf<T: MpiDatatype>(s: &[T]) -> (*const c_void, i64, i32) {
201    (s.as_ptr().cast(), s.len() as i64, T::TAG as i32)
202}
203
204/// See [`buf`].
205pub(crate) fn buf_mut<T: MpiDatatype>(s: &mut [T]) -> (*mut c_void, i64, i32) {
206    (s.as_mut_ptr().cast(), s.len() as i64, T::TAG as i32)
207}

230/// Trait for types that can be used with `MPI_MAXLOC` / `MPI_MINLOC`.
231///
232/// This is a **sealed trait** — it cannot be implemented outside this crate.
233/// Only the six MPI predefined paired value+index types implement it.
234///
235/// Use these types exclusively with [`Communicator::allreduce_indexed`](crate::Communicator::allreduce_indexed) and
236/// [`ReduceOp::MaxLoc`](crate::ReduceOp::MaxLoc) / [`ReduceOp::MinLoc`](crate::ReduceOp::MinLoc). They are **not** valid for
237/// `broadcast`, `send`, `recv`, or other collectives: they implement
238/// [`MpiIndexedDatatype`], not [`MpiDatatype`], which those APIs require.
239///
240/// # Example
241///
242/// ```no_run
243/// use ferrompi::{Mpi, ReduceOp, DoubleInt, MpiIndexedDatatype};
244///
245/// let mpi = Mpi::init().unwrap();
246/// let world = mpi.world();
247/// let rank = world.rank();
248///
249/// let send = [DoubleInt { value: rank as f64, index: rank }];
250/// let mut recv = [DoubleInt { value: 0.0, index: 0 }];
251/// world.allreduce_indexed(&send, &mut recv, ReduceOp::MaxLoc).unwrap();
252/// // Every rank now holds { value: (size-1) as f64, index: size-1 }
253/// ```
254///
255/// # Sealed: cannot implement for external types
256///
257/// ```compile_fail
258/// use ferrompi::MpiIndexedDatatype;
259/// use ferrompi::DatatypeTag;
260///
261/// // This must not compile — MpiIndexedDatatype is sealed.
262/// impl MpiIndexedDatatype for i32 {
263///     const TAG: DatatypeTag = DatatypeTag::Int2;
264/// }
265/// ```
266pub trait MpiIndexedDatatype: sealed_indexed::Sealed + Copy + Send + 'static {
267    /// The datatype tag used for FFI dispatch to the C layer.
268    const TAG: DatatypeTag;
269}

499/// Trait for types eligible for use with `MPI_Compare_and_swap`.
500///
501/// This is a **sealed trait** — it cannot be implemented outside this crate.
502/// Per MPI 4.1 section 12.5.4, the predefined types guaranteed to support
503/// `MPI_Compare_and_swap` are the integer and byte types: `i32`, `i64`,
504/// `u32`, `u64`, and `u8`.
505///
506/// **Floating-point types (`f32`, `f64`) are intentionally excluded.** The
507/// MPI standard does not require implementations to support floating-point
508/// CAS. Accepting them would introduce silent portability bugs across MPI
509/// implementations.
510///
511/// This trait is used as a bound on [`Win::compare_and_swap`](crate::Win::compare_and_swap)
512/// to ensure the method only exists for windows whose element type is
513/// CAS-eligible.
514///
515/// # Sealed: cannot implement for external types
516///
517/// ```compile_fail
518/// fn _check(w: &ferrompi::Win<f64>) -> ferrompi::Result<f64> {
519///     w.compare_and_swap(0.0, 0.0, 0, 0)
520/// }
521/// ```
522#[cfg(feature = "rma")]
523pub trait AtomicMpiDatatype: sealed_atomic::Sealed + Copy + Send + 'static {}
524
525#[cfg(feature = "rma")]
526macro_rules! impl_atomic_mpi_datatype {
527    ($ty:ty) => {
528        impl sealed_atomic::Sealed for $ty {}
529        impl AtomicMpiDatatype for $ty {}
530    };
531}

638    #[test]
639    fn indexed_datatype_struct_layouts() {
640        use std::mem::{align_of, size_of};
641
642        // FloatInt: { f32, i32 } — both 4-byte aligned, no padding
643        // sizeof == 8, alignof == 4
644        assert_eq!(size_of::<FloatInt>(), 8, "FloatInt size");
645        assert_eq!(align_of::<FloatInt>(), 4, "FloatInt align");
646
647        // DoubleInt: { f64, i32 } — f64 is 8-byte aligned, i32 is 4-byte;
648        // 4 bytes trailing padding to align the next array element at 8 bytes.
649        // sizeof == 16, alignof == 8 on x86_64/aarch64 Linux.
650        assert!(
651            size_of::<DoubleInt>() >= 12,
652            "DoubleInt must hold at least f64 + i32"
653        );
654        assert!(
655            align_of::<DoubleInt>() >= 8,
656            "DoubleInt alignment must be at least f64 alignment"
657        );
658        #[cfg(any(target_arch = "x86_64", target_arch = "aarch64"))]
659        {
660            assert_eq!(
661                size_of::<DoubleInt>(),
662                16,
663                "DoubleInt size on x86_64/aarch64"
664            );
665            assert_eq!(
666                align_of::<DoubleInt>(),
667                8,
668                "DoubleInt align on x86_64/aarch64"
669            );
670        }
671
672        // Int2: { i32, i32 } — no padding
673        // sizeof == 8, alignof == 4
674        assert_eq!(size_of::<Int2>(), 8, "Int2 size");
675        assert_eq!(align_of::<Int2>(), 4, "Int2 align");
676
677        // ShortInt: { i16, i32 } — 2 bytes padding between fields for i32 alignment
678        // sizeof == 8, alignof == 4
679        assert!(
680            size_of::<ShortInt>() >= 6,
681            "ShortInt must hold at least i16 + i32"
682        );
683        assert!(
684            align_of::<ShortInt>() >= 4,
685            "ShortInt alignment must be at least i32 alignment"
686        );
687        #[cfg(any(target_arch = "x86_64", target_arch = "aarch64"))]
688        {
689            assert_eq!(size_of::<ShortInt>(), 8, "ShortInt size on x86_64/aarch64");
690            assert_eq!(
691                align_of::<ShortInt>(),
692                4,
693                "ShortInt align on x86_64/aarch64"
694            );
695        }
696    }
