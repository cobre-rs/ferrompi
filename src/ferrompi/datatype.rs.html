1//! MPI datatype trait and type tag mapping.
2//!
3//! This module provides three sealed traits:
4//!
5//! - [`MpiDatatype`]: maps primitive Rust types to MPI datatype tags for all
6//!   communication operations (broadcast, send, recv, allreduce with scalar ops).
7//! - [`MpiIndexedDatatype`]: marks the six MPI paired value+index structs that are
8//!   only valid with [`MPI_MAXLOC`/`MPI_MINLOC`](crate::CollectiveOp::MAX_LOC) reductions.
9//! - [`BytePermutable`]: marks types safe for byte-level bitwise reductions via
10//!   `MPI_BYTE` (`BitwiseOr`, `BitwiseAnd`, `BitwiseXor`).
11//!
12//! # Primitive Types (`MpiDatatype`)
13//!
14//! | Rust Type | MPI Equivalent     | Tag Value |
15//! |-----------|-------------------|-----------|
16//! | `f32`     | `MPI_FLOAT`       | 0         |
17//! | `f64`     | `MPI_DOUBLE`      | 1         |
18//! | `i32`     | `MPI_INT32_T`     | 2         |
19//! | `i64`     | `MPI_INT64_T`     | 3         |
20//! | `u8`      | `MPI_UINT8_T`     | 4         |
21//! | `u32`     | `MPI_UINT32_T`    | 5         |
22//! | `u64`     | `MPI_UINT64_T`    | 6         |
23//!
24//! # Paired Value+Index Types (`MpiIndexedDatatype`)
25//!
26//! | Rust Struct      | MPI Equivalent        | Tag Value |
27//! |------------------|-----------------------|-----------|
28//! | [`FloatInt`]     | `MPI_FLOAT_INT`       | 7         |
29//! | [`DoubleInt`]    | `MPI_DOUBLE_INT`      | 8         |
30//! | [`LongInt`] (Linux x86_64/aarch64/ppc64le only) | `MPI_LONG_INT` | 9 |
31//! | [`Int2`]         | `MPI_2INT`            | 10        |
32//! | [`ShortInt`]     | `MPI_SHORT_INT`       | 11        |
33//! | [`LongDoubleInt`] (Linux x86_64/aarch64/ppc64le only) | `MPI_LONG_DOUBLE_INT` | 12 |
34//!
35//! # Byte-Permutable Types (`BytePermutable`)
36//!
37//! | Rust Type       | MPI Equivalent | Tag Value |
38//! |-----------------|----------------|-----------|
39//! | u8-like bytes   | `MPI_BYTE`     | 13        |

81/// Tag values matching C-side `FERROMPI_*` defines.
82///
83/// These discriminants must stay in sync with the `#define FERROMPI_*` values
84/// in `csrc/ferrompi.h`.
85#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
86#[repr(i32)]
87pub enum DatatypeTag {
88    /// 32-bit floating point (`MPI_FLOAT`)
89    F32 = 0,
90    /// 64-bit floating point (`MPI_DOUBLE`)
91    F64 = 1,
92    /// 32-bit signed integer (`MPI_INT32_T`)
93    I32 = 2,
94    /// 64-bit signed integer (`MPI_INT64_T`)
95    I64 = 3,
96    /// 8-bit unsigned integer (`MPI_UINT8_T`)
97    U8 = 4,
98    /// 32-bit unsigned integer (`MPI_UINT32_T`)
99    U32 = 5,
100    /// 64-bit unsigned integer (`MPI_UINT64_T`)
101    U64 = 6,
102    /// Paired `{ f32 value; i32 index }` for `MPI_FLOAT_INT` (MAXLOC/MINLOC)
103    FloatInt = 7,
104    /// Paired `{ f64 value; i32 index }` for `MPI_DOUBLE_INT` (MAXLOC/MINLOC)
105    DoubleInt = 8,
106    /// Paired `{ i64 value; i32 index }` for `MPI_LONG_INT` (MAXLOC/MINLOC).
107    /// Note: on most 64-bit platforms `long` is 8 bytes, matching `i64`.
108    /// The matching `LongInt` struct exists only on Linux x86_64, aarch64
109    /// and little-endian powerpc64.
110    LongInt = 9,
111    /// Paired `{ i32 value; i32 index }` for `MPI_2INT` (MAXLOC/MINLOC)
112    Int2 = 10,
113    /// Paired `{ i16 value; i32 index }` for `MPI_SHORT_INT` (MAXLOC/MINLOC)
114    ShortInt = 11,
115    /// Paired `{ f128-equivalent value; i32 index }` for `MPI_LONG_DOUBLE_INT`
116    /// (MAXLOC/MINLOC). Uses `[u8; 16]` on x86_64 where `long double` is 80-bit
117    /// extended precision stored in 16 bytes.
118    /// The matching `LongDoubleInt` struct exists only on Linux x86_64,
119    /// aarch64 and little-endian powerpc64.
120    LongDoubleInt = 12,
121    /// Opaque 1-byte unit (`MPI_BYTE`) for type-erased bitwise reductions.
122    ///
123    /// Used exclusively with [`BytePermutable`]-bounded types and
124    /// [`Communicator::allreduce_bytes`](crate::Communicator::allreduce_bytes). The count passed to MPI is the
125    /// total byte count (`element_count * size_of::<T>()`), so MPI treats
126    /// the buffer as a flat array of bytes.
127    Byte = 13,
128}

130/// Trait for types that can be used in MPI communication operations.
131///
132/// This is a **sealed trait** — it cannot be implemented outside this crate.
133/// Supported types: [`f32`], [`f64`], [`i32`], [`i64`], [`u8`], [`u32`], [`u64`].
134///
135/// # Example
136///
137/// ```no_run
138/// use ferrompi::{Mpi, MpiDatatype};
139///
140/// let mpi = Mpi::init().unwrap();
141/// let world = mpi.world();
142///
143/// // Works with f64
144/// let mut data_f64 = vec![1.0f64; 10];
145/// world.broadcast(&mut data_f64, 0).unwrap();
146///
147/// // Works with i32
148/// let mut data_i32 = vec![42i32; 10];
149/// world.broadcast(&mut data_i32, 0).unwrap();
150/// ```
151///
152/// # The datatype code is private
153///
154/// The code the C layer dispatches on has no public name:
155///
156/// ```compile_fail,E0432
157/// use ferrompi::DatatypeTag;
158/// ```
159///
160/// ```no_run
161/// use ferrompi::MpiDatatype;
162/// ```
163///
164/// and it cannot be read through a bound on this trait:
165///
166/// ```compile_fail,E0599
167/// fn f<T: ferrompi::MpiDatatype>() {
168///     let _ = T::TAG;
169/// }
170/// ```
171///
172/// ```no_run
173/// fn f<T: ferrompi::MpiDatatype>() {
174///     let _ = ferrompi::CustomDatatype::contiguous::<T>(1);
175/// }
176/// ```
177pub trait MpiDatatype: sealed::Sealed + Copy + Send + 'static {}
178
179/// The tag the C layer dispatches on for `T`.
180#[inline]
181pub(crate) fn tag_of<T: MpiDatatype>() -> DatatypeTag {
182    T::tag(sealed::Token)
183}

186#[inline]

191/// Bound for the element type of the custom-datatype point-to-point methods
192/// ([`Communicator::send_custom`](crate::Communicator::send_custom),
193/// [`recv_custom`](crate::Communicator::recv_custom),
194/// [`isend_custom`](crate::Communicator::isend_custom) and
195/// [`irecv_custom`](crate::Communicator::irecv_custom)).
196///
197/// # Safety
198///
199/// Implementors guarantee that every bit pattern of `size_of::<Self>()`
200/// bytes is a valid `Self`. The type must therefore have no references, no
201/// raw pointers that MPI's write could invalidate, no `Box`, `Vec`, `bool`,
202/// `char`, enum, or `NonZero*` fields. Padding bytes are allowed. The type
203/// should be `#[repr(C)]` so its layout matches the datatype built for it.
204///
205/// This trait is blanket-implemented for every [`MpiDatatype`] and for
206/// fixed-size arrays `[T; N]` of any `PlainData` type.
207///
208/// # Example
209///
210/// ```compile_fail,E0277
211/// # use ferrompi::{CustomDatatype, Mpi};
212/// # let _mpi = Mpi::init().unwrap();
213/// # let world = _mpi.world();
214/// let dt8 = CustomDatatype::contiguous::<u8>(8).unwrap();
215/// let mut boxes: [Box<u64>; 1] = [Box::new(5)];
216/// world.recv_custom(&mut boxes, &dt8, 0, 2).unwrap();
217/// ```
218///
219/// ```no_run
220/// # use ferrompi::{CustomDatatype, Mpi, PlainData};
221/// #[repr(C)]
222/// #[derive(Clone, Copy)]
223/// struct Pair { v: f64, i: i32 }
224/// // SAFETY: Pair is #[repr(C)] of an f64 and an i32, so any bit pattern is valid.
225/// unsafe impl PlainData for Pair {}
226/// # let _mpi = Mpi::init().unwrap();
227/// # let world = _mpi.world();
228/// let dt8 = CustomDatatype::contiguous::<u8>(8).unwrap();
229/// let mut boxes: [Pair; 1] = [Pair { v: 0.0, i: 0 }];
230/// world.recv_custom(&mut boxes, &dt8, 0, 2).unwrap();
231/// ```
232pub unsafe trait PlainData: Copy + 'static {}
233// SAFETY: every MpiDatatype is a primitive integer or float, and any bit pattern of its size is a valid value.
234unsafe impl<T: MpiDatatype> PlainData for T {}
235// SAFETY: an array of plain data is plain data — its bytes are exactly N back-to-back elements, each valid for any bit pattern.
236unsafe impl<T: PlainData, const N: usize> PlainData for [T; N] {}

238/// Describes a typed slice for MPI FFI as `(pointer, count, tag)`.
239///
240/// The pointer is valid for `s.len()` elements of `T` for as long as the
241/// borrow of `s` lives (for [`buf_mut`], it is also valid for writes and
242/// unaliased). The count is exact: a slice length never exceeds
243/// `isize::MAX <= i64::MAX`. The tag is the MPI datatype that the sealed
244/// `MpiDatatype` impl assigns to `T`, so MPI's element size and layout equal
245/// `T`'s. The pointer is never null, so a NULL passed elsewhere as an in-place
246/// marker stays unambiguous; see [`buf_ptr`] for an empty slice.
247// PERF: with no inline attribute on `buf` and `buf_mut` (and `#[inline]` on
248// `buf_ptr`), `isend` and `irecv` are no longer inlined into a caller's scope
249// body.
250#[inline(always)]
251pub(crate) fn buf<T: MpiDatatype>(s: &[T]) -> (*const c_void, i64, i32) {
252    (
253        buf_ptr(s.as_ptr(), s.len(), false),
254        s.len() as i64,
255        tag_of::<T>() as i32,
256    )
257}
258
259/// See [`buf`].
260#[inline(always)]
261pub(crate) fn buf_mut<T: MpiDatatype>(s: &mut [T]) -> (*mut c_void, i64, i32) {
262    (
263        buf_ptr(s.as_mut_ptr(), s.len(), true).cast_mut(),
264        s.len() as i64,
265        tag_of::<T>() as i32,
266    )
267}

269/// The addresses MPI gets in place of an empty buffer's own: the first for a
270/// buffer MPI reads, the second for one it writes, so that two substituted
271/// buffers of one call never share an address. [`buf_ptr`] and
272/// [`empty_buf_ptr`] substitute them.
273static EMPTY: [u64; 2] = [0; 2];
274
275/// `ptr`, or the address of a private static when the buffer holds no bytes
276/// (`len` is zero or `T` is zero-sized) and `T` has alignment 1. Such a buffer
277/// usually points at address 1, which is `MPI_IN_PLACE` in Open MPI and in the
278/// MPI-5 ABI; a type with a larger alignment has an even address, and
279/// `MPI_IN_PLACE` is an odd address there and in MPICH. `written` selects the
280/// second address, so that two substituted buffers of one call never share
281/// one; [`empty_buf_ptr`] names the calls that need that. MPI reads and writes
282/// nothing through a buffer of count zero.
283#[inline(always)]
284pub(crate) fn buf_ptr<T>(ptr: *const T, len: usize, written: bool) -> *const c_void {
285    if align_of::<T>() == 1 && (len == 0 || size_of::<T>() == 0) {
286        std::ptr::from_ref(&EMPTY[usize::from(written)]).cast()
287    } else {
288        ptr.cast()
289    }
290}
291
292/// `ptr`, or, when `empty`, an address of [`EMPTY`] that is not `other`, the
293/// pointer of the call's other buffer: the first for a buffer MPI reads and the
294/// second (`written`) for one it writes, or the remaining one of the two when
295/// `other` is that address. It is [`buf_ptr`]'s substitution for an element
296/// type of any alignment.
297///
298/// [`buf_ptr`] leaves an empty buffer of a type with alignment 2 or more at its
299/// own address: a test on the length would cost the blocking collectives a
300/// branch. That address can be the other buffer's: two empty `Vec`s usually
301/// share one dangling address, and an empty slice can start where the other
302/// buffer begins. MPICH's `MPI_Alltoallv` and `MPI_Ialltoallv` can refuse equal
303/// send and receive buffers whatever the counts, and so can Open MPI 5's
304/// `MPI_Reduce_init` at the root. So `alltoallv` and `ialltoallv` pass each of
305/// their two pointers through here, `empty` when that buffer is, and
306/// `reduce_init`, whose two `Vec`s share an address only when both are empty,
307/// its receive pointer, `empty` when both are. MPI reads and writes nothing
308/// through a buffer of count zero.
309///
310/// MPICH's `MPI_Alltoallv_init` makes the same test. `alltoallv_init` needs no
311/// call here: it hands MPI its own end-to-end copy of the four arrays, in which
312/// the send and the receive counts never share an address.
313///
314/// `other` is needed because the two addresses are not this crate's alone: a
315/// build with link-time optimization can place a caller's read-only constant
316/// of sixteen zero bytes where [`EMPTY`] is, and a send buffer can then start
317/// at the second address.
318#[inline]
319pub(crate) fn empty_buf_ptr(
320    ptr: *const c_void,
321    empty: bool,
322    written: bool,
323    other: *const c_void,
324) -> *const c_void {
325    let fixed = |second: bool| std::ptr::from_ref(&EMPTY[usize::from(second)]).cast::<c_void>();
326    if !empty {
327        ptr
328    } else if fixed(written) == other {
329        fixed(!written)
330    } else {
331        fixed(written)
332    }
333}

359/// Trait for types that can be used with `MPI_MAXLOC` / `MPI_MINLOC`.
360///
361/// This is a **sealed trait** — it cannot be implemented outside this crate.
362/// Only the six MPI predefined paired value+index types implement it.
363///
364/// Use these types exclusively with [`Communicator::allreduce_indexed`](crate::Communicator::allreduce_indexed) and
365/// [`CollectiveOp::MAX_LOC`](crate::CollectiveOp::MAX_LOC) / [`CollectiveOp::MIN_LOC`](crate::CollectiveOp::MIN_LOC). They are **not** valid for
366/// `broadcast`, `send`, `recv`, or other collectives: they implement
367/// [`MpiIndexedDatatype`], not [`MpiDatatype`], which those APIs require.
368///
369/// # Example
370///
371/// ```no_run
372/// use ferrompi::{CollectiveOp, Mpi, DoubleInt, MpiIndexedDatatype};
373///
374/// let mpi = Mpi::init().unwrap();
375/// let world = mpi.world();
376/// let rank = world.rank();
377///
378/// let send = [DoubleInt { value: rank as f64, index: rank }];
379/// let mut recv = [DoubleInt { value: 0.0, index: 0 }];
380/// world.allreduce_indexed(&send, &mut recv, CollectiveOp::MAX_LOC).unwrap();
381/// // Every rank now holds { value: (size-1) as f64, index: size-1 }
382/// ```
383///
384/// # Sealed: cannot implement for external types
385///
386/// ```compile_fail,E0277
387/// use ferrompi::MpiIndexedDatatype;
388///
389/// #[derive(Clone, Copy)]
390/// struct Mine;
391/// // This must not compile — MpiIndexedDatatype is sealed.
392/// impl MpiIndexedDatatype for Mine {}
393/// ```
394pub trait MpiIndexedDatatype: sealed_indexed::Sealed + Copy + Send + 'static {}

575// ============================================================
576// Byte-permutable types for MPI_BYTE bitwise reductions
577// ============================================================
578
579/// Trait for types whose byte representation is valid for use with
580/// `MPI_BYTE`-typed bitwise reductions.
581///
582/// This is a **sealed trait** — only types whose memory layout
583/// consists entirely of meaningful bytes (no padding, no
584/// discriminants, no uninhabited cases) can implement it. ferrompi
585/// provides blanket impls for `u8`, `u16`, `u32`, `u64`, `i8`,
586/// `i16`, `i32`, `i64`, plus `[T; N]` where `T: BytePermutable`.
587///
588/// # Safety invariants (enforced by sealing)
589///
590/// - No padding bytes inside the type (every byte is meaningful).
591/// - No niche optimizations: every bit pattern is a valid value.
592/// - `Copy + Send + 'static`.
593///
594/// # Sealed: cannot implement for external types
595///
596/// ```compile_fail,E0277
597/// use ferrompi::BytePermutable;
598///
599/// #[derive(Clone, Copy)]
600/// struct Mine;
601/// // This must not compile — BytePermutable is sealed.
602/// impl BytePermutable for Mine {}
603/// ```
604pub trait BytePermutable: sealed_byte::Sealed + Copy + Send + 'static {}
605
606macro_rules! impl_byte_permutable {
607    ($ty:ty) => {
608        impl sealed_byte::Sealed for $ty {}
609        impl BytePermutable for $ty {}
610    };
611}
612
613impl_byte_permutable!(u8);
614impl_byte_permutable!(u16);
615impl_byte_permutable!(u32);
616impl_byte_permutable!(u64);
617impl_byte_permutable!(i8);
618impl_byte_permutable!(i16);
619impl_byte_permutable!(i32);
620impl_byte_permutable!(i64);
621
622impl<T: BytePermutable, const N: usize> sealed_byte::Sealed for [T; N] {}
623impl<T: BytePermutable, const N: usize> BytePermutable for [T; N] {}

625// ============================================================
626// Atomic-eligible types for MPI_Compare_and_swap
627// ============================================================
628
629/// Trait for types eligible for use with `MPI_Compare_and_swap`.
630///
631/// This is a **sealed trait** — it cannot be implemented outside this crate.
632/// MPI-4.1 §12.3.4 allows only integer, logical and byte datatypes in
633/// `MPI_Compare_and_swap`; of ferrompi's element types these are `i32`, `i64`,
634/// `u32`, `u64` and `u8`.
635///
636/// **Floating-point types (`f32`, `f64`) are intentionally excluded.** The
637/// MPI standard does not allow them in compare-and-swap. Accepting them would
638/// introduce silent portability bugs across MPI implementations.
639///
640/// This trait is used as a bound on
641/// [`ActiveEpoch::compare_and_swap`](crate::ActiveEpoch::compare_and_swap) and
642/// [`PassiveEpoch::compare_and_swap`](crate::PassiveEpoch::compare_and_swap)
643/// to ensure the method only exists for windows whose element type is
644/// CAS-eligible.
645///
646/// # Eligible types only
647///
648/// `compare_and_swap` exists only for integer and byte elements: on an `f64`
649/// window the method is not found.
650///
651/// ```compile_fail,E0599
652/// fn _check(e: &ferrompi::ActiveEpoch<'_, '_, f64>, v: &'static f64, r: &'static mut f64) -> ferrompi::Result<()> {
653///     e.compare_and_swap(v, v, r, 0, 0)
654/// }
655/// ```
656///
657/// The same call on an `i64` window compiles:
658///
659/// ```no_run
660/// fn _check(e: &ferrompi::ActiveEpoch<'_, '_, i64>, v: &'static i64, r: &'static mut i64) -> ferrompi::Result<()> {
661///     e.compare_and_swap(v, v, r, 0, 0)
662/// }
663/// ```
664#[cfg(feature = "rma")]
665pub trait AtomicMpiDatatype: sealed_atomic::Sealed + Copy + Send + 'static {}
666
667#[cfg(feature = "rma")]
668macro_rules! impl_atomic_mpi_datatype {
669    ($ty:ty) => {
670        impl sealed_atomic::Sealed for $ty {}
671        impl AtomicMpiDatatype for $ty {}
672    };
673}
674
675#[cfg(feature = "rma")]
676impl_atomic_mpi_datatype!(i32);
677#[cfg(feature = "rma")]
678impl_atomic_mpi_datatype!(i64);
679#[cfg(feature = "rma")]
680impl_atomic_mpi_datatype!(u32);
681#[cfg(feature = "rma")]
682impl_atomic_mpi_datatype!(u64);
683#[cfg(feature = "rma")]
684impl_atomic_mpi_datatype!(u8);

774    #[test]
775    fn empty_buffer_of_any_type_gets_a_fixed_address() {
776        let mut data = [0.0f64; 4];
777        let (none, some) = data.split_at_mut(0);
778        let full = buf(some).0;
779        let empty = buf_mut(none).0.cast_const();
780        // The defect: an empty slice of an alignment-8 type can start where the
781        // other buffer begins.
782        assert_eq!(full, empty);
783        let read = empty_buf_ptr(empty, true, false, full);
784        let written = empty_buf_ptr(empty, true, true, full);
785        assert!(read != full && written != full && read != written);
786        // They are the addresses `buf` and `buf_mut` give empty byte buffers.
787        let mut no_bytes: Vec<u8> = Vec::new();
788        assert_eq!(read, buf(&no_bytes).0);
789        assert_eq!(written, buf_mut(&mut no_bytes).0.cast_const());
790        // An empty buffer never gets the address the other buffer is at.
791        assert_eq!(empty_buf_ptr(empty, true, false, read), written);
792        assert_eq!(empty_buf_ptr(empty, true, true, written), read);
793        // A buffer that is not empty keeps its pointer.
794        assert_eq!(empty_buf_ptr(full, false, false, empty), full);
795        assert_eq!(empty_buf_ptr(full, false, true, full), full);
796    }
