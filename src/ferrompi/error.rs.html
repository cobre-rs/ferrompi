8/// Result type for MPI operations.

11// Internal resource-exhaustion sentinels returned by the C layer when a
12// fixed-size handle table is full. Negative so they never collide with MPI
13// return codes (which are non-negative); intercepted in [`Error::from_code`]
14// and mapped to [`Error::ResourceExhausted`]. These MUST match the
15// `FERROMPI_ERR_*_FULL` defines in `csrc/ferrompi.h`.
16const FERROMPI_ERR_REQUESTS_FULL: i32 = -7001;
17const FERROMPI_ERR_COMMS_FULL: i32 = -7002;
18const FERROMPI_ERR_DATATYPES_FULL: i32 = -7003;
19const FERROMPI_ERR_OPS_FULL: i32 = -7004;
20const FERROMPI_ERR_WINDOWS_FULL: i32 = -7005;
21const FERROMPI_ERR_GROUPS_FULL: i32 = -7006;
22const FERROMPI_ERR_INFOS_FULL: i32 = -7007;
23
24// Returned by the persistent-collective and comm_create_from_group stubs
25// compiled without FERROMPI_HAVE_MPI4_COLLECTIVES (the library lacks the
26// MPI 4.0 operation). Intercepted in [`Error::from_code`]/
27// [`Error::from_code_with_op`] and mapped to [`Error::NotSupported`]. This
28// MUST match `FERROMPI_ERR_NOT_SUPPORTED` in `csrc/ferrompi.h`.
29const FERROMPI_ERR_NOT_SUPPORTED: i32 = -7008;
30
31// Rust-only lifecycle-guard sentinels. Produced only by the Rust lifecycle
32// guard (never returned by the C layer), and outside the -7001..-7099 range
33// the C sentinels above use.
34pub(crate) const FERROMPI_ERR_FINALIZED: i32 = -7101;
35pub(crate) const FERROMPI_ERR_THREAD_LEVEL: i32 = -7102;

37/// Identifies which internal ferrompi handle table was exhausted in an
38/// [`Error::ResourceExhausted`].
39///
40/// Each variant corresponds to a fixed-size table in the C layer (e.g.
41/// `MAX_COMMS`, `MAX_DATATYPES`, `MAX_OPS`, `MAX_REQUESTS`). The `Display`
42/// form is the lowercase resource noun used in the error message.
43#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash, thiserror::Error)]
44pub enum ResourceKind {
45    /// Nonblocking / persistent request table (`MAX_REQUESTS`).
46    #[error("request")]
47    Request,
48    /// Communicator table (`MAX_COMMS`).
49    #[error("communicator")]
50    Communicator,
51    /// Custom datatype table (`MAX_DATATYPES`).
52    #[error("datatype")]
53    Datatype,
54    /// Custom reduction-operation table (`MAX_OPS`).
55    #[error("operation")]
56    Operation,
57    /// RMA window table.
58    #[error("window")]
59    Window,
60    /// Group table.
61    #[error("group")]
62    Group,
63    /// Info-object table.
64    #[error("info")]
65    Info,
66}
67
68impl ResourceKind {
69    /// Map a C-layer resource-exhaustion sentinel to its `ResourceKind`, or
70    /// `None` if `code` is not one of the `FERROMPI_ERR_*_FULL` sentinels.
71    fn from_sentinel(code: i32) -> Option<Self> {
72        match code {
73            FERROMPI_ERR_REQUESTS_FULL => Some(ResourceKind::Request),
74            FERROMPI_ERR_COMMS_FULL => Some(ResourceKind::Communicator),
75            FERROMPI_ERR_DATATYPES_FULL => Some(ResourceKind::Datatype),
76            FERROMPI_ERR_OPS_FULL => Some(ResourceKind::Operation),
77            FERROMPI_ERR_WINDOWS_FULL => Some(ResourceKind::Window),
78            FERROMPI_ERR_GROUPS_FULL => Some(ResourceKind::Group),
79            FERROMPI_ERR_INFOS_FULL => Some(ResourceKind::Info),
80            _ => None,
81        }
82    }
83}

85/// MPI error class, categorizing the type of MPI error.
86///
87/// The named variants correspond to the MPI error classes, compared against
88/// the linked MPI library's own `MPI_ERR_*` constants (see [`Self::from_raw`]).
89/// [`Self::Raw`] carries a class value this enum does not name.
90#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash, thiserror::Error)]
91pub enum MpiErrorClass {
92    /// `MPI_SUCCESS` — no error
93    #[error("SUCCESS")]
94    Success,
95    /// `MPI_ERR_BUFFER` — invalid buffer pointer
96    #[error("ERR_BUFFER")]
97    Buffer,
98    /// `MPI_ERR_COUNT` — invalid count argument
99    #[error("ERR_COUNT")]
100    Count,
101    /// `MPI_ERR_TYPE` — invalid datatype argument
102    #[error("ERR_TYPE")]
103    Type,
104    /// `MPI_ERR_TAG` — invalid tag argument
105    #[error("ERR_TAG")]
106    Tag,
107    /// `MPI_ERR_COMM` — invalid communicator
108    #[error("ERR_COMM")]
109    Comm,
110    /// `MPI_ERR_RANK` — invalid rank
111    #[error("ERR_RANK")]
112    Rank,
113    /// `MPI_ERR_REQUEST` — invalid request handle
114    #[error("ERR_REQUEST")]
115    Request,
116    /// `MPI_ERR_ROOT` — invalid root
117    #[error("ERR_ROOT")]
118    Root,
119    /// `MPI_ERR_GROUP` — invalid group
120    #[error("ERR_GROUP")]
121    Group,
122    /// `MPI_ERR_OP` — invalid operation
123    #[error("ERR_OP")]
124    Op,
125    /// `MPI_ERR_TOPOLOGY` — invalid topology
126    #[error("ERR_TOPOLOGY")]
127    Topology,
128    /// `MPI_ERR_DIMS` — invalid dimension argument
129    #[error("ERR_DIMS")]
130    Dims,
131    /// `MPI_ERR_ARG` — invalid argument
132    #[error("ERR_ARG")]
133    Arg,
134    /// `MPI_ERR_UNKNOWN` — unknown error
135    #[error("ERR_UNKNOWN")]
136    Unknown,
137    /// `MPI_ERR_TRUNCATE` — message truncated
138    #[error("ERR_TRUNCATE")]
139    Truncate,
140    /// `MPI_ERR_OTHER` — other error
141    #[error("ERR_OTHER")]
142    Other,
143    /// `MPI_ERR_INTERN` — internal MPI error
144    #[error("ERR_INTERN")]
145    Intern,
146    /// `MPI_ERR_IN_STATUS` — error code is in status
147    #[error("ERR_IN_STATUS")]
148    InStatus,
149    /// `MPI_ERR_PENDING` — pending request
150    #[error("ERR_PENDING")]
151    Pending,
152    /// `MPI_ERR_WIN` — invalid window
153    #[error("ERR_WIN")]
154    Win,
155    /// `MPI_ERR_INFO` — invalid info object
156    #[error("ERR_INFO")]
157    Info,
158    /// `MPI_ERR_FILE` — invalid file handle
159    #[error("ERR_FILE")]
160    File,
161    /// Unrecognized error class from the MPI implementation
162    #[error("ERR_CLASS({0})")]
163    Raw(i32),
164}
165
166impl MpiErrorClass {
167    /// Map a class value of the linked MPI library (as returned by
168    /// `MPI_Error_class`) to the enum variant.
169    ///
170    /// Only `MPI_SUCCESS = 0` is fixed by the MPI standard; every other class
171    /// value is implementation-defined and differs between MPI libraries
172    /// (e.g. MPICH numbers `MPI_ERR_ROOT` 7, Open MPI numbers it 8). This
173    /// delegates to the C layer, which compares `class` against the linked
174    /// library's own `MPI_ERR_*` constants, so the mapping is correct on
175    /// MPICH-derived libraries, Open MPI and the MPI 5 standard ABI.
176    pub fn from_raw(class: i32) -> Self {
177        // SAFETY: ferrompi_error_class_index only compares its argument
178        // against compile-time MPI_ERR_* constants; there is no pointer,
179        // buffer, or initialization precondition to uphold.
180        let index = unsafe { ffi::ferrompi_error_class_index(class) };
181        match index {
182            0 => MpiErrorClass::Success,
183            1 => MpiErrorClass::Buffer,
184            2 => MpiErrorClass::Count,
185            3 => MpiErrorClass::Type,
186            4 => MpiErrorClass::Tag,
187            5 => MpiErrorClass::Comm,
188            6 => MpiErrorClass::Rank,
189            7 => MpiErrorClass::Request,
190            8 => MpiErrorClass::Root,
191            9 => MpiErrorClass::Group,
192            10 => MpiErrorClass::Op,
193            11 => MpiErrorClass::Topology,
194            12 => MpiErrorClass::Dims,
195            13 => MpiErrorClass::Arg,
196            14 => MpiErrorClass::Unknown,
197            15 => MpiErrorClass::Truncate,
198            16 => MpiErrorClass::Other,
199            17 => MpiErrorClass::Intern,
200            18 => MpiErrorClass::InStatus,
201            19 => MpiErrorClass::Pending,
202            20 => MpiErrorClass::Win,
203            21 => MpiErrorClass::Info,
204            22 => MpiErrorClass::File,
205            _ => MpiErrorClass::Raw(class),
206        }
207    }
208}

222/// Error types for MPI operations.
223///
224/// This enum is non-exhaustive: downstream `match`es need a wildcard arm.
225///
226/// # Which failures return `Err`
227///
228/// `ferrompi` installs `MPI_ERRORS_RETURN` on `MPI_COMM_WORLD`, on
229/// `MPI_COMM_SELF` (which receives errors the MPI standard does not
230/// attribute to any communicator or window), and on every communicator and
231/// window it creates (windows on a best-effort basis — see below), so
232/// failures on those objects return `Err` instead of aborting through MPI's
233/// default handler.
234///
235/// Nine paths end the process instead of returning `Err`:
236/// - an error inside `MPI_Init_thread` itself, before any error handler is
237///   installed: MPI's default handler is `MPI_ERRORS_ARE_FATAL`, so
238///   [`Mpi::init`](crate::Mpi::init)/[`Mpi::init_thread`](crate::Mpi::init_thread)
239///   return `Err(Error::Mpi)` only when MPI's initial error handler returns
240///   instead of aborting;
241/// - a failure to install `MPI_ERRORS_RETURN` on `MPI_COMM_WORLD` or
242///   `MPI_COMM_SELF` right after `MPI_Init_thread` succeeds calls
243///   `MPI_Abort`, since MPI cannot be initialized a second time; if
244///   `MPI_Abort` itself returns on a non-conforming implementation, the
245///   process ends by `SIGABRT` from a fallback abort instead;
246/// - a window whose own `MPI_Win_set_errhandler` call failed (a stderr
247///   warning is printed; a known Open MPI 4.x quirk) keeps MPI's default
248///   handler, so a later RMA error on that window aborts instead of
249///   returning `Err`;
250/// - a panic inside a [`UserOp`](crate::UserOp) reduction closure aborts
251///   the process;
252/// - dropping a handle on the wrong thread aborts the process; see the
253///   [`Mpi`](crate::Mpi) lifecycle section;
254/// - under a fault-tolerant MPI, a receive from any source that a process
255///   failure leaves pending (`MPI_ERR_PROC_FAILED_PENDING`) makes the
256///   completion call that reports it print a message and abort the
257///   process, since no `ferrompi` call can complete it; so does the
258///   nonblocking receive call itself when the request table was full,
259///   since it then waits for the receive before returning;
260/// - `Win::create` on a rank whose window table is full aborts the process
261///   (`MPI_Abort`, falling back to `SIGABRT` if `MPI_Abort` itself returns),
262///   since the new window already exposes the caller's buffer to its peers;
263/// - in a debug build at
264///   [`ThreadLevel::Serialized`](crate::ThreadLevel::Serialized), a
265///   [`Request::wait`](crate::Request::wait) that overlaps another thread's MPI
266///   call aborts the process, since it can neither run nor hand the request back;
267/// - [`Communicator::abort`](crate::Communicator::abort) aborts the process
268///   by design.
269///
270/// `Drop` implementations that call MPI never return an error from that
271/// call; apart from a stderr line when freeing a [`UserOp`](crate::UserOp)
272/// fails, an error during drop is not reported.
273#[derive(Debug, thiserror::Error)]
274#[non_exhaustive]
275pub enum Error {
276    /// MPI has already been initialized.
277    #[error("MPI has already been initialized")]
278    AlreadyInitialized,
279
280    /// MPI has been finalized.
281    ///
282    /// Returned by any MPI-calling method invoked after the [`Mpi`](crate::Mpi)
283    /// handle was dropped, and by `Mpi::init`/`Mpi::init_thread` once MPI has
284    /// been finalized. No MPI call is made.
285    #[error("MPI has been finalized")]
286    Finalized,
287
288    /// MPI call not permitted on this thread at the provided thread level.
289    ///
290    /// Returned by any MPI-calling method invoked from a thread other than
291    /// the one that called `Mpi::init_thread` while the provided level is
292    /// below [`ThreadLevel::Serialized`](crate::ThreadLevel::Serialized),
293    /// and, in debug builds only, by a call that overlaps another thread's
294    /// call at [`ThreadLevel::Serialized`](crate::ThreadLevel::Serialized).
295    /// No MPI call is made.
296    #[error("MPI call not permitted on this thread at the provided thread level")]
297    ThreadLevelViolation,
298
299    /// MPI error with class, code, descriptive message, and optional operation name.
300    #[error("{}", fmt_mpi(.class, .code, .message, .operation))]
301    Mpi {
302        /// The error class (category of error).
303        class: MpiErrorClass,
304        /// The raw MPI error code.
305        code: i32,
306        /// Human-readable error message from `MPI_Error_string`.
307        message: String,
308        /// The ferrompi operation that produced the error, if known.
309        operation: Option<&'static str>,
310    },
311
312    /// Invalid buffer provided (e.g., send/recv buffer size mismatch).
313    #[error("Invalid buffer")]
314    InvalidBuffer,
315
316    /// Invalid reduction operation for the method being called
317    /// (e.g., passing a non-MAXLOC/MINLOC op to `allreduce_indexed`).
318    #[error("Invalid reduction operation for this method")]
319    InvalidOp,
320
321    /// Operation not supported. Two sources:
322    /// - an MPI 4.0 operation (a persistent collective or
323    ///   [`Mpi::create_from_group`](crate::Mpi::create_from_group)) when
324    ///   ferrompi was built against an MPI older than 4.0 other than Open
325    ///   MPI 5, which implements both;
326    /// - [`Request::cancel`](crate::Request::cancel) on a nonblocking-collective
327    ///   or RMA request.
328    ///
329    /// The string names the operation.
330    #[error("Operation not supported: {0}")]
331    NotSupported(String),
332
333    /// An internal ferrompi handle table is full (too many concurrently live
334    /// requests / communicators / datatypes / etc.).
335    ///
336    /// Distinct from a genuine MPI fault ([`Error::Mpi`] with class
337    /// `ERR_OTHER`) so a long-running caller can recognize an internal cap,
338    /// release handles, back off, or retry rather than treating it as fatal.
339    ///
340    /// A communicator or window constructor that returns this error ran its
341    /// collective MPI call first: MPI created the object on every rank whose
342    /// call succeeded, and this rank leaks its copy (never freed; a window
343    /// counts as alive, so [`Mpi`](crate::Mpi) skips `MPI_Finalize`). Peers
344    /// whose call succeeded hold theirs, so collective calls on it cannot
345    /// complete; retrying is a new collective call on every rank.
346    #[error("ferrompi {resource} table is full")]
347    ResourceExhausted {
348        /// Which internal handle table overflowed.
349        resource: ResourceKind,
350    },
351
352    /// Internal ferrompi error.
353    #[error("Internal error: {0}")]
354    Internal(String),
355}
356
357impl Error {
358    /// Create a structured error from an MPI error code.
359    ///
360    /// Calls `ferrompi_error_info` to obtain the error class and human-readable
361    /// message from the MPI runtime.
362    ///
363    /// # Returns
364    ///
365    /// If called with `MPI_SUCCESS` (code 0), returns
366    /// `Error::Internal("from_code called with success code 0")`.
367    /// Callers should use [`Error::check`] or [`Error::check_with_op`]
368    /// for the standard "Ok on success, Err on failure" idiom.
369    ///
370    /// Marked `#[cold]` + `#[inline(never)]`: this is only reached on the error
371    /// path of [`Error::check`]/[`Error::check_with_op`], and its body (a 512-byte
372    /// stack buffer, an FFI call to `ferrompi_error_info`, and a `format!`
373    /// fallback) must stay out of line so the inlined success-path check folds
374    /// to a single return-code branch in the caller.
375    #[cold]
376    #[inline(never)]
377    pub fn from_code(code: i32) -> Self {
378        if code == 0 {
379            return Error::Internal("from_code called with success code 0".into());
380        }
381
382        // Resource-exhaustion sentinels from the C layer are negative; MPI
383        // codes are non-negative. Map them before touching the MPI runtime —
384        // MPI_Error_class on a synthetic negative code is undefined behavior.
385        if let Some(resource) = ResourceKind::from_sentinel(code) {
386            return Error::ResourceExhausted { resource };
387        }
388        if code == FERROMPI_ERR_FINALIZED {
389            return Error::Finalized;
390        }
391        if code == FERROMPI_ERR_THREAD_LEVEL {
392            return Error::ThreadLevelViolation;
393        }
394        if code == FERROMPI_ERR_NOT_SUPPORTED {
395            return Error::NotSupported("MPI 4.0 operation".into());
396        }
397
398        let mut class: i32 = 0;
399        let mut msg_buf = [0u8; 512];
400        let mut msg_len: i32 = 0;
401
402        // SAFETY: class and msg_len are local out-parameters; msg_buf is a
403        // local 512-byte buffer; MPI writes at most MPI_MAX_ERROR_STRING bytes,
404        // which the C layer asserts at build time is at most 512, reporting
405        // the written length through msg_len.
406        let ret = unsafe {
407            ffi::ferrompi_error_info(
408                code,
409                &mut class,
410                msg_buf.as_mut_ptr().cast::<std::ffi::c_char>(),
411                &mut msg_len,
412            )
413        };
414
415        if ret == 0 {
416            let len = (msg_len.max(0) as usize).min(msg_buf.len());
417            let len = msg_buf[..len].iter().position(|&b| b == 0).unwrap_or(len);
418            let message = std::str::from_utf8(&msg_buf[..len])
419                .unwrap_or("unknown error")
420                .to_string();
421            Error::Mpi {
422                class: MpiErrorClass::from_raw(class),
423                code,
424                message,
425                operation: None,
426            }
427        } else {
428            // ferrompi_error_info itself failed — provide a fallback
429            Error::Mpi {
430                class: MpiErrorClass::Raw(code),
431                code,
432                message: format!("MPI error code {code}"),
433                operation: None,
434            }
435        }
436    }
437
438    /// Create a structured error from an MPI error code with an operation name.
439    ///
440    /// Behaves identically to [`Error::from_code`] but populates `operation:
441    /// Some(operation)` on the resulting [`Error::Mpi`] variant, so callers can
442    /// record which ferrompi wrapper produced the error.
443    ///
444    /// # Contract
445    ///
446    /// Callers must not pass `code = 0`. The canonical success-vs-error
447    /// idiom is [`Error::check_with_op`], which short-circuits to `Ok(())`
448    /// when `code == 0`. Passing 0 here returns `Error::Internal("from_code
449    /// called with success code 0")` as a defensive fallback — this signals
450    /// a misuse of the API, not an MPI error condition.
451    ///
452    /// # Returns
453    ///
454    /// - `Error::Mpi { .. }` for an MPI error code, with the `operation`
455    ///   field populated; a ferrompi sentinel maps to the same variant as
456    ///   in [`Error::from_code`] (`NotSupported` carries `operation`).
457    /// - `Error::Internal` if called with `code = 0` (delegated from
458    ///   [`Error::from_code`]; treat this as a programming error in the
459    ///   caller, not a runtime MPI failure).
460    ///
461    /// Marked `#[cold]` + `#[inline(never)]` for the same reason as
462    /// [`Error::from_code`]: it is only invoked on the error path, so keeping
463    /// it out of line lets [`Error::check_with_op`] inline to a bare branch.
464    #[cold]
465    #[inline(never)]
466    pub fn from_code_with_op(code: i32, operation: &'static str) -> Self {
467        if code == FERROMPI_ERR_NOT_SUPPORTED {
468            return Error::NotSupported(operation.into());
469        }
470        match Error::from_code(code) {
471            Error::Mpi {
472                class,
473                code,
474                message,
475                ..
476            } => Error::Mpi {
477                class,
478                code,
479                message,
480                operation: Some(operation),
481            },
482            // Every other variant (a sentinel's variant, or Internal for
483            // code 0) is preserved verbatim.
484            other => other,
485        }
486    }
487
488    /// Check an MPI return code, returning `Ok(())` for success.
489    ///
490    /// Returns `Err(Error::from_code(code))` for a non-zero code: `Error::Mpi`
491    /// for an MPI error code, or the variant [`Error::from_code`] maps a
492    /// ferrompi sentinel to.
493    ///
494    /// `#[inline]` so the success-path check (`code == 0`) folds into the
495    /// caller across codegen-unit/crate boundaries; the cold error
496    /// construction is delegated to the `#[cold]` [`Error::from_code`].
497    #[inline]
498    pub fn check(code: i32) -> Result<()> {
499        if code == 0 {
500            Ok(())
501        } else {
502            Err(Error::from_code(code))
503        }
504    }
505
506    /// Check an MPI return code with an operation name, returning `Ok(())` for
507    /// success.
508    ///
509    /// Returns `Err(Error::from_code_with_op(code, operation))` for a
510    /// non-zero code: `Error::Mpi` for an MPI error code, or the variant
511    /// [`Error::from_code_with_op`] maps a ferrompi sentinel to.
512    ///
513    /// `#[inline]` so the success-path check folds into each call site; the
514    /// cold error construction stays out of line in the `#[cold]`
515    /// [`Error::from_code_with_op`].
516    #[inline]
517    pub fn check_with_op(code: i32, operation: &'static str) -> Result<()> {
518        if code == 0 {
519            Ok(())
520        } else {
521            Err(Error::from_code_with_op(code, operation))
522        }
523    }
524}
