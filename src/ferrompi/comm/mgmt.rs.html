9    /// Get the processor name for this process.
10    pub fn processor_name(&self) -> Result<String> {
11        let mut buf = [0u8; 256];
12        let mut len: i32 = 0;
13        // SAFETY: buf is a 256-byte stack array, exclusively writable; MPI writes at most
14        // MPI_MAX_PROCESSOR_NAME bytes, which the C layer asserts at build time is at most 256.
15        let ret = unsafe {
16            ffi::ferrompi_get_processor_name(buf.as_mut_ptr().cast::<std::ffi::c_char>(), &mut len)
17        };
18        Error::check_with_op(ret, "get_processor_name")?;
19        let len = (len.max(0) as usize).min(buf.len());
20        let len = buf[..len].iter().position(|&b| b == 0).unwrap_or(len);
21        let s = std::str::from_utf8(&buf[..len])
22            .map_err(|_| Error::Internal("Invalid UTF-8 in processor name".into()))?;
23        Ok(s.to_string())
24    }

26    /// Gather topology information from all ranks in this communicator.
27    ///
28    /// This is a **collective operation** — all ranks must call it.
29    /// Every rank receives the complete [`TopologyInfo`], which includes the
30    /// rank-to-host mapping, MPI version metadata, and (with the `numa`
31    /// feature) SLURM job information.
32    ///
33    /// [`TopologyInfo`]: crate::TopologyInfo
34    ///
35    /// # Errors
36    ///
37    /// If the processor-name or version query fails on any rank, every rank
38    /// returns `Err`: that rank its own error, every other rank
39    /// [`Error::Internal`] naming the lowest such rank.
40    ///
41    /// # Example
42    ///
43    /// ```no_run
44    /// use ferrompi::Mpi;
45    ///
46    /// let mpi = Mpi::init().unwrap();
47    /// let world = mpi.world();
48    /// let topo = world.topology(&mpi).unwrap();
49    /// if world.rank() == 0 {
50    ///     println!("{topo}");
51    /// }
52    /// ```
53    pub fn topology(&self, mpi: &crate::Mpi) -> Result<crate::TopologyInfo> {
54        crate::topology::gather_topology(self, mpi)
55    }

57    /// Get the group associated with this communicator.
58    ///
59    /// Returns a [`Group`] containing all processes in this communicator.
60    /// Use [`Group::include`] or [`Group::exclude`] on the returned group
61    /// to derive sub-groups, which can then be used with
62    /// [`Communicator::create_from_group`](Self::create_from_group).
63    ///
64    /// # Errors
65    ///
66    /// Returns an error if the underlying `MPI_Comm_group` call fails or if
67    /// the C-side group table is full.
68    ///
69    /// # Example
70    ///
71    /// ```no_run
72    /// use ferrompi::Mpi;
73    ///
74    /// let mpi = Mpi::init().unwrap();
75    /// let world = mpi.world();
76    /// let g = world.group().unwrap();
77    /// assert_eq!(g.size().unwrap(), world.size());
78    /// ```
79    pub fn group(&self) -> Result<Group> {
80        let mut group_handle: i32 = 0;
81        // SAFETY: self.handle is owned by this Communicator; group_handle is an exclusive output.
82        let ret = unsafe { ffi::ferrompi_comm_group(self.handle, &mut group_handle) };
83        Error::check_with_op(ret, "comm_group")?;
84        Ok(Group {
85            handle: group_handle,
86        })
87    }

89    /// Duplicate this communicator.
90    pub fn duplicate(&self) -> Result<Self> {
91        let mut new_handle: i32 = 0;
92        // SAFETY: self.handle is owned by this Communicator; new_handle is an exclusive output.
93        let ret = unsafe { ffi::ferrompi_comm_dup(self.handle, &mut new_handle) };
94        Error::check_with_op(ret, "comm_dup")?;
95        Self::from_handle(new_handle)
96    }

98    /// Convert a communicator-handle result into `Option<Self>`, treating a
99    /// negative handle as `MPI_COMM_NULL` (used by split, split_type, and
100    /// create_from_group).
101    fn from_optional_handle(new_handle: i32) -> Result<Option<Self>> {
102        if new_handle < 0 {
103            Ok(None)
104        } else {
105            Self::from_handle(new_handle).map(Some)
106        }
107    }

109    /// Split this communicator into sub-communicators based on color and key.
110    ///
111    /// Processes with the same `color` are placed in the same new communicator.
112    /// The `key` controls the rank ordering within the new communicator.
113    ///
114    /// Returns `None` if this process used [`Communicator::UNDEFINED`] as color.
115    ///
116    /// # Example
117    ///
118    /// ```no_run
119    /// use ferrompi::Mpi;
120    ///
121    /// let mpi = Mpi::init().unwrap();
122    /// let world = mpi.world();
123    /// let color = world.rank() % 2; // Even/odd split
124    /// if let Some(sub) = world.split(color, world.rank()).unwrap() {
125    ///     println!("Rank {} in sub-communicator of size {}", sub.rank(), sub.size());
126    /// }
127    /// ```
128    pub fn split(&self, color: i32, key: i32) -> Result<Option<Communicator>> {
129        let mut new_handle: i32 = 0;
130        // SAFETY: self.handle is owned by this Communicator; remaining arguments are scalars.
131        let ret = unsafe { ffi::ferrompi_comm_split(self.handle, color, key, &mut new_handle) };
132        Error::check_with_op(ret, "comm_split")?;
133        Self::from_optional_handle(new_handle)
134    }

136    /// Split this communicator by type.
137    ///
138    /// Processes that share the same resource (determined by `split_type`) are
139    /// placed in the same new communicator. The `key` controls the rank ordering
140    /// within the new communicator.
141    ///
142    /// Returns `None` if MPI returns `MPI_COMM_NULL` for this process.
143    ///
144    /// # Example
145    ///
146    /// ```no_run
147    /// use ferrompi::{Mpi, SplitType};
148    ///
149    /// let mpi = Mpi::init().unwrap();
150    /// let world = mpi.world();
151    /// if let Some(node) = world.split_type(SplitType::Shared, world.rank()).unwrap() {
152    ///     println!("Node has {} processes", node.size());
153    /// }
154    /// ```
155    pub fn split_type(&self, split_type: SplitType, key: i32) -> Result<Option<Communicator>> {
156        let mut new_handle: i32 = 0;
157        // SAFETY: self.handle is owned by this Communicator; remaining arguments are scalars.
158        let ret = unsafe {
159            ffi::ferrompi_comm_split_type(self.handle, split_type as i32, key, &mut new_handle)
160        };
161        Error::check_with_op(ret, "comm_split_type")?;
162        Self::from_optional_handle(new_handle)
163    }

165    /// Create a communicator containing only processes that share memory.
166    ///
167    /// This is equivalent to `split_type(SplitType::Shared, self.rank())`.
168    /// All processes on the same physical node will be in the same communicator.
169    ///
170    /// # Errors
171    ///
172    /// Returns [`Error::Internal`] if MPI unexpectedly returns a null communicator,
173    /// which should not happen for `MPI_COMM_TYPE_SHARED` under normal conditions.
174    ///
175    /// # Example
176    ///
177    /// ```no_run
178    /// use ferrompi::Mpi;
179    ///
180    /// let mpi = Mpi::init().unwrap();
181    /// let world = mpi.world();
182    /// let node = world.split_shared().unwrap();
183    /// println!("Node has {} processes, I am local rank {}", node.size(), node.rank());
184    /// ```
185    pub fn split_shared(&self) -> Result<Communicator> {
186        self.split_type(SplitType::Shared, self.rank())?
187            .ok_or_else(|| Error::Internal("split_shared returned null communicator".into()))
188    }

190    /// Create a sub-communicator from a group.
191    ///
192    /// `group` must be a subset of `self`'s group. This call is collective
193    /// over `self` — every rank in the parent communicator must invoke it,
194    /// even ranks not in `group`.
195    ///
196    /// Returns `Ok(None)` for ranks that are not members of `group`
197    /// (MPI returns `MPI_COMM_NULL` for those ranks). Returns
198    /// `Ok(Some(comm))` for ranks that are members.
199    ///
200    /// # Errors
201    ///
202    /// Returns an error if `group` contains a rank not present in the parent
203    /// communicator (`MPI_ERR_GROUP`), or
204    /// [`Error::ResourceExhausted`](crate::Error::ResourceExhausted) if the
205    /// communicator table is full.
206    ///
207    /// # Example
208    ///
209    /// ```no_run
210    /// use ferrompi::Mpi;
211    ///
212    /// let mpi = Mpi::init().unwrap();
213    /// let world = mpi.world();
214    /// let g = world.group().unwrap().include(&[0, 2]).unwrap();
215    /// // Collective: every rank calls create_from_group
216    /// if let Some(sub) = world.create_from_group(&g).unwrap() {
217    ///     assert_eq!(sub.size(), 2);
218    /// }
219    /// ```
220    pub fn create_from_group(&self, group: &Group) -> Result<Option<Communicator>> {
221        let mut new_handle: i32 = -1;
222        // SAFETY: self.handle is owned by this Communicator; group.handle is owned by the Group
223        // argument and remains valid for the duration of this call.
224        let ret = unsafe {
225            ffi::ferrompi_comm_create_from_group_parent(self.handle, group.handle, &mut new_handle)
226        };
227        Error::check_with_op(ret, "comm_create_from_group_parent")?;
228        Self::from_optional_handle(new_handle)
229    }

231    /// Abort MPI execution across all processes in this communicator.
232    ///
233    /// This function terminates all processes associated with the communicator.
234    /// It calls `MPI_Abort` and then exits the process. This function never
235    /// returns.
236    ///
237    /// # Arguments
238    ///
239    /// * `errorcode` - Error code to return to the invoking environment
240    ///
241    /// # Example
242    ///
243    /// ```no_run
244    /// # use ferrompi::Mpi;
245    /// # let mpi = Mpi::init().unwrap();
246    /// # let world = mpi.world();
247    /// if world.rank() == 0 {
248    ///     // Fatal error detected, abort all processes
249    ///     world.abort(1);
250    /// }
251    /// ```
252    ///
253    /// # Fallback Behavior
254    ///
255    /// In the standard case `MPI_Abort` terminates the entire process
256    /// group and this function never returns. If `MPI_Abort` itself
257    /// returns (the MPI standard asks it to end the processes, but that is
258    /// not guaranteed, and some implementations return), or if ferrompi
259    /// rejects the call before it reaches MPI (the lifecycle/thread-level
260    /// conditions of
261    /// [`Error::Finalized`](crate::Error::Finalized) and
262    /// [`Error::ThreadLevelViolation`](crate::Error::ThreadLevelViolation)),
263    /// this function falls back to [`std::process::abort`], which raises
264    /// `SIGABRT` and terminates the current process immediately. The
265    /// original `errorcode` is lost in that fallback path because
266    /// `process::abort` does not accept an exit code; the signal code
267    /// (usually 134 = 128 + 6) is the best evidence the caller has that
268    /// the fallback triggered.
269    pub fn abort(&self, errorcode: i32) -> ! {
270        // SAFETY: ferrompi_abort passes self.handle, which is valid because
271        // Communicator is only constructed through safe methods that guarantee
272        // it, to MPI_Abort. MPI_Abort should terminate every process in the
273        // communicator but is not guaranteed to; if it returns, the
274        // std::process::abort below ends this process.
275        unsafe { ffi::ferrompi_abort(self.handle, errorcode) };
276
277        // Defense in depth: the MPI standard says MPI_Abort "should"
278        // terminate all processes but does not strictly guarantee the
279        // calling process aborts before return. If MPI_Abort ever
280        // returns (allowed by the standard, and observed in practice), or
281        // if the lifecycle guard rejected the call above before it reached
282        // MPI, we must still honor the `-> !` contract. std::process::abort()
283        // raises SIGABRT and is guaranteed to diverge without running
284        // destructors — which is the right outcome, because any
285        // destructor that touches MPI state after a failed or rejected
286        // MPI_Abort has undefined behavior.
287        std::process::abort()
288    }
