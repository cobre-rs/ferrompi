20    /// Barrier synchronization.
21    ///
22    /// All processes in the communicator must call this function. No process
23    /// will return until all processes have entered the barrier.
24    #[inline]
25    pub fn barrier(&self) -> Result<()> {
26        // SAFETY: self.handle is a valid communicator handle registered in the C-side
27        // communicator table. ferrompi_barrier delegates to MPI_Barrier which requires no
28        // buffers; only a valid communicator handle is needed.
29        let ret = unsafe { ffi::ferrompi_barrier(self.handle) };
30        Error::check_with_op(ret, "barrier")
31    }

37    /// Broadcast a slice from root to all processes.
38    ///
39    /// # Arguments
40    ///
41    /// * `data` - Buffer to broadcast (input at root, output at others)
42    /// * `root` - Rank of the root process
43    ///
44    /// # Example
45    ///
46    /// ```no_run
47    /// # use ferrompi::Mpi;
48    /// # let mpi = Mpi::init().unwrap();
49    /// # let world = mpi.world();
50    /// let mut data = vec![0.0f64; 100];
51    /// if world.rank() == 0 {
52    ///     data.fill(42.0);
53    /// }
54    /// world.broadcast(&mut data, 0).unwrap();
55    /// ```
56    #[inline]
57    pub fn broadcast<T: MpiDatatype>(&self, data: &mut [T], root: i32) -> Result<()> {
58        let (p, n, dt) = buf_mut(data);
59        // SAFETY: this blocking call returns only after MPI is done with the buffer.
60        let ret = unsafe { ffi::ferrompi_bcast(p, n, dt, root, self.handle) };
61        Error::check_with_op(ret, "bcast")
62    }

64    /// Reduce values to the root process.
65    ///
66    /// # Arguments
67    ///
68    /// * `send` - Data to send from this process
69    /// * `recv` - Buffer for the result at the root, which must have
70    ///   `send.len()` elements there; ignored at other ranks, where it may be
71    ///   empty
72    /// * `op` - a [`ReduceOp`], a `&`[`UserOp<T>`](crate::UserOp), or a [`CollectiveOp`]
73    /// * `root` - Rank of the root process
74    ///
75    /// # Errors
76    ///
77    /// - [`Error::InvalidArgument`] with `arg: "op"` if `op` is a predefined op MPI does not
78    ///   define on `T` (a bitwise or logical op on `f32`/`f64`)
79    /// - [`Error::BufferSize`] if this rank is `root` and `recv.len() != send.len()`
80    /// - [`Error::Mpi`] with class [`MpiErrorClass::Count`](crate::MpiErrorClass::Count) if
81    ///   `op` is a user op and `send.len()` exceeds `i32::MAX`, on every MPI version
82    ///
83    /// # Example
84    ///
85    /// ```no_run
86    /// # use ferrompi::{Mpi, ReduceOp};
87    /// # let mpi = Mpi::init().unwrap();
88    /// # let world = mpi.world();
89    /// let send = vec![1.0f64; 10];
90    /// let mut recv = vec![0.0f64; 10];
91    /// world.reduce(&send, &mut recv, ReduceOp::Sum, 0).unwrap();
92    /// ```
93    pub fn reduce<'a, T: MpiDatatype>(
94        &self,
95        send: &[T],
96        recv: &mut [T],
97        op: impl Into<CollectiveOp<'a, T>>,
98        root: i32,
99    ) -> Result<()> {
100        let op = op.into().code(tag_of::<T>())?;
101        if self.rank == root {
102            check_same_len("recv", send.len(), recv.len())?;
103        }
104        let (sp, n, dt) = buf(send);
105        let (rp, _, _) = buf_mut(recv);
106        // SAFETY: send.len() == recv.len() is verified above at the root; MPI ignores recvbuf at
107        // every other rank (MPI-4.1 section 6.9.1). The two slices cannot alias (&[T] vs &mut [T]).
108        let ret = unsafe { ffi::ferrompi_reduce(sp, rp, n, dt, op, root, self.handle) };
109        Error::check_with_op(ret, "reduce")
110    }

151    /// In-place reduce to the root process.
152    ///
153    /// At root: `data` is both input and output (the reduction result overwrites
154    /// the input).
155    /// At non-root: `data` is the send buffer only.
156    ///
157    /// This avoids allocating a separate receive buffer at the root, which is
158    /// useful for large reductions where memory is a concern.
159    ///
160    /// # Arguments
161    ///
162    /// * `data` - Buffer to reduce (input on all ranks, output only at root)
163    /// * `op` - a [`ReduceOp`], a `&`[`UserOp<T>`](crate::UserOp), or a [`CollectiveOp`]
164    /// * `root` - Rank of the root process
165    ///
166    /// # Errors
167    ///
168    /// - [`Error::InvalidArgument`] with `arg: "op"` if `op` is a predefined op MPI does not
169    ///   define on `T` (a bitwise or logical op on `f32`/`f64`)
170    /// - [`Error::Mpi`] with class [`MpiErrorClass::Count`](crate::MpiErrorClass::Count) if
171    ///   `op` is a user op and `data.len()` exceeds `i32::MAX`, on every MPI version
172    ///
173    /// # Example
174    ///
175    /// ```no_run
176    /// # use ferrompi::{Mpi, ReduceOp};
177    /// # let mpi = Mpi::init().unwrap();
178    /// # let world = mpi.world();
179    /// let mut data = vec![world.rank() as f64; 10];
180    /// world.reduce_inplace(&mut data, ReduceOp::Sum, 0).unwrap();
181    /// if world.rank() == 0 {
182    ///     println!("Reduced result: {:?}", &data[..3]);
183    /// }
184    /// ```
185    pub fn reduce_inplace<'a, T: MpiDatatype>(
186        &self,
187        data: &mut [T],
188        op: impl Into<CollectiveOp<'a, T>>,
189        root: i32,
190    ) -> Result<()> {
191        let op = op.into().code(tag_of::<T>())?;
192        let (p, n, dt) = buf_mut(data);
193        let ret = if self.rank() == root {
194            // SAFETY: NULL sendbuf is the in-place marker (buf_mut's pointer is never null, so
195            // this NULL is unambiguous); ferrompi_reduce maps it to MPI_IN_PLACE, so data is
196            // both input and output at root.
197            unsafe { ffi::ferrompi_reduce(std::ptr::null(), p, n, dt, op, root, self.handle) }
198        } else {
199            // SAFETY: data is passed as both sendbuf and recvbuf because strict MPI builds
200            // reject a NULL recvbuf at non-root; MPI ignores recvbuf there, so the computation
201            // is unaffected.
202            unsafe { ffi::ferrompi_reduce(p, p, n, dt, op, root, self.handle) }
203        };
204        Error::check_with_op(ret, "reduce_inplace")
205    }

212    /// * `recv` - Buffer for result
213    /// * `op` - a [`ReduceOp`], a `&`[`UserOp<T>`](crate::UserOp), or a [`CollectiveOp`]
214    ///
215    /// # Errors
216    ///
217    /// - [`Error::InvalidArgument`] with `arg: "op"` if `op` is a predefined op MPI does not
218    ///   define on `T` (a bitwise or logical op on `f32`/`f64`)
219    /// - [`Error::BufferSize`] if `send.len() != recv.len()`
220    /// - [`Error::Mpi`] with class [`MpiErrorClass::Count`](crate::MpiErrorClass::Count) if
221    ///   `op` is a user op and `send.len()` exceeds `i32::MAX`, on every MPI version
222    ///
223    /// # Example
224    ///
225    /// ```no_run
226    /// # use ferrompi::{Mpi, ReduceOp};
227    /// # let mpi = Mpi::init().unwrap();
228    /// # let world = mpi.world();
229    /// let send = vec![world.rank() as f64; 10];
230    /// let mut recv = vec![0.0f64; 10];
231    /// world.allreduce(&send, &mut recv, ReduceOp::Sum).unwrap();
232    /// ```
233    ///
234    /// A user-defined op is passed by reference:
235    ///
236    /// ```no_run
237    /// use ferrompi::{Mpi, UserOp};
238    ///
239    /// let mpi = Mpi::init().unwrap();
240    /// let world = mpi.world();
241    ///
242    /// let op: UserOp<f64> = UserOp::new(|invec: &[f64], inoutvec: &mut [f64]| {
243    ///     for (x, y) in invec.iter().zip(inoutvec.iter_mut()) {
244    ///         *y = x.max(*y);
245    ///     }
246    /// }).unwrap();
247    ///
248    /// let send = vec![world.rank() as f64 + 1.5_f64];
249    /// let mut recv = vec![0.0_f64];
250    /// world.allreduce(&send, &mut recv, &op).unwrap();
251    /// // recv[0] == (world.size() - 1) as f64 + 1.5
252    /// ```
253    ///
254    /// `MAX_LOC` is not defined on a primitive type:
255    ///
256    /// ```compile_fail,E0277
257    /// use ferrompi::{CollectiveOp, Mpi};
258    ///
259    /// let mpi = Mpi::init().unwrap();
260    /// let world = mpi.world();
261    ///
262    /// let send = [1.0f64];
263    /// let mut recv = [0.0f64];
264    /// world.allreduce(&send, &mut recv, CollectiveOp::MAX_LOC).unwrap();
265    /// ```
266    #[inline]

279        let ret = unsafe { ffi::ferrompi_allreduce(sp, rp, n, dt, op, self.handle) };
280        Error::check_with_op(ret, "allreduce")
281    }

344    /// All-reduce paired value+index types using `MPI_MAXLOC` or `MPI_MINLOC`.
345    ///
346    /// This method finds the global maximum (or minimum) value across all ranks
347    /// together with the rank index where it occurred. Only
348    /// [`CollectiveOp::MAX_LOC`] and [`CollectiveOp::MIN_LOC`] are accepted; any
349    /// other op on a pair type does not compile.
350    ///
351    /// The type parameter `T` must implement [`MpiIndexedDatatype`], which is
352    /// only satisfied by the six MPI predefined paired types: [`FloatInt`],
353    /// [`DoubleInt`], [`LongInt`], [`Int2`], [`ShortInt`], [`LongDoubleInt`].
354    /// These types are **not** interchangeable with the primitive types used by
355    /// `allreduce` — they are distinct at the type-system level.
356    ///
357    /// # Arguments
358    ///
359    /// * `send` - Slice of paired values contributed by this process
360    /// * `recv` - Output buffer; must be the same length as `send`
361    /// * `op` - [`CollectiveOp::MAX_LOC`] or [`CollectiveOp::MIN_LOC`]
362    ///
363    /// # Errors
364    ///
365    /// - [`Error::BufferSize`] if `send.len() != recv.len()`
366    /// - An MPI error if the call fails
367    ///
368    /// # Example
369    ///
370    /// ```no_run
371    /// use ferrompi::{CollectiveOp, DoubleInt, Mpi};
372    ///
373    /// let mpi = Mpi::init().unwrap();
374    /// let world = mpi.world();
375    /// let rank = world.rank();
376    ///
377    /// // Each rank contributes its rank as value and index.
378    /// let send = [DoubleInt { value: rank as f64, index: rank }];
379    /// let mut recv = [DoubleInt { value: 0.0, index: 0 }];
380    /// world.allreduce_indexed(&send, &mut recv, CollectiveOp::MAX_LOC).unwrap();
381    /// // Every rank now holds { value: (size-1) as f64, index: size-1 }
382    /// ```
383    ///
384    /// `Sum` is not defined on a pair type:
385    ///
386    /// ```compile_fail,E0277
387    /// use ferrompi::{DoubleInt, Mpi, ReduceOp};
388    ///
389    /// let mpi = Mpi::init().unwrap();
390    /// let world = mpi.world();
391    /// let rank = world.rank();
392    ///
393    /// let send = [DoubleInt { value: rank as f64, index: rank }];
394    /// let mut recv = [DoubleInt { value: 0.0, index: 0 }];
395    /// world.allreduce_indexed(&send, &mut recv, ReduceOp::Sum).unwrap();
396    /// ```
397    ///
398    /// [`FloatInt`]: crate::FloatInt
399    /// [`DoubleInt`]: crate::DoubleInt
400    /// [`LongInt`]: crate::LongInt
401    /// [`Int2`]: crate::Int2
402    /// [`ShortInt`]: crate::ShortInt
403    /// [`LongDoubleInt`]: crate::LongDoubleInt
404    pub fn allreduce_indexed<'a, T: MpiIndexedDatatype>(
405        &self,
406        send: &[T],
407        recv: &mut [T],
408        op: impl Into<CollectiveOp<'a, T>>,
409    ) -> Result<()> {
410        let op = op.into().code(pair_tag_of::<T>())?;
411        check_same_len("recv", send.len(), recv.len())?;
412        // SAFETY: send is a valid shared slice and recv is a valid exclusive slice of T
413        // (T: MpiIndexedDatatype — one of the six predefined MPI paired types). They cannot
414        // alias (Rust borrow rules). send.len() == recv.len() verified above. op is MAX_LOC
415        // or MIN_LOC, the only ops CollectiveOp offers for a pair type.
416        // The tag is T's MPI paired datatype. Slices outlive this call.
417        let ret = unsafe {
418            ffi::ferrompi_allreduce(
419                buf_ptr(send.as_ptr(), send.len(), false),
420                buf_ptr(recv.as_mut_ptr(), recv.len(), true).cast_mut(),
421                send.len() as i64,
422                pair_tag_of::<T>() as i32,
423                op,
424                self.handle,
425            )
426        };
427        Error::check_with_op(ret, "allreduce_indexed")
428    }

430    /// All-reduce arbitrary `Copy` types using `MPI_BYTE`-typed bitwise reductions.
431    ///
432    /// Performs a bitwise reduction across all ranks in the communicator. Each
433    /// element of `recv` receives the result of applying `op` element-wise across
434    /// the corresponding elements of each rank's `send` buffer.
435    ///
436    /// The buffer is transmitted as a flat array of bytes via `MPI_BYTE`, so the
437    /// count passed to MPI is `send.len() * size_of::<T>()`.
438    ///
439    /// Only `BitwiseOr`, `BitwiseAnd`, `BitwiseXor` are accepted. For
440    /// floating-point or indexed reductions, use [`allreduce`] or
441    /// [`allreduce_indexed`].
442    ///
443    /// # Arguments
444    ///
445    /// * `send` - Data contributed by this process
446    /// * `recv` - Output buffer; must be the same length as `send`
447    /// * `op` - Must be [`ReduceOp::BitwiseOr`], [`ReduceOp::BitwiseAnd`], or
448    ///   [`ReduceOp::BitwiseXor`]
449    ///
450    /// # Errors
451    ///
452    /// - [`Error::InvalidArgument`] if `op` is not one of the three bitwise ops
453    /// - [`Error::BufferSize`] if `send.len() != recv.len()`
454    /// - [`Error::Mpi`] if the MPI layer rejects the call
455    ///
456    /// # Example
457    ///
458    /// ```no_run
459    /// use ferrompi::{Mpi, ReduceOp};
460    ///
461    /// let mpi = Mpi::init().unwrap();
462    /// let world = mpi.world();
463    /// let rank = world.rank() as u64;
464    ///
465    /// // Each rank contributes a different bit; OR across all ranks gives 0b1111
466    /// let data: [u64; 4] = [1u64 << rank; 4];
467    /// let mut recv = [0u64; 4];
468    /// world.allreduce_bytes(&data, &mut recv, ReduceOp::BitwiseOr).unwrap();
469    /// assert_eq!(recv, [0b1111u64; 4]);
470    /// ```
471    ///
472    /// [`allreduce`]: Communicator::allreduce
473    /// [`allreduce_indexed`]: Communicator::allreduce_indexed
474    pub fn allreduce_bytes<T: BytePermutable>(
475        &self,
476        send: &[T],
477        recv: &mut [T],
478        op: ReduceOp,
479    ) -> Result<()> {
480        let op = Builtin::Reduce(op);
481        check_op(op, DatatypeTag::Byte)?;
482        check_same_len("recv", send.len(), recv.len())?;
483        let byte_count = std::mem::size_of_val(send);
484        // SAFETY:
485        // - send and recv are valid slices of T where T: BytePermutable (Copy + Send + 'static).
486        // - byte_count = size_of_val(send), which equals send.len() * size_of::<T>() and is
487        //   the exact memory footprint of each slice, at most isize::MAX. The cast to
488        //   *const c_void / *mut c_void is safe because we pass the byte count to MPI
489        //   (MPI_BYTE datatype), so MPI treats the buffer as raw bytes matching exactly
490        //   the memory of the slices.
491        // - DatatypeTag::Byte maps to MPI_BYTE in the C layer (case FERROMPI_BYTE).
492        // - send and recv do not alias (send is &[T], recv is &mut [T]).
493        let ret = unsafe {
494            ffi::ferrompi_allreduce(
495                buf_ptr(send.as_ptr(), send.len(), false),
496                buf_ptr(recv.as_mut_ptr(), recv.len(), true).cast_mut(),
497                byte_count as i64,
498                DatatypeTag::Byte as i32,
499                op.code(),
500                self.handle,
501            )
502        };
503        Error::check_with_op(ret, "allreduce_bytes")
504    }

506    /// Inclusive prefix reduction (scan).
507    ///
508    /// On rank `i`, `recv` contains the reduction of `send` values from ranks
509    /// `0..=i`. This is the inclusive variant: every rank's own contribution is
510    /// included in its result.
511    ///
512    /// # Arguments
513    ///
514    /// * `send` - Data to contribute from this process
515    /// * `recv` - Buffer for the prefix-reduced result (must be same length as `send`)
516    /// * `op` - a [`ReduceOp`], a `&`[`UserOp<T>`](crate::UserOp), or a [`CollectiveOp`]
517    ///
518    /// # Errors
519    ///
520    /// - [`Error::InvalidArgument`] with `arg: "op"` if `op` is a predefined op MPI does not
521    ///   define on `T` (a bitwise or logical op on `f32`/`f64`)
522    /// - [`Error::BufferSize`] if `send.len() != recv.len()`
523    /// - [`Error::Mpi`] with class [`MpiErrorClass::Count`](crate::MpiErrorClass::Count) if
524    ///   `op` is a user op and `send.len()` exceeds `i32::MAX`, on every MPI version
525    ///
526    /// # Example
527    ///
528    /// ```no_run
529    /// # use ferrompi::{Mpi, ReduceOp};
530    /// # let mpi = Mpi::init().unwrap();
531    /// # let world = mpi.world();
532    /// let send = vec![1.0f64; 10];
533    /// let mut recv = vec![0.0f64; 10];
534    /// world.scan(&send, &mut recv, ReduceOp::Sum).unwrap();
535    /// // On rank i, recv[j] == (i + 1) * send[j]
536    /// ```
537    pub fn scan<'a, T: MpiDatatype>(
538        &self,
539        send: &[T],
540        recv: &mut [T],
541        op: impl Into<CollectiveOp<'a, T>>,
542    ) -> Result<()> {
543        let op = op.into().code(tag_of::<T>())?;
544        check_same_len("recv", send.len(), recv.len())?;
545        let (sp, n, dt) = buf(send);
546        let (rp, _, _) = buf_mut(recv);
547        // SAFETY: send.len() == recv.len() is verified above; the two slices cannot alias
548        // (&[T] vs &mut [T]).
549        let ret = unsafe { ffi::ferrompi_scan(sp, rp, n, dt, op, self.handle) };
550        Error::check_with_op(ret, "scan")
551    }

553    /// Exclusive prefix reduction (exscan).
554    ///
555    /// On rank `i`, `recv` contains the reduction of `send` values from ranks
556    /// `0..i` (i.e., excluding rank `i`'s own contribution).
557    ///
558    /// # Rank 0 Behavior
559    ///
560    /// **Per the MPI standard, the contents of `recv` on rank 0 are undefined.**
561    /// Callers must not rely on the receive buffer contents on rank 0.
562    ///
563    /// # Arguments
564    ///
565    /// * `send` - Data to contribute from this process
566    /// * `recv` - Buffer for the prefix-reduced result (must be same length as `send`;
567    ///   **undefined on rank 0**)
568    /// * `op` - a [`ReduceOp`], a `&`[`UserOp<T>`](crate::UserOp), or a [`CollectiveOp`]
569    ///
570    /// # Errors
571    ///
572    /// - [`Error::InvalidArgument`] with `arg: "op"` if `op` is a predefined op MPI does not
573    ///   define on `T` (a bitwise or logical op on `f32`/`f64`)
574    /// - [`Error::BufferSize`] if `send.len() != recv.len()`
575    /// - [`Error::Mpi`] with class [`MpiErrorClass::Count`](crate::MpiErrorClass::Count) if
576    ///   `op` is a user op and `send.len()` exceeds `i32::MAX`, on every MPI version
577    ///
578    /// # Example
579    ///
580    /// ```no_run
581    /// # use ferrompi::{Mpi, ReduceOp};
582    /// # let mpi = Mpi::init().unwrap();
583    /// # let world = mpi.world();
584    /// let send = vec![1.0f64; 10];
585    /// let mut recv = vec![0.0f64; 10];
586    /// world.exscan(&send, &mut recv, ReduceOp::Sum).unwrap();
587    /// // On rank i > 0, recv[j] == i * send[j]
588    /// // On rank 0, recv is undefined per the MPI standard.
589    /// ```
590    pub fn exscan<'a, T: MpiDatatype>(
591        &self,
592        send: &[T],
593        recv: &mut [T],
594        op: impl Into<CollectiveOp<'a, T>>,
595    ) -> Result<()> {
596        let op = op.into().code(tag_of::<T>())?;
597        check_same_len("recv", send.len(), recv.len())?;
598        let (sp, n, dt) = buf(send);
599        let (rp, _, _) = buf_mut(recv);
600        // SAFETY: send.len() == recv.len() is verified above; the two slices cannot alias
601        // (&[T] vs &mut [T]). MPI leaves recv undefined on rank 0, documented above.
602        let ret = unsafe { ffi::ferrompi_exscan(sp, rp, n, dt, op, self.handle) };
603        Error::check_with_op(ret, "exscan")
604    }

672    /// Gather values to the root process.
673    ///
674    /// Each process sends `send.len()` elements. Root receives
675    /// `send.len() * size` elements total.
676    ///
677    /// # Arguments
678    ///
679    /// * `send` - Data to send from this process
680    /// * `recv` - Buffer for received data (only significant at root, must be at
681    ///   least `send.len() * size` elements)
682    /// * `root` - Rank of the root process
683    ///
684    /// # Errors
685    ///
686    /// [`Error::BufferSize`] if this rank is `root` and `recv.len() < send.len() *

688    ///
689    /// # Example
690    ///
691    /// ```no_run

693    /// # let mpi = Mpi::init().unwrap();

698    /// ```

700        if self.rank == root {
701            check_rank_slots("recv", recv.len(), send.len(), self.size)?;

704        let (rp, _, _) = buf_mut(recv);
705        // SAFETY: send and recv cannot alias (&[T] vs &mut [T]); at non-root, recv is ignored by
706        // MPI. The root-side receive-length relation (recv.len() >= send.len() * size) is
707        // checked above.
708        let ret = unsafe { ffi::ferrompi_gather(sp, n, rp, n, dt, root, self.handle) };
709        Error::check_with_op(ret, "gather")
710    }

738    /// Gather values in place. At root, `data` is both the send contribution and the
739    /// receive buffer. At non-root, `data` is that rank's own block (here its send
740    /// block), as for [`scatter_inplace`](Self::scatter_inplace) and
741    /// [`reduce_inplace`](Self::reduce_inplace).
742    ///
743    /// # Buffer Layout
744    ///
745    /// At root, `data` must have length `recvcount * size()` where `recvcount` is the
746    /// per-rank count. Rank `r`'s contribution lives at offset `r * recvcount`. Root's own
747    /// contribution must be pre-written into `data[rank() * recvcount .. (rank()+1) *
748    /// recvcount]`.
749    ///
750    /// At non-root, `data` is the rank's block of `recvcount` elements. Only root knows
751    /// `recvcount`, so a block of another length is not checked locally; MPI reports the
752    /// mismatch.
753    ///
754    /// # Errors
755    ///
756    /// `Error::InvalidArgument` at root if `data.len()` is not divisible by `size()`.
757    ///
758    /// # Example
759    ///
760    /// ```no_run
761    /// # use ferrompi::Mpi;
762    /// # let mpi = Mpi::init().unwrap();
763    /// # let world = mpi.world();
764    /// let rank = world.rank() as usize;
765    /// let size = world.size() as usize;
766    /// // Root allocates the full buffer; each rank's slot is at offset rank * recvcount.
767    /// // recvcount = 1 in this example; every other rank passes its own block.
768    /// if world.rank() == 0 {
769    ///     let mut data = vec![0i32; size]; // slot 0..size
770    ///     data[rank] = rank as i32 * 10;   // root pre-writes its own slot
771    ///     world.gather_inplace(&mut data, 0).unwrap();
772    ///     // data[r] == r * 10 for all r
773    /// } else {
774    ///     let mut data = vec![rank as i32 * 10];
775    ///     world.gather_inplace(&mut data, 0).unwrap();
776    /// }
777    /// ```
778    pub fn gather_inplace<T: MpiDatatype>(&self, data: &mut [T], root: i32) -> Result<()> {
779        let ret = if self.rank() == root {
780            let recvcount = rank_block("data", data.len(), self.size)? as i64;
781            let (p, _, dt) = buf_mut(data);
782            // SAFETY: NULL sendbuf is the in-place marker (buf_mut's pointer is never null, so
783            // this NULL is unambiguous); ferrompi_gather maps it to MPI_IN_PLACE, so data serves
784            // as both root's send contribution (at offset rank*recvcount) and the receive
785            // buffer. recvcount is checked to evenly divide data.len() above, and this branch
786            // runs only at root, the only rank MPI_IN_PLACE is valid for in MPI_Gather.
787            unsafe {
788                ffi::ferrompi_gather(std::ptr::null(), 0, p, recvcount, dt, root, self.handle)
789            }
790        } else {
791            let (p, n, dt) = buf_mut(data);
792            // SAFETY: p is non-null, so ferrompi_gather does not map it to MPI_IN_PLACE; data is
793            // this rank's send block. It is also passed as recvbuf because strict MPI builds
794            // reject a NULL recvbuf at non-root; MPI ignores recvbuf and recvcount there, so the
795            // gather is unaffected.
796            unsafe { ffi::ferrompi_gather(p, n, p, n, dt, root, self.handle) }
797        };
798        Error::check_with_op(ret, "gather_inplace")
799    }

806    /// `data` must have length `recvcount * size()`. Rank `r`'s contribution lives at

839    /// Scatter values in place. At root, `data` is the `sendcount * size()` send buffer;
840    /// root's own slot is retained in place. At non-root, `data` is the
841    /// `recvcount`-element receive buffer.
842    ///
843    /// # Buffer Layout (root)
844    ///
845    /// `data` must have length `sendcount * size()`. Rank `r`'s slot is
846    /// `data[r*sendcount .. (r+1)*sendcount]`. After the call, only root's own slot is
847    /// guaranteed to remain intact; other slots are unspecified.
848    ///
849    /// # Errors
850    ///
851    /// Returns `Error::InvalidArgument` at root if `data.len()` is not divisible by
852    /// `size()`.
853    ///
854    /// # Example
855    ///
856    /// ```no_run
857    /// # use ferrompi::Mpi;
858    /// # let mpi = Mpi::init().unwrap();
859    /// # let world = mpi.world();
860    /// // With 4 ranks: root pre-populates [0, 10, 20, 30], each non-root has a 1-element buf.
861    /// // After the call: rank 0 retains data[0]==0, rank 1 gets [10], rank 2 [20], rank 3 [30].
862    /// if world.rank() == 0 {
863    ///     let mut data = vec![0i32, 10, 20, 30];
864    ///     world.scatter_inplace(&mut data, 0).unwrap();
865    ///     assert_eq!(data[0], 0); // root retains its own slot
866    /// } else {
867    ///     let mut data = vec![0i32; 1];
868    ///     world.scatter_inplace(&mut data, 0).unwrap();
869    /// }
870    /// ```
871    pub fn scatter_inplace<T: MpiDatatype>(&self, data: &mut [T], root: i32) -> Result<()> {
872        let (sendbuf, sendcount, recvbuf, recvcount, dt) =
873            scatter_inplace_args(data, self.rank() == root, self.size)?;
874        // SAFETY: at root, recvbuf is NULL, the in-place marker (buf's pointer is never null,
875        // so this NULL is unambiguous); ferrompi_scatter maps it to MPI_IN_PLACE so root's own
876        // slot is retained. scatter_inplace_args checks that the block size evenly divides
877        // data.len(). At non-root, sendbuf is null, which the MPI standard ignores on non-root
878        // scatter.
879        let ret = unsafe {
880            ffi::ferrompi_scatter(
881                sendbuf,
882                sendcount,
883                recvbuf,
884                recvcount,
885                dt,
886                root,
887                self.handle,
888            )
889        };
890        Error::check_with_op(ret, "scatter_inplace")
891    }

1007    /// Reduce-scatter with uniform block size.
1008    ///
1009    /// Performs an element-wise reduction across all processes, then scatters
1010    /// the result so that each process receives `recv.len()` elements.
1011    /// `send` must have exactly `recv.len() * size` elements.
1012    ///
1013    /// This is equivalent to [`allreduce`](Self::allreduce) followed by each
1014    /// process keeping only its portion, but is more efficient because the MPI
1015    /// implementation can fuse the two operations.
1016    ///
1017    /// # Errors
1018    ///
1019    /// - [`Error::InvalidArgument`] with `arg: "op"` if `op` is a predefined op MPI does not
1020    ///   define on `T` (a bitwise or logical op on `f32`/`f64`)
1021    /// - [`Error::InvalidArgument`] if `send.len()` is not evenly divisible by the
1022    ///   communicator size
1023    /// - [`Error::BufferSize`] if `send.len() != recv.len() * size`
1024    /// - [`Error::Mpi`] with class [`MpiErrorClass::Count`](crate::MpiErrorClass::Count) if
1025    ///   `op` is a user op and `send.len()` exceeds `i32::MAX`, on every MPI version
1026    ///
1027    /// # Example
1028    ///
1029    /// ```no_run
1030    /// # use ferrompi::{Mpi, ReduceOp};
1031    /// # let mpi = Mpi::init().unwrap();
1032    /// # let world = mpi.world();
1033    /// let size = world.size() as usize;
1034    /// let send = vec![1.0f64; size * 5];
1035    /// let mut recv = vec![0.0f64; 5];
1036    /// world.reduce_scatter_block(&send, &mut recv, ReduceOp::Sum).unwrap();
1037    /// ```
1038    pub fn reduce_scatter_block<'a, T: MpiDatatype>(
1039        &self,
1040        send: &[T],
1041        recv: &mut [T],
1042        op: impl Into<CollectiveOp<'a, T>>,
1043    ) -> Result<()> {
1044        let op = op.into().code(tag_of::<T>())?;
1045        check_same_len(
1046            "recv",
1047            rank_block("send", send.len(), self.size)?,
1048            recv.len(),
1049        )?;
1050        let (sp, _, _) = buf(send);
1051        let (rp, n, dt) = buf_mut(recv);
1052        // SAFETY: send and recv cannot alias (&[T] vs &mut [T]). send.len() == recv.len() * size
1053        // is verified above.
1054        let ret = unsafe { ffi::ferrompi_reduce_scatter_block(sp, rp, n, dt, op, self.handle) };
1055        Error::check_with_op(ret, "reduce_scatter_block")
1056    }
1057}
