19/// Checks that a buffer of `whole` elements holds one `block`-element slot per
20/// rank of a `size`-rank communicator (`whole >= block * size`, with the product
21/// computed by checked arithmetic).
22fn check_rank_slots(whole: usize, block: usize, size: i32) -> Result<()> {
23    let needed = block
24        .checked_mul(size as usize)
25        .ok_or(Error::InvalidBuffer)?;
26    if whole < needed {
27        return Err(Error::InvalidBuffer);
28    }
29    Ok(())
30}

32/// Checks that two buffers hold the same number of elements.
33fn check_same_len(a: usize, b: usize) -> Result<()> {
34    if a != b {
35        return Err(Error::InvalidBuffer);
36    }
37    Ok(())
38}

40/// Returns the per-rank block count of a buffer of `whole` elements split
41/// evenly across a `size`-rank communicator; Err when `size <= 0` or
42/// `whole % size != 0`.
43fn rank_block(whole: usize, size: i32) -> Result<usize> {
44    if size <= 0 {
45        return Err(Error::InvalidBuffer);
46    }
47    let size = size as usize;
48    if whole % size != 0 {
49        return Err(Error::InvalidBuffer);
50    }
51    Ok(whole / size)
52}

54/// Arguments of an in-place scatter as `(sendbuf, sendcount, recvbuf, recvcount, tag)`:
55/// at root, `data` is the send buffer split in `size` blocks and the NULL
56/// receive buffer is the in-place marker; elsewhere, `data` is the receive
57/// buffer and the send side is NULL and ignored.
58fn scatter_inplace_args<T: MpiDatatype>(
59    data: &mut [T],
60    is_root: bool,
61    size: i32,
62) -> Result<(*const c_void, i64, *mut c_void, i64, i32)> {
63    if is_root {
64        let per = rank_block(data.len(), size)? as i64;
65        let (sp, _, dt) = buf(data);
66        Ok((sp, per, std::ptr::null_mut::<std::ffi::c_void>(), 0i64, dt))
67    } else {
68        let (rp, rn, dt) = buf_mut(data);
69        Ok((std::ptr::null::<std::ffi::c_void>(), 0i64, rp, rn, dt))
70    }
71}

85/// An MPI communicator.
86///
87/// This type wraps an MPI communicator handle and provides safe methods for
88/// collective and point-to-point communication operations.
89///
90/// # Thread Safety
91///
92/// `Communicator` is `Send + Sync`, matching the thread-safety model of
93/// C/Fortran MPI. The actual thread-safety guarantees depend on the
94/// thread level provided by [`Mpi::init_thread()`](crate::Mpi::init_thread):
95///
96/// - [`ThreadLevel::Single`](crate::ThreadLevel::Single) /
97///   [`Funneled`](crate::ThreadLevel::Funneled): MPI calls only from main thread
98/// - [`ThreadLevel::Serialized`](crate::ThreadLevel::Serialized): MPI calls
99///   from any thread, but serialized by the user (e.g., via a `Mutex`)
100/// - [`ThreadLevel::Multiple`](crate::ThreadLevel::Multiple): MPI calls from
101///   any thread concurrently without external synchronization
102///
103/// For hybrid MPI + threads programs, request at least
104/// [`ThreadLevel::Funneled`](crate::ThreadLevel::Funneled) (master thread
105/// makes MPI calls) or [`ThreadLevel::Serialized`](crate::ThreadLevel::Serialized)
106/// (any thread, but only one at a time).
107///
108/// # Argument validation
109///
110/// Collective methods that document an [`Error::InvalidBuffer`] condition check it
111/// on the calling rank before any MPI call. The check is local: ranks that already
112/// entered the collective are not told about the failure and may block.
113///
114/// # Errors in collective calls
115///
116/// MPI may return an error from a collective call on some ranks only. The
117/// ranks that returned `Ok` go on to their next collective call and can
118/// block there, waiting for a rank that returned `Err`. Methods that
119/// document an error "on every rank" guarantee that for their own checks
120/// once their collective calls have succeeded on every rank.
121///
122/// # Example
123///
124/// ```no_run
125/// use ferrompi::Mpi;
126///
127/// let mpi = Mpi::init().unwrap();
128/// let world = mpi.world();
129///
130/// println!("I am rank {} of {}", world.rank(), world.size());
131/// ```
132pub struct Communicator {
133    pub(crate) handle: i32,
134    pub(crate) rank: i32,
135    pub(crate) size: i32,
136}
137
138// SAFETY: Communicator handles are integer indices into a C-side table.
139// Every MPI call goes through the lifecycle guard (`rt::enter`): below
140// `ThreadLevel::Serialized` (Single or Funneled), it rejects a call from any
141// thread other than the one that called `Mpi::init`/`init_thread`, with
142// `Err(Error::ThreadLevelViolation)`, without touching MPI. A `Drop` cannot
143// return `Err`, so the equivalent guard (`rt::drop_guard`) instead aborts the
144// process on a wrong-thread drop below `Serialized`. At `Serialized`, the
145// caller must serialize its own calls (debug builds detect two overlapping
146// calls). At `Multiple`, calls from any thread may run concurrently.
147//
148// The C layer's handle tables claim slots lock-free: six with a C11
149// atomic compare-exchange, the request table with an atomic bitmap
150// (`fetch_or`), so there are no data races under MPI_THREAD_MULTIPLE.
151// See docs/adr/0002-handle-tables.md for the full rationale and design.
152unsafe impl Send for Communicator {}
153// SAFETY: &Communicator exposes only reads of immutable fields and FFI calls
154// gated by the same lifecycle guard described above, so its thread safety
155// follows the requested `ThreadLevel` the same way.
156unsafe impl Sync for Communicator {}

158/// Handle of `MPI_COMM_WORLD`: slot 0 of the C communicator table, set at
159/// init and never freed.
160const WORLD_HANDLE: i32 = 0;
161
162impl Communicator {
163    /// Constant for opting out of a communicator split.
164    ///
165    /// Processes passing this as the `color` to [`split()`](Self::split) will not be
166    /// included in any resulting communicator.
167    pub const UNDEFINED: i32 = -1;
168
169    /// Get a handle to `MPI_COMM_WORLD`.
170    pub(crate) fn world() -> Self {
171        Self::from_handle(WORLD_HANDLE).expect("COMM_WORLD must be valid post-init")
172    }
173
174    /// Construct a `Communicator` by querying rank and size once from MPI.
175    ///
176    /// This is the canonical internal constructor. All construction sites use it
177    /// so that `rank` and `size` are cached at creation time and subsequent calls
178    /// to [`rank()`](Self::rank) / [`size()`](Self::size) require no FFI round-trip.
179    ///
180    /// Returns `Err` if either `MPI_Comm_rank` or `MPI_Comm_size` fails.
181    pub(crate) fn from_handle(handle: i32) -> Result<Self> {
182        let mut rank: i32 = 0;
183        // SAFETY: `handle` is a valid MPI communicator handle obtained from MPI
184        // functions. `rank` is a local variable so the pointer is valid for writes.
185        let ret = unsafe { ffi::ferrompi_comm_rank(handle, &mut rank) };
186        Error::check_with_op(ret, "comm_rank")?;
187        let mut size: i32 = 0;
188        // SAFETY: same as above — `size` is a local variable, `handle` is valid.
189        let ret = unsafe { ffi::ferrompi_comm_size(handle, &mut size) };
190        Error::check_with_op(ret, "comm_size")?;
191        Ok(Communicator { handle, rank, size })
192    }
193
194    /// Get the raw communicator handle (for advanced use).
195    pub fn raw_handle(&self) -> i32 {
196        self.handle
197    }
198
199    /// Get the rank of the calling process in this communicator.
200    ///
201    /// Returns the cached value stored at construction time. No FFI call is made.
202    #[inline]
203    pub fn rank(&self) -> i32 {
204        self.rank
205    }
206
207    /// Get the number of processes in this communicator.
208    ///
209    /// Returns the cached value stored at construction time. No FFI call is made.
210    #[inline]
211    pub fn size(&self) -> i32 {
212        self.size
213    }
214}

216impl Drop for Communicator {
217    fn drop(&mut self) {
218        // Don't free COMM_WORLD
219        if self.handle != WORLD_HANDLE {
220            if !rt::drop_guard("Communicator") {
221                return;
222            }
223            // SAFETY: self.handle is a valid, non-zero communicator handle
224            // registered in the C-side comm table (checked above); Drop takes
225            // &mut self and runs at most once per value, so this cannot
226            // double-free the handle.
227            unsafe { ffi::ferrompi_comm_free(self.handle) };
228        }
229    }
230}
