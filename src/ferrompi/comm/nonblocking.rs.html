18    /// Nonblocking broadcast.
19    ///
20    /// Initiates the broadcast and returns immediately with a [`Request`] that
21    /// belongs to the scope `s`. `data` is borrowed mutably for the scope: it
22    /// cannot be read or changed until [`scope`](crate::scope) returns, even
23    /// after the request was waited, and the scope completes the broadcast at
24    /// the latest then.
25    ///
26    /// # Arguments
27    ///
28    /// * `s` - The scope that owns the request
29    /// * `data` - The payload at the root, and the buffer that receives it at every
30    ///   other rank, borrowed for the scope
31    /// * `root` - Rank of the root process
32    ///
33    /// # Example
34    ///
35    /// ```no_run
36    /// # use ferrompi::Mpi;
37    /// # let mpi = Mpi::init().unwrap();
38    /// # let world = mpi.world();
39    /// let mut data = vec![0.0f64; 100];
40    /// ferrompi::scope(|s| {
41    ///     let req = world.ibroadcast(s, &mut data, 0)?;
42    ///     // ... do other work ...
43    ///     req.wait()?;
44    ///     Ok(())
45    /// })
46    /// .unwrap();
47    /// // data is readable again here.
48    /// ```
49    #[inline]
50    pub fn ibroadcast<'s, T: MpiDatatype>(
51        &self,
52        s: &'s Scope<'s, '_>,
53        data: &'s mut [T],
54        root: i32,
55    ) -> Result<Request<'s>> {
56        let mut request_handle: i64 = 0;
57        let (p, n, dt) = buf_mut(data);
58        // SAFETY: data is borrowed mutably for 's. The scope completes every request it holds
59        // before 's ends, so the buffer outlives the span in which MPI may use it, and nothing
60        // touches it meanwhile.
61        let ret = unsafe { ffi::ferrompi_ibcast(p, n, dt, root, self.handle, &mut request_handle) };
62        Error::check_with_op(ret, "ibcast")?;
63        Ok(s.request(request_handle, RequestKind::Collective))
64    }

66    /// Nonblocking all-reduce.
67    ///
68    /// Initiates the reduction and returns immediately with a [`Request`] that
69    /// belongs to the scope `s`. `send` and a user `op` are borrowed for the
70    /// scope, and `recv` is borrowed mutably for it: `recv` cannot be read until
71    /// [`scope`](crate::scope) returns, even after the request was waited, and
72    /// the scope completes the reduction at the latest then.
73    ///
74    /// # Arguments
75    ///
76    /// * `s` - The scope that owns the request
77    /// * `send` - Data to send from this process, borrowed for the scope
78    /// * `recv` - Buffer for the result, borrowed for the scope
79    /// * `op` - a [`ReduceOp`](crate::ReduceOp), a `&`[`UserOp<T>`](crate::UserOp) (borrowed
80    ///   until the scope returns), or a [`CollectiveOp`]
81    ///
82    /// # Errors
83    ///
84    /// - [`Error::InvalidArgument`] with `arg: "op"` if `op` is a predefined op MPI does not
85    ///   define on `T` (a bitwise or logical op on `f32`/`f64`)
86    /// - [`Error::BufferSize`] if `send.len() != recv.len()`
87    /// - [`Error::Mpi`] with class [`MpiErrorClass::Count`](crate::MpiErrorClass::Count) if
88    ///   `op` is a user op and `send.len()` exceeds `i32::MAX`, on every MPI version
89    ///
90    /// # Example
91    ///
92    /// ```no_run
93    /// # use ferrompi::{Mpi, ReduceOp};
94    /// # let mpi = Mpi::init().unwrap();
95    /// # let world = mpi.world();
96    /// let send = vec![1.0f64; 10];
97    /// let mut recv = vec![0.0f64; 10];
98    /// ferrompi::scope(|s| {
99    ///     let req = world.iallreduce(s, &send, &mut recv, ReduceOp::Sum)?;
100    ///     // ... do other work ...
101    ///     req.wait()?;
102    ///     Ok(())
103    /// })
104    /// .unwrap();
105    /// // recv is readable again here.
106    /// ```
107    #[inline]
108    pub fn iallreduce<'s, T: MpiDatatype>(
109        &self,
110        s: &'s Scope<'s, '_>,
111        send: &'s [T],
112        recv: &'s mut [T],
113        op: impl Into<CollectiveOp<'s, T>>,
114    ) -> Result<Request<'s>> {
115        let op = op.into().code(tag_of::<T>())?;
116        check_same_len("recv", send.len(), recv.len())?;
117        let mut request_handle: i64 = 0;
118        let (sp, n, dt) = buf(send);
119        let (rp, _, _) = buf_mut(recv);
120        // SAFETY: send.len() == recv.len() is verified above; the two slices cannot alias
121        // (&[T] vs &mut [T]). send and recv are borrowed for 's (recv mutably), and a user op
122        // is borrowed for 's through `op`. The scope completes every request it holds before 's
123        // ends, so the buffers and the op outlive the span in which MPI may use them, and
124        // nothing touches the buffers meanwhile.
125        let ret = unsafe {
126            ffi::ferrompi_iallreduce(sp, rp, n, dt, op, self.handle, &mut request_handle)
127        };
128        Error::check_with_op(ret, "iallreduce")?;
129        Ok(s.request(request_handle, RequestKind::Collective))
130    }

156    /// - [`Error::Mpi`] with class [`MpiErrorClass::Count`](crate::MpiErrorClass::Count) if

166    /// let mut recv = vec![0.0f64; 10];
167    /// ferrompi::scope(|s| {
168    ///     let req = world.ireduce(s, &send, &mut recv, ReduceOp::Sum, 0)?;
169    ///     // ... do other work ...
170    ///     req.wait()?;
171    ///     Ok(())
172    /// })

174    /// // recv is readable again here.

177    pub fn ireduce<'s, T: MpiDatatype>(
178        &self,
179        s: &'s Scope<'s, '_>,
180        send: &'s [T],
181        recv: &'s mut [T],

184    ) -> Result<Request<'s>> {

191        let (rp, _, _) = buf_mut(recv);

193        // every other rank (MPI-4.1 section 6.9.1). The two slices cannot alias (&[T] vs
194        // &mut [T]). send and recv are borrowed for 's (recv mutably), and a user op is borrowed
195        // for 's through `op`. The scope completes every request it holds before 's ends, so the
196        // buffers and the op outlive the span in which MPI may use them, and nothing touches the
197        // buffers meanwhile.

199            ffi::ferrompi_ireduce(sp, rp, n, dt, op, root, self.handle, &mut request_handle)
200        };
201        Error::check_with_op(ret, "ireduce")?;
202        Ok(s.request(request_handle, RequestKind::Collective))
203    }

205    /// Nonblocking gather to root.
206    ///
207    /// Initiates a gather operation and returns immediately with a [`Request`]
208    /// that belongs to the scope `s`. Each process sends `send.len()` elements.
209    /// Root receives `send.len() * size` elements total. `send` is borrowed for
210    /// the scope, and `recv` is borrowed mutably for it: `recv` cannot be read
211    /// until [`scope`](crate::scope) returns, even after the request was
212    /// waited, and the scope completes the gather at the latest then.
213    ///
214    /// # Arguments
215    ///
216    /// * `s` - The scope that owns the request
217    /// * `send` - Data to send from this process, borrowed for the scope
218    /// * `recv` - Buffer for received data (only significant at root), borrowed for
219    ///   the scope on every rank
220    /// * `root` - Rank of the root process
221    ///
222    /// # Errors
223    ///
224    /// [`Error::BufferSize`] if this rank is `root` and `recv.len() < send.len() *
225    /// size()`.
226    ///
227    /// # Example
228    ///
229    /// ```no_run
230    /// # use ferrompi::Mpi;
231    /// # let mpi = Mpi::init().unwrap();
232    /// # let world = mpi.world();
233    /// let send = vec![world.rank() as f64; 5];
234    /// let mut recv = vec![0.0f64; 5 * world.size() as usize];
235    /// ferrompi::scope(|s| {
236    ///     let req = world.igather(s, &send, &mut recv, 0)?;
237    ///     req.wait()?;
238    ///     Ok(())
239    /// })
240    /// .unwrap();
241    /// // recv is readable again here.
242    /// ```
243    #[inline]
244    pub fn igather<'s, T: MpiDatatype>(
245        &self,
246        s: &'s Scope<'s, '_>,
247        send: &'s [T],
248        recv: &'s mut [T],
249        root: i32,
250    ) -> Result<Request<'s>> {
251        if self.rank == root {
252            check_rank_slots("recv", recv.len(), send.len(), self.size)?;
253        }
254        let mut request_handle: i64 = 0;
255        let (sp, n, dt) = buf(send);
256        let (rp, _, _) = buf_mut(recv);
257        // SAFETY: send and recv cannot alias (&[T] vs &mut [T]); at non-root, recv is
258        // ignored by MPI. The root-side receive-length relation (recv.len() >= send.len()
259        // * size) is checked above. send and recv are borrowed for 's (recv mutably). The
260        // scope completes every request it holds before 's ends, so the buffers outlive the
261        // span in which MPI may use them, and nothing touches them meanwhile.
262        let ret = unsafe {
263            ffi::ferrompi_igather(sp, n, rp, n, dt, root, self.handle, &mut request_handle)
264        };
265        Error::check_with_op(ret, "igather")?;
266        Ok(s.request(request_handle, RequestKind::Collective))
267    }

518    ///   `op` is a user op and `send.len()` exceeds `i32::MAX`, on every MPI version
519    ///
520    /// # Example
521    ///
522    /// ```no_run

524    /// # let mpi = Mpi::init().unwrap();

527    /// let mut recv = vec![0.0f64; 10];
528    /// ferrompi::scope(|s| {
529    ///     let req = world.iexscan(s, &send, &mut recv, ReduceOp::Sum)?;
530    ///     // ... do other work ...
531    ///     req.wait()?;
532    ///     Ok(())
533    /// })
534    /// .unwrap();
535    /// // On rank i > 0, recv[j] == i * send[j]
536    /// // On rank 0, recv is undefined per the MPI standard.
537    /// ```
538    #[inline]
539    pub fn iexscan<'s, T: MpiDatatype>(
540        &self,
541        s: &'s Scope<'s, '_>,
542        send: &'s [T],
543        recv: &'s mut [T],
544        op: impl Into<CollectiveOp<'s, T>>,
545    ) -> Result<Request<'s>> {
546        let op = op.into().code(tag_of::<T>())?;
547        check_same_len("recv", send.len(), recv.len())?;
548        let mut request_handle: i64 = 0;
549        let (sp, n, dt) = buf(send);
550        let (rp, _, _) = buf_mut(recv);
551        // SAFETY: send.len() == recv.len() is verified above; the two slices cannot alias
552        // (&[T] vs &mut [T]). send and recv are borrowed for 's (recv mutably), and a user op
553        // is borrowed for 's through `op`. The scope completes every request it holds before 's
554        // ends, so the buffers and the op outlive the span in which MPI may use them, and
555        // nothing touches the buffers meanwhile. MPI leaves recv undefined on rank 0,
556        // documented above.
557        let ret =
558            unsafe { ffi::ferrompi_iexscan(sp, rp, n, dt, op, self.handle, &mut request_handle) };
559        Error::check_with_op(ret, "iexscan")?;
560        Ok(s.request(request_handle, RequestKind::Collective))
561    }

670    /// .unwrap();
671    /// // recv is readable again here.
672    /// ```

702    /// Nonblocking in-place gather. At root, `data` is both the send contribution
703    /// and the receive buffer. At non-root, `data` is that rank's own block (here its
704    /// send block), as for [`iscatter_inplace`](Self::iscatter_inplace).
705    ///
706    /// The returned [`Request`] belongs to the scope `s`. `data` is borrowed
707    /// mutably for the scope: it cannot be read or changed until
708    /// [`scope`](crate::scope) returns, even after the request was waited, and
709    /// the scope completes the gather at the latest then.
710    ///
711    /// # Buffer Layout
712    ///
713    /// At root, `data` must have length `recvcount * size()` where `recvcount` is
714    /// the per-rank count. Rank `r`'s contribution lives at offset
715    /// `r * recvcount`. Root's own contribution must be pre-written into
716    /// `data[rank() * recvcount .. (rank()+1) * recvcount]` before the call.
717    ///
718    /// At non-root, `data` is the rank's block of `recvcount` elements. Only root
719    /// knows `recvcount`, so a block of another length is not checked locally; MPI
720    /// reports the mismatch.
721    ///
722    /// # Errors
723    ///
724    /// `Error::InvalidArgument` at root if `data.len()` is not divisible by `size()`.
725    ///
726    /// # Example
727    ///
728    /// ```no_run
729    /// # use ferrompi::Mpi;
730    /// # let mpi = Mpi::init().unwrap();
731    /// # let world = mpi.world();
732    /// let mut data = if world.rank() == 0 {
733    ///     vec![0i32; 4 * world.size() as usize]
734    /// } else {
735    ///     vec![world.rank(); 4]
736    /// };
737    /// ferrompi::scope(|s| {
738    ///     let req = world.igather_inplace(s, &mut data, 0)?;
739    ///     req.wait()?;
740    ///     Ok(())
741    /// })
742    /// .unwrap();
743    /// ```
744    #[inline]
745    pub fn igather_inplace<'s, T: MpiDatatype>(
746        &self,
747        s: &'s Scope<'s, '_>,
748        data: &'s mut [T],
749        root: i32,
750    ) -> Result<Request<'s>> {
751        let mut request_handle: i64 = 0;
752        let ret = if self.rank() == root {
753            let recvcount = rank_block("data", data.len(), self.size)? as i64;
754            let (p, _, dt) = buf_mut(data);
755            // SAFETY: NULL sendbuf is the in-place marker (buf_mut's pointer is never null, so
756            // this NULL is unambiguous); ferrompi_igather maps it to MPI_IN_PLACE, so data
757            // serves as both root's send contribution and the receive buffer. recvcount is
758            // checked to evenly divide data.len() above, and this branch runs only at root,
759            // the only rank MPI_IN_PLACE is valid for in MPI_Igather. data is borrowed mutably
760            // for 's. The scope completes every request it holds before 's ends, so the buffer
761            // outlives the span in which MPI may use it, and nothing touches it meanwhile.
762            unsafe {
763                ffi::ferrompi_igather(
764                    std::ptr::null(),
765                    0,
766                    p,
767                    recvcount,
768                    dt,
769                    root,
770                    self.handle,
771                    &mut request_handle,
772                )
773            }
774        } else {
775            let (p, n, dt) = buf_mut(data);
776            // SAFETY: p is non-null, so ferrompi_igather does not map it to MPI_IN_PLACE; data is
777            // this rank's send block. It is also passed as recvbuf because strict MPI builds
778            // reject a NULL recvbuf at non-root; MPI ignores recvbuf and recvcount there. data
779            // is borrowed mutably for 's. The scope completes every request it holds before 's
780            // ends, so the buffer outlives the span in which MPI may use it, and nothing
781            // touches it meanwhile.
782            unsafe { ffi::ferrompi_igather(p, n, p, n, dt, root, self.handle, &mut request_handle) }
783        };
784        Error::check_with_op(ret, "igather_inplace")?;
785        Ok(s.request(request_handle, RequestKind::Collective))
786    }

1060
1061    /// Nonblocking in-place reduce to the root process.
1062    ///
1063    /// At root, `data` is both the input and the result. At every other rank
1064    /// `data` is the send buffer only, as for
1065    /// [`reduce_inplace`](Self::reduce_inplace).
1066    ///
1067    /// The returned [`Request`] belongs to the scope `s`. `data` is borrowed
1068    /// mutably for the scope on every rank, and a user `op` is borrowed for it:
1069    /// `data` cannot be read or changed until [`scope`](crate::scope) returns,
1070    /// even after the request was waited, and the scope completes the reduction
1071    /// at the latest then.
1072    ///
1073    /// # Arguments
1074    ///
1075    /// * `s` - The scope that owns the request
1076    /// * `data` - This process's values on entry; at the root, the reduced
1077    ///   values once the request completed. Borrowed for the scope
1078    /// * `op` - a [`ReduceOp`](crate::ReduceOp), a `&`[`UserOp<T>`](crate::UserOp) (borrowed
1079    ///   until the scope returns), or a [`CollectiveOp`]
1080    /// * `root` - Rank of the root process
1081    ///
1082    /// # Errors
1083    ///
1084    /// - [`Error::InvalidArgument`] with `arg: "op"` if `op` is a predefined op MPI does not
1085    ///   define on `T` (a bitwise or logical op on `f32`/`f64`)
1086    /// - [`Error::Mpi`] with class [`MpiErrorClass::Count`](crate::MpiErrorClass::Count) if
1087    ///   `op` is a user op and `data.len()` exceeds `i32::MAX`, on every MPI version
1088    ///
1089    /// # Example
1090    ///
1091    /// ```no_run
1092    /// # use ferrompi::{Mpi, ReduceOp};
1093    /// # let mpi = Mpi::init().unwrap();
1094    /// # let world = mpi.world();
1095    /// let mut data = vec![world.rank() as f64; 10];
1096    /// ferrompi::scope(|s| {
1097    ///     let req = world.ireduce_inplace(s, &mut data, ReduceOp::Sum, 0)?;
1098    ///     // ... do other work ...
1099    ///     req.wait()?;
1100    ///     Ok(())
1101    /// })
1102    /// .unwrap();
1103    /// if world.rank() == 0 {
1104    ///     println!("Reduced result: {:?}", &data[..3]);
1105    /// }
1106    /// ```
1107    #[inline]
1108    pub fn ireduce_inplace<'s, T: MpiDatatype>(
1109        &self,
1110        s: &'s Scope<'s, '_>,
1111        data: &'s mut [T],
1112        op: impl Into<CollectiveOp<'s, T>>,
1113        root: i32,
1114    ) -> Result<Request<'s>> {
1115        let op = op.into().code(tag_of::<T>())?;
1116        let mut request_handle: i64 = 0;
1117        let (p, n, dt) = buf_mut(data);
1118        let sendbuf = if self.rank == root {
1119            std::ptr::null()
1120        } else {
1121            p.cast_const()
1122        };
1123        // SAFETY: at root, NULL sendbuf is the in-place marker (buf_mut's pointer is never
1124        // null, so this NULL is unambiguous); ferrompi_ireduce maps it to MPI_IN_PLACE, so data
1125        // is both input and output there. At every other rank data is the send buffer, and it
1126        // is passed as recvbuf too, as reduce_inplace does: recvbuf is significant only at the
1127        // root (MPI-4.1 section 6.9.1), so a non-root rank's data is only read. data is borrowed
1128        // mutably for 's, and a user op is borrowed for 's through `op`. The scope completes every
1129        // request it holds before 's ends, so the buffer and the op outlive the span in which
1130        // MPI may use them, and nothing touches the buffer meanwhile.
1131        let ret = unsafe {
1132            ffi::ferrompi_ireduce(
1133                sendbuf,
1134                p,
1135                n,
1136                dt,
1137                op,
1138                root,
1139                self.handle,
1140                &mut request_handle,
1141            )
1142        };
1143        Error::check_with_op(ret, "ireduce_inplace")?;
1144        Ok(s.request(request_handle, RequestKind::Collective))
1145    }
1146}
