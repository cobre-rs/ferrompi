3use crate::error::{Error, Result, FERROMPI_ERR_THREAD_LEVEL};
4use crate::ffi;
5use crate::rt;

7/// A persistent MPI request handle.
8///
9/// This type represents a persistent collective operation that has been
10/// initialized but not yet started. Unlike regular nonblocking operations,
11/// persistent operations can be started multiple times.
12///
13/// # Lifecycle
14///
15/// 1. Create with `comm.bcast_init()` or similar
16/// 2. Start with `start()` or `start_all()`
17/// 3. Wait for completion with `wait()`
18/// 4. Repeat steps 2-3 as needed
19/// 5. Free on drop
20///
21/// # Example
22///
23/// ```no_run
24/// use ferrompi::{Mpi, ReduceOp};
25///
26/// let mpi = Mpi::init().unwrap();
27/// let world = mpi.world();
28///
29/// // Buffer that will be used for all broadcasts
30/// let mut data = vec![0.0f64; 1000];
31///
32/// // Initialize persistent broadcast (MPI 4.0, or Open MPI 5)
33/// let mut persistent = world.bcast_init(&mut data, 0).unwrap();
34///
35/// // Run many iterations
36/// for iter in 0..1000 {
37///     // Update data on root
38///     if world.rank() == 0 {
39///         for (i, x) in data.iter_mut().enumerate() {
40///             *x = (iter * 1000 + i) as f64;
41///         }
42///     }
43///
44///     // Start the broadcast
45///     persistent.start().unwrap();
46///
47///     // Optionally do other work here...
48///
49///     // Wait for completion
50///     persistent.wait().unwrap();
51///
52///     // data now contains broadcast result on all ranks
53/// }
54///
55/// // Cleanup happens automatically on drop
56/// ```
57pub struct PersistentRequest {
58    handle: i64,
59    active: bool, // True if started but not yet waited
60}
61
62impl PersistentRequest {
63    /// Create a new persistent request from a raw handle.
64    pub(crate) fn new(handle: i64) -> Self {
65        PersistentRequest {
66            handle,
67            active: false,
68        }
69    }
70
71    /// Get the raw request handle (for advanced use).
72    pub fn raw_handle(&self) -> i64 {
73        self.handle
74    }
75
76    /// Check if this request is currently active (started but not waited).
77    pub fn is_active(&self) -> bool {
78        self.active
79    }
80
81    /// Start the persistent operation.
82    ///
83    /// This initiates the communication. You must call `wait()` before starting
84    /// again or accessing the buffers.
85    ///
86    /// # Errors
87    ///
88    /// Returns an error if the operation is already active or if the start fails.
89    #[inline]
90    pub fn start(&mut self) -> Result<()> {
91        if self.active {
92            return Err(Error::Internal("Request is already active".into()));
93        }
94        // SAFETY: self.handle is a valid persistent MPI request handle
95        // registered in the C-side request table by the *_init constructor
96        // that produced this PersistentRequest; self.active is false, so
97        // MPI_Start is not being called on an already-active request.
98        let ret = unsafe { ffi::ferrompi_start(self.handle) };
99        Error::check_with_op(ret, "start")?;
100        self.active = true;
101        Ok(())
102    }
103
104    /// Wait for the operation to complete.
105    ///
106    /// This blocks until the communication started by `start()` is finished.
107    /// After this returns, the buffers can be safely accessed and the operation
108    /// can be started again.
109    ///
110    /// # Errors
111    ///
112    /// Returns an error if the wait fails. A wait that MPI ran leaves the
113    /// request inactive either way: MPI completed it with that error. A call
114    /// rejected before reaching MPI (wrong thread, after finalize, or, in a
115    /// debug build at `Serialized`, overlapping another thread's call)
116    /// leaves it active.
117    #[inline]
118    pub fn wait(&mut self) -> Result<()> {
119        if !self.active {
120            return Ok(());
121        }
122        Error::check_with_op(rt::enter(), "wait")?;
123        // SAFETY: self.handle is a valid persistent MPI request handle
124        // registered in the C-side request table; self.active was true on
125        // entry (checked above), so start() was called and MPI holds an
126        // in-flight operation on this handle for ferrompi_wait to complete.
127        let ret = unsafe { ffi::ferrompi_wait(self.handle) };
128        // A debug build's Serialized overlap check rejects the call before MPI
129        // sees it; the request is then still active. Otherwise MPI completed it
130        // whatever it returned: it is inactive now, or freed if the library
131        // frees failed persistent requests.
132        if ret != FERROMPI_ERR_THREAD_LEVEL {
133            self.active = false;
134        }
135        Error::check_with_op(ret, "wait")
136    }
137
138    /// Test if the operation has completed without blocking.
139    ///
140    /// Returns `true` if complete, `false` if still in progress. A failed
141    /// `test` that MPI completed also leaves the request inactive.
142    #[inline]
143    pub fn test(&mut self) -> Result<bool> {
144        if !self.active {
145            return Ok(true);
146        }
147        let mut flag: i32 = 0;
148        // SAFETY: self.handle is a valid persistent MPI request handle
149        // registered in the C-side request table; self.active was true on
150        // entry (checked above). flag is a local out-parameter written by
151        // ferrompi_test before this function reads it below.
152        let ret = unsafe { ffi::ferrompi_test(self.handle, &mut flag) };
153        // flag is set when MPI completed the request, even with an error.
154        if flag != 0 {
155            self.active = false;
156        }
157        Error::check_with_op(ret, "test")?;
158        Ok(flag != 0)
159    }
160
161    /// Start multiple persistent operations.
162    ///
163    /// This is more efficient than starting each operation individually.
164    ///
165    /// # Errors
166    ///
167    /// Returns `Err(Internal)` without calling MPI if any request is already
168    /// active. If MPI reports an error, it may have started some of the
169    /// requests; every request is then marked active, so `wait`, `wait_all`
170    /// or `Drop` completes whichever did start.
171    pub fn start_all(requests: &mut [PersistentRequest]) -> Result<()> {
172        if requests.is_empty() {
173            return Ok(());
174        }
175
176        if requests.iter().any(|req| req.active) {
177            return Err(Error::Internal(
178                "One or more requests already active".into(),
179            ));
180        }
181
182        // SAFETY: with_handles provides a valid, contiguous [i64] of the
183        // persistent-request handles and a same-length [u8] started buffer,
184        // both sized to the count we pass.
185        let ret = crate::request::with_handles(
186            requests,
187            |r| r.handle,
188            |handles, started| unsafe {
189                ffi::ferrompi_startall(handles.len() as i64, handles.as_ptr(), started.as_mut_ptr())
190            },
191            |r| r.active = true,
192        );
193        Error::check_with_op(ret, "startall")
194    }
195
196    /// Wait for all persistent operations to complete.
197    ///
198    /// Whatever the result, every request MPI completed is marked inactive
199    /// in place; the others stay active. This is the same policy
200    /// [`Request::wait_all`](crate::Request::wait_all) applies.
201    /// Inactive requests are skipped, so a request whose earlier `wait` or
202    /// `test` failed can stay in the slice.
203    ///
204    /// On a failed request, the returned error carries that request's own
205    /// class and code, and its message ends with `(request N)`, `N` being
206    /// its index in `requests`.
207    pub fn wait_all(requests: &mut [PersistentRequest]) -> Result<()> {
208        if requests.is_empty() {
209            return Ok(());
210        }
211
212        let mut failed: i64 = -1;
213        // SAFETY: with_handles provides a valid, contiguous [i64] of the
214        // persistent-request handles and a same-length [u8] done buffer, both
215        // sized to the count we pass; failed is a valid stack-allocated i64
216        // output parameter.
217        let ret = crate::request::with_handles(
218            requests,
219            |r| if r.active { r.handle } else { -1 },
220            |handles, done| unsafe {
221                ffi::ferrompi_waitall(
222                    handles.len() as i64,
223                    handles.as_ptr(),
224                    done.as_mut_ptr(),
225                    &mut failed,
226                )
227            },
228            |r| r.active = false,
229        );
230        crate::request::check_batch(ret, "waitall", failed)
231    }
232}

234impl Drop for PersistentRequest {
235    /// Complete any in-flight operation, then free the persistent request handle.
236    ///
237    /// When `self.active` is `true` (i.e., `start()` was called but `wait()` has
238    /// not yet returned), this calls `MPI_Wait` before freeing the handle.
239    /// **`MPI_Wait` blocks** until the peer operation completes; if the peer is
240    /// unreachable, this deadlocks. This two-step sequence upholds the MPI
241    /// standard requirement that `MPI_Request_free` must not be called on an
242    /// active request.
243    ///
244    /// See ADR-0004 §"Drop behavior: wait before free" for the full rationale.
245    fn drop(&mut self) {
246        if !rt::drop_guard("PersistentRequest") {
247            return;
248        }
249        if self.active {
250            // SAFETY: self.handle is a valid MPI request handle registered in the
251            // C-side request table by the *_init constructor. self.active is true,
252            // so start() was called and MPI holds an in-flight operation on this
253            // handle. ferrompi_wait calls MPI_Wait which completes the operation
254            // and releases the handle's active state before request_free below.
255            // Calls the unguarded raw wrapper (not the lifecycle-guarded one):
256            // rt::drop_guard above already handles the FFI lifecycle check, so
257            // this call must still attempt the wait once reached.
258            unsafe { ffi::raw::ferrompi_wait(self.handle) };
259        }
260        // SAFETY: self.handle is a valid persistent MPI request handle. If it was
261        // active, ferrompi_wait above has already completed the operation, so
262        // MPI_Request_free is safe to call. If it was inactive, no operation is
263        // in flight and MPI_Request_free is unconditionally safe on the handle.
264        unsafe { ffi::ferrompi_request_free(self.handle) };
265    }
266}

274    #[test]
275    fn start_when_already_active_returns_error() {
276        let mut req = PersistentRequest {
277            handle: 0,
278            active: true,
279        };
280        let result = req.start();
281        assert!(result.is_err());
282        let err = result.unwrap_err();
283        assert!(
284            matches!(&err, Error::Internal(msg) if msg.contains("already active")),
285            "expected Error::Internal containing 'already active', got: {err}"
286        );
287        forget(req);
288    }

296            "expected Ok(true), got: {result:?}"

299    }
