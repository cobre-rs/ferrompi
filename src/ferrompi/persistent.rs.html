12/// Describes the heap allocation of `v` for MPI FFI as `(pointer, count, tag)`,
13/// like `datatype::buf` but with the pointer taken from the `Vec` itself
14/// (`Vec::as_ptr`) instead of from a slice reborrow. For a `Vec` that holds no
15/// bytes, `datatype::buf_ptr` may replace that pointer.
16///
17/// A pointer that MPI keeps across calls, as a persistent request does, must
18/// be derived this way: no slice reference then sits between it and the
19/// allocation, so the slices the request's accessors create later are its
20/// children. A pointer taken through a slice reference is invalidated by those
21/// accessors under the Stacked and Tree Borrows models.
22#[allow(
23    clippy::ptr_arg,
24    reason = "the pointer must come from the Vec, not from a slice reborrow"
25)]
26pub(crate) fn vec_buf<T: MpiDatatype>(v: &Vec<T>) -> (*const c_void, i64, i32) {
27    (
28        buf_ptr(v.as_ptr(), v.len(), false),
29        v.len() as i64,
30        tag_of::<T>() as i32,
31    )
32}
33
34/// See [`vec_buf`]; the pointer comes from `Vec::as_mut_ptr`.
35pub(crate) fn vec_buf_mut<T: MpiDatatype>(v: &mut Vec<T>) -> (*mut c_void, i64, i32) {
36    (
37        buf_ptr(v.as_mut_ptr(), v.len(), true).cast_mut(),
38        v.len() as i64,
39        tag_of::<T>() as i32,
40    )
41}

43/// A persistent MPI request that owns its buffers.
44///
45/// A persistent request is initialized once and then started and completed
46/// many times, so one setup serves a whole loop of iterations. The `*_init`
47/// constructors take the buffers by value and the request keeps them until it
48/// is freed: MPI holds a pointer into each allocation, so no safe call can
49/// resize or free one behind MPI's back. Reach the buffer with
50/// [`buffer`](Self::buffer) and [`buffer_mut`](Self::buffer_mut) while the
51/// request is inactive, or take it back with
52/// [`into_buffer`](Self::into_buffer).
53///
54/// A collective with separate send and receive buffers owns both. `buffer` and
55/// `buffer_mut` reach the receive buffer, [`send_buffer_mut`](Self::send_buffer_mut)
56/// reaches the send buffer, and [`into_buffers`](Self::into_buffers) returns
57/// both. A request that owns one buffer (a point-to-point request, a
58/// broadcast or an in-place collective) has an empty send buffer.
59///
60/// [`Communicator::barrier_init`](crate::Communicator::barrier_init) returns a
61/// `PersistentRequest<()>`, which owns no data. [`start_all`](Self::start_all)
62/// and [`wait_all`](Self::wait_all) take requests of one element type, so a
63/// barrier request is started and completed on its own, not in one slice with
64/// requests of another type.
65///
66/// Creating a persistent collective is itself a collective call. Every process
67/// of the communicator makes the same `*_init` call, ordered with its other
68/// collective calls on that communicator, and the call can block until the
69/// others make it (MPI-4.1 §6.13). The starts are not ordered: persistent
70/// collectives can be started in any order, and the order can differ between
71/// processes.
72///
73/// # Lifecycle
74///
75/// 1. Create with `comm.send_init(vec, ..)`, `comm.recv_init(vec, ..)` or
76///    another `*_init`, which takes the buffers by value.
77/// 2. Update the buffer through `buffer_mut()` (or `send_buffer_mut()`). The
78///    accessors work only while the request is inactive and return
79///    [`Error::InvalidState`] otherwise.
80/// 3. Start with `start()` or `start_all()`.
81/// 4. Wait for completion with `wait()`, then read the result through
82///    `buffer()`.
83/// 5. Repeat steps 2-4 as needed.
84/// 6. Free on drop, or take the buffer back with `into_buffer()` or
85///    `into_buffers()`.
86///
87/// # Example
88///
89/// ```no_run
90/// use ferrompi::Mpi;
91///
92/// # fn main() -> ferrompi::Result<()> {
93/// let mpi = Mpi::init()?;
94/// let world = mpi.world();
95/// let rank = world.rank();
96///
97/// // Initialize a self-send pair once; each request owns its buffer.
98/// let mut send = world.send_init(vec![0.0f64; 1000], rank, 0)?;
99/// let mut recv = world.recv_init(vec![0.0f64; 1000], rank, 0)?;
100///
101/// for iter in 0..1000 {
102///     for (i, x) in send.buffer_mut()?.iter_mut().enumerate() {
103///         *x = (iter * 1000 + i) as f64;
104///     }
105///
106///     recv.start()?;
107///     send.start()?;
108///     send.wait()?;
109///     recv.wait()?;
110///
111///     assert_eq!(recv.buffer()?[0], (iter * 1000) as f64);
112/// }
113///
114/// // Cleanup happens automatically on drop.
115/// # Ok(())
116/// # }
117/// ```
118///
119/// # Ownership
120///
121/// MPI keeps a pointer into each buffer from the `*_init` call until the
122/// request is freed, so safe code must have no way to resize or free a buffer
123/// while the request exists, or to touch it while the request is active. The
124/// constructors take each buffer by value, the accessors hand out slices that
125/// cannot be resized, and while the request is active they return
126/// [`Error::InvalidState`] instead of a slice. Each program below does not
127/// compile, and is followed by the same program made correct.
128///
129/// The `Vec` a constructor consumed cannot be resized afterwards, so its
130/// allocation cannot move while MPI holds a pointer into it:
131///
132/// ```compile_fail,E0382
133/// # fn f(world: &ferrompi::Communicator) -> ferrompi::Result<()> {
134/// let mut data = vec![0u64; 4];
135/// let _p = world.recv_init(data, 1, 9)?;
136/// data.reserve(4096);
137/// # Ok(())
138/// # }
139/// ```
140///
141/// ```no_run
142/// # fn f(world: &ferrompi::Communicator) -> ferrompi::Result<()> {
143/// let mut data = world.recv_init(vec![0u64; 4], 1, 9)?.into_buffer()?;
144/// data.reserve(4096);
145/// # Ok(())
146/// # }
147/// ```
148///
149/// Nor can the buffer be resized through an accessor, which returns a slice:
150///
151/// ```compile_fail,E0599
152/// # fn f(world: &ferrompi::Communicator) -> ferrompi::Result<()> {
153/// let mut p = world.recv_init(vec![0u64; 4], 1, 9)?;
154/// p.buffer_mut()?.reserve(4096);
155/// # Ok(())
156/// # }
157/// ```
158///
159/// ```no_run
160/// # fn f(world: &ferrompi::Communicator) -> ferrompi::Result<()> {
161/// let mut p = world.recv_init(vec![0u64; 4], 1, 9)?;
162/// p.buffer_mut()?.copy_from_slice(&[1, 2, 3, 4]);
163/// # Ok(())
164/// # }
165/// ```
166///
167/// A slice an accessor returned cannot be held across `start`, because the
168/// slice borrows the request and `start` borrows it mutably:
169///
170/// ```compile_fail,E0499
171/// # fn f(world: &ferrompi::Communicator) -> ferrompi::Result<()> {
172/// let mut p = world.send_init(vec![0u64; 4], 1, 9)?;
173/// let b = p.buffer_mut()?;
174/// p.start()?;
175/// b[0] = 1;
176/// # Ok(())
177/// # }
178/// ```
179///
180/// ```no_run
181/// # fn f(world: &ferrompi::Communicator) -> ferrompi::Result<()> {
182/// let mut p = world.send_init(vec![0u64; 4], 1, 9)?;
183/// p.buffer_mut()?[0] = 1;
184/// p.start()?;
185/// p.wait()?;
186/// # Ok(())
187/// # }
188/// ```
189///
190/// The `test_persistent_ownership` example checks the rest at run time: the
191/// accessors fail with [`Error::InvalidState`] while the request is active,
192/// `into_buffer` on an active request waits for it, a request dropped while
193/// active completes before its buffer is freed, a request forgotten while
194/// active leaks its buffer instead of freeing it, and a loop that rewrites the
195/// buffers between iterations allocates nothing per iteration.
196///
197/// # Dropping
198///
199/// Dropping an active request completes it (`MPI_Wait`, which blocks until the
200/// peer operation completes), then frees it (`MPI_Request_free`, which MPI does
201/// not allow on an active collective request), and only then frees the
202/// buffers, and after them the user op, which MPI may invoke at every start.
203///
204/// - At `Single`/`Funneled`, drop it on the thread that initialized MPI;
205///   another thread aborts the process.
206/// - At `Serialized` and `Multiple`, the drop is an MPI call, which the
207///   program serializes like any other.
208/// - Once [`Mpi`](crate::Mpi) is being dropped, or after it was dropped, the
209///   drop makes no MPI call. It leaks the request handle and, when the request
210///   is active, the send and receive buffers, the copy of a v-collective's
211///   count and displacement arrays, and the user op, because MPI may still use
212///   them. The buffers of an inactive request are freed normally.
213/// - A drop on another thread while `Mpi` is being dropped either completes
214///   before `MPI_Finalize`, because it holds a per-call count the finalizer
215///   waits for, or makes no MPI call and leaks as above. A drop that is blocked
216///   in `MPI_Wait` for a peer that never arrives blocks `Mpi`'s drop the same
217///   way a blocked `wait()` does. MPI requires every call in flight to complete
218///   before `MPI_Finalize` (MPI-4.1 §11.2.2 and §11.6.2 p.516).
219/// - Dropping `Mpi` while the request is active skips `MPI_Finalize`, whatever
220///   thread holds the request (see [`Mpi`](crate::Mpi)). The request can still
221///   be completed with `wait` or `test` afterwards; dropping it then makes no
222///   MPI call, as above.
223/// - The drop does not assume that a failed wait completed the request
224///   (MPI-4.1 §9.4). If completing an active request fails, it makes no
225///   further MPI call and leaks the request handle, the buffers and the user
226///   op, which the request keeps alive (see [`UserOp`](crate::UserOp)). The
227///   request also keeps counting as active, so a later [`Mpi`](crate::Mpi)
228///   drop skips `MPI_Finalize`.
229pub struct PersistentRequest<T> {
230    handle: i64,
231    token: Option<rt::HandleToken>,
232    send: Vec<T>,
233    buffer: Vec<T>,
234    counts: Vec<i32>,
235    pub(crate) op: Option<Arc<OpRegistration>>,
236}

238impl<T> PersistentRequest<T> {
239    /// Create a new persistent request from a raw handle and the one buffer it
240    /// owns.
241    pub(crate) fn new(handle: i64, buffer: Vec<T>) -> Self {
242        Self::new_pair(handle, Vec::new(), buffer, Vec::new())
243    }

245    /// Create a persistent request that owns a send buffer, a receive buffer
246    /// and the copy of the count and displacement arrays a v-collective was
247    /// given (empty for the other collectives).
248    ///
249    /// The `*_init` constructors hand MPI pointers into these `Vec`s, taken
250    /// from each `Vec` itself ([`vec_buf`], [`vec_buf_mut`], or `as_ptr` for
251    /// `counts`), and their `SAFETY` comments rely on this ownership invariant:
252    /// the request owns the `Vec`s until it is freed, and moving a `Vec` leaves
253    /// its allocation in place. The allocations are never resized, because the
254    /// accessors hand out slices only and nothing reaches `counts`. From
255    /// start() to completion MPI is the only user of the buffers, because the
256    /// accessors refuse while the request is active, and `Drop` leaks them when
257    /// MPI may still use them. If the init call fails, MPI recorded nothing (or
258    /// the shim freed the inactive request), and the constructor drops its
259    /// `Vec`s without building a request.
260    pub(crate) fn new_pair(handle: i64, send: Vec<T>, buffer: Vec<T>, counts: Vec<i32>) -> Self {
261        PersistentRequest {
262            handle,
263            token: None,
264            send,
265            buffer,
266            counts,
267            op: None,
268        }
269    }

271    /// Leaks what MPI may still use: the buffers, the count and displacement
272    /// copy, the user op the request keeps alive, and the token that makes
273    /// `Mpi::drop` skip `MPI_Finalize` while the request may be active.
274    fn leak(&mut self) {
275        std::mem::forget(self.token.take());
276        std::mem::forget(std::mem::take(&mut self.send));
277        std::mem::forget(std::mem::take(&mut self.buffer));
278        std::mem::forget(std::mem::take(&mut self.counts));
279        std::mem::forget(self.op.take());
280    }

282    /// Check if this request is currently active (started but not waited).
283    pub fn is_active(&self) -> bool {
284        self.token.is_some()
285    }

287    /// Start the persistent operation.
288    ///
289    /// This initiates the communication. You must call `wait()` before starting
290    /// again or reaching the buffer.
291    ///
292    /// # Errors
293    ///
294    /// Returns [`Error::InvalidState`] if the operation is already active,
295    /// [`Error::Finalized`] once [`Mpi`](crate::Mpi) is being dropped,
296    /// [`Error::ThreadLevelViolation`] on a thread other than the one that
297    /// initialized MPI at `Single`/`Funneled`, or the error of a failed start.
298    #[inline]
299    pub fn start(&mut self) -> Result<()> {
300        if self.is_active() {
301            return Err(Error::InvalidState {
302                reason: "request is already active",
303            });
304        }
305        let tokens =
306            rt::open_handle_tokens(1).map_err(|code| Error::from_code_with_op(code, "start"))?;
307        // SAFETY: self.handle is a valid persistent MPI request handle
308        // registered in the C-side request table by the *_init constructor
309        // that produced this PersistentRequest; the request is inactive (no
310        // token), so MPI_Start is not being called on an already-active
311        // request.
312        let ret = unsafe { ffi::ferrompi_start(self.handle) };
313        Error::check_with_op(ret, "start")?;
314        self.token = tokens.take();
315        Ok(())
316    }

318    /// Wait for the operation to complete.
319    ///
320    /// This blocks until the communication started by `start()` is finished.
321    /// After this returns, the buffer can be reached with
322    /// [`buffer`](Self::buffer) and [`buffer_mut`](Self::buffer_mut), and the
323    /// operation can be started again.
324    ///
325    /// For a request made by `recv_init` the returned [`Status`] holds the
326    /// matched source, tag and element count of that start
327    /// ([`Source::ProcNull`](crate::Source::ProcNull) for a receive from
328    /// `ProcNull`). For every send-side or collective request the `Status` is
329    /// the empty status, and so is an inactive request, as `MPI_Wait` returns
330    /// for one; only `error` has meaning, and these calls leave it `None`.
331    ///
332    /// # Errors
333    ///
334    /// Returns an error if the wait fails. A wait that MPI ran leaves the
335    /// request inactive either way: MPI completed it with that error. This
336    /// assumes that a request MPI reported an error for is complete, as MPICH
337    /// and Open MPI leave it; MPI-4.1 §9.4 (p.460) says an operation that
338    /// raised an error may not have satisfied its specification, so the
339    /// standard does not promise it. The same assumption holds for
340    /// [`test`](Self::test) and [`wait_all`](Self::wait_all); only `Drop`
341    /// does not make it. A call rejected before reaching MPI (wrong thread,
342    /// after finalize, or, in a debug build at `Serialized`, overlapping
343    /// another thread's call) leaves the request active.
344    #[inline]
345    pub fn wait(&mut self) -> Result<Status> {
346        if !self.is_active() {
347            return Ok(Status::EMPTY);
348        }
349        let mut status = ffi::FerrompiStatus::default();
350        self.wait_raw(&mut status)?;
351        Ok(Status::from_ffi(status))
352    }
353
354    /// The body of [`wait`](Self::wait) for an active request, out of line so
355    /// that the inline wrapper decodes `status` only when its caller uses the
356    /// result.
357    #[inline(never)]
358    fn wait_raw(&mut self, status: &mut ffi::FerrompiStatus) -> Result<()> {
359        Error::check_with_op(rt::check_completion(), "wait")?;
360        // SAFETY: self.handle is a valid persistent MPI request handle
361        // registered in the C-side request table; the request was active on
362        // entry (checked by wait), so start() was called and MPI holds an
363        // in-flight operation on this handle for ferrompi_wait to complete.
364        // status is a valid out-parameter.
365        let ret = unsafe { ffi::ferrompi_wait(self.handle, status) };
366        // A call the guard rejects before MPI sees it leaves the request
367        // active, token included: a debug build's Serialized overlap check, or
368        // a finalize that stored `Finalized` after the pre-check above.
369        // Otherwise MPI completed it whatever it returned: it is inactive now,
370        // or freed if the library frees failed persistent requests.
371        if ret != FERROMPI_ERR_THREAD_LEVEL && ret != FERROMPI_ERR_FINALIZED {
372            self.token = None;
373        }
374        Error::check_with_op(ret, "wait")
375    }

377    /// Test if the operation has completed without blocking.
378    ///
379    /// Returns `Some(`[`Status`]`)` if complete, `None` if still in progress.
380    /// The `Status` is the one [`wait`](Self::wait) returns: the matched source,
381    /// tag and element count for a `recv_init` request, the empty status for
382    /// every other request, and the empty status for an inactive request. A
383    /// failed `test` that MPI completed also leaves the request inactive, on
384    /// the assumption described under [`wait`](Self::wait).
385    #[inline]
386    pub fn test(&mut self) -> Result<Option<Status>> {
387        if !self.is_active() {
388            return Ok(Some(Status::EMPTY));
389        }
390        let mut status = ffi::FerrompiStatus::default();
391        Ok(self
392            .test_raw(&mut status)?
393            .then(|| Status::from_ffi(status)))
394    }

396    /// The body of [`test`](Self::test) for an active request, out of line like
397    /// [`wait_raw`](Self::wait_raw). Returns whether MPI completed it.
398    #[inline(never)]
399    fn test_raw(&mut self, status: &mut ffi::FerrompiStatus) -> Result<bool> {
400        let mut flag: i32 = 0;
401        // SAFETY: self.handle is a valid persistent MPI request handle
402        // registered in the C-side request table; the request was active on
403        // entry (checked by test). flag is a local out-parameter written by
404        // ferrompi_test before this function reads it below; status is a valid
405        // out-parameter.
406        let ret = unsafe { ffi::ferrompi_test(self.handle, &mut flag, status) };
407        // flag is set when MPI completed the request, even with an error.
408        if flag != 0 {
409            self.token = None;
410        }
411        Error::check_with_op(ret, "test")?;
412        Ok(flag != 0)
413    }

415    /// The request's buffer, while the request is inactive.
416    ///
417    /// This is the send data of a send request, and the received data of a
418    /// receive request after its `wait()`. For a collective with separate send
419    /// and receive buffers it is the receive buffer; reach the send buffer with
420    /// [`send_buffer_mut`](Self::send_buffer_mut). The slice cannot be resized,
421    /// because MPI holds a pointer into the allocation.
422    ///
423    /// # Errors
424    ///
425    /// Returns [`Error::InvalidState`] while the request is active: MPI owns
426    /// the buffer from `start()` until `wait()`, `wait_all()`, or a `test()`
427    /// that reports completion, returns.
428    #[inline]
429    pub fn buffer(&self) -> Result<&[T]> {
430        if self.is_active() {
431            return Err(Error::InvalidState {
432                reason: "request is active",
433            });
434        }
435        Ok(&self.buffer)
436    }

438    /// The request's buffer for writing, while the request is inactive: the
439    /// buffer [`buffer`](Self::buffer) reads.
440    ///
441    /// Write the next iteration's input here between `wait()` and the next
442    /// `start()`.
443    ///
444    /// # Errors
445    ///
446    /// Returns [`Error::InvalidState`] while the request is active.
447    #[inline]
448    pub fn buffer_mut(&mut self) -> Result<&mut [T]> {
449        if self.is_active() {
450            return Err(Error::InvalidState {
451                reason: "request is active",
452            });
453        }
454        Ok(&mut self.buffer)
455    }

457    /// The send buffer of a collective with separate send and receive buffers,
458    /// for writing, while the request is inactive.
459    ///
460    /// Write the next iteration's input here between `wait()` and the next
461    /// `start()`. The slice is empty for a request that owns one buffer: a
462    /// point-to-point request, a broadcast or an in-place collective.
463    ///
464    /// # Errors
465    ///
466    /// Returns [`Error::InvalidState`] while the request is active.
467    #[inline]
468    pub fn send_buffer_mut(&mut self) -> Result<&mut [T]> {
469        if self.is_active() {
470            return Err(Error::InvalidState {
471                reason: "request is active",
472            });
473        }
474        Ok(&mut self.send)
475    }

477    /// Complete the request if it is active, free it, and return its buffer.
478    ///
479    /// The request is freed before this returns. Freeing a request is an MPI
480    /// call, so `Drop`'s thread rule applies: at `Single`/`Funneled`, on a
481    /// thread other than the one that initialized MPI, the process aborts.
482    ///
483    /// For a collective with separate send and receive buffers this returns the
484    /// receive buffer and drops the send buffer after the free; use
485    /// [`into_buffers`](Self::into_buffers) to keep both.
486    ///
487    /// It discards the final [`Status`]: a `recv_init` caller that needs the
488    /// received count or source calls [`wait`](Self::wait) first, which returns
489    /// it, and then `into_buffer`. The same holds for
490    /// [`into_buffers`](Self::into_buffers).
491    ///
492    /// # Errors
493    ///
494    /// Returns the error of [`wait`](Self::wait) if completing an active
495    /// request fails. The request is then dropped as `Drop` does, and the
496    /// buffer is not returned.
497    pub fn into_buffer(mut self) -> Result<Vec<T>> {
498        self.wait()?;
499        Ok(std::mem::take(&mut self.buffer))
500    }
501
502    /// Complete the request if it is active, free it, and return its send and
503    /// receive buffers as `(send, buffer)`.
504    ///
505    /// The send buffer is empty for a request that owns one buffer. Otherwise
506    /// this behaves as [`into_buffer`](Self::into_buffer).
507    ///
508    /// # Errors
509    ///
510    /// Returns the error of [`wait`](Self::wait) if completing an active
511    /// request fails. The request is then dropped as `Drop` does, and the
512    /// buffers are not returned.
513    pub fn into_buffers(mut self) -> Result<(Vec<T>, Vec<T>)> {
514        self.wait()?;
515        Ok((
516            std::mem::take(&mut self.send),
517            std::mem::take(&mut self.buffer),
518        ))
519    }

521    /// Start multiple persistent operations.
522    ///
523    /// This is more efficient than starting each operation individually.
524    ///
525    /// # Errors
526    ///
527    /// Returns `Err(`[`Error::InvalidState`]`)` without calling MPI if any
528    /// request is already active, and [`Error::Finalized`] or
529    /// [`Error::ThreadLevelViolation`] as [`start`](Self::start) does. If MPI
530    /// reports an error, it may have started some of the requests; every
531    /// request is then marked active, so `wait`, `wait_all` or `Drop`
532    /// completes whichever did start.
533    pub fn start_all(requests: &mut [PersistentRequest<T>]) -> Result<()> {
534        if requests.is_empty() {
535            return Ok(());
536        }
537
538        if requests.iter().any(PersistentRequest::is_active) {
539            return Err(Error::InvalidState {
540                reason: "a request is already active",
541            });
542        }
543
544        let tokens = rt::open_handle_tokens(requests.len())
545            .map_err(|code| Error::from_code_with_op(code, "startall"))?;
546
547        // SAFETY: with_handles provides a valid, contiguous [i64] of the
548        // persistent-request handles and a same-length [u8] started buffer,
549        // both sized to the count we pass.
550        let ret = crate::request::with_handles(
551            requests,
552            |r| r.handle,
553            |handles, started| unsafe {
554                ffi::ferrompi_startall(handles.len() as i64, handles.as_ptr(), started.as_mut_ptr())
555            },
556            |r| r.token = tokens.take(),
557        );
558        Error::check_with_op(ret, "startall")
559    }

561    /// Wait for all persistent operations to complete.
562    ///
563    /// Whatever the result, every request MPI completed is marked inactive
564    /// in place; the others stay active. This is the same policy
565    /// [`Request::wait_all`](crate::Request::wait_all) applies.
566    /// Inactive requests are skipped, so a request whose earlier `wait` or
567    /// `test` failed can stay in the slice.
568    ///
569    /// On a failed request, the returned error carries that request's own
570    /// class and code, and its message ends with `(request N)`, `N` being
571    /// its index in `requests`.
572    ///
573    /// This returns no status, like
574    /// [`Request::wait_all`](crate::Request::wait_all). A request it completed
575    /// is inactive, so a later [`wait`](Self::wait) on it returns the empty
576    /// status; to read a receive's [`Status`], complete that request with
577    /// `wait` or [`test`](Self::test) instead.
578    pub fn wait_all(requests: &mut [PersistentRequest<T>]) -> Result<()> {
579        if requests.is_empty() {
580            return Ok(());
581        }
582
583        let mut failed: i64 = -1;
584        // SAFETY: with_handles provides a valid, contiguous [i64] of the
585        // persistent-request handles and a same-length [u8] done buffer, both
586        // sized to the count we pass; failed is a valid stack-allocated i64
587        // output parameter.
588        let ret = crate::request::with_handles(
589            requests,
590            |r| if r.is_active() { r.handle } else { -1 },
591            |handles, done| unsafe {
592                ffi::ferrompi_waitall(
593                    handles.len() as i64,
594                    handles.as_ptr(),
595                    done.as_mut_ptr(),
596                    &mut failed,
597                )
598            },
599            |r| r.token = None,
600        );
601        crate::request::check_batch(ret, "waitall", failed)
602    }
603}
604
605impl<T> Drop for PersistentRequest<T> {
606    /// Completes an active request, frees it, then frees its buffers and
607    /// releases its user op, as described under [Dropping](Self#dropping).
608    fn drop(&mut self) {
609        let Some(_call) = rt::drop_guard("PersistentRequest") else {
610            if self.is_active() {
611                self.leak();
612            }
613            return;
614        };
615        if self.is_active() {
616            // SAFETY: self.handle is a valid MPI request handle registered in the
617            // C-side request table by the *_init constructor. The request is
618            // active, so start() was called and MPI holds an in-flight operation
619            // on this handle. ferrompi_wait calls MPI_Wait which completes the
620            // operation and releases the handle's active state before
621            // request_free below.
622            // Calls the unguarded raw wrapper (not the lifecycle-guarded one):
623            // rt::drop_guard above already handles the FFI lifecycle check, so
624            // this call must still attempt the wait once reached.
625            let ret = unsafe { ffi::raw::ferrompi_wait(self.handle, std::ptr::null_mut()) };
626            if ret != 0 {
627                self.leak();
628                return;
629            }
630        }
631        // SAFETY: self.handle is a valid persistent MPI request handle. If it was
632        // active, ferrompi_wait above has already completed the operation, so
633        // MPI_Request_free is safe to call. If it was inactive, no operation is
634        // in flight and MPI_Request_free is unconditionally safe on the handle.
635        unsafe { ffi::ferrompi_request_free(self.handle) };
636        // Released before `_call`, so a finalizer that waited for `_call` also
637        // sees the request gone.
638        self.token = None;
639    }
640}

942    /// A drop does not assume that a failed raw wait completed the request, so
943    /// it must neither free the request nor release what MPI may still use,
944    /// the token that keeps `Mpi::drop` from finalizing included.
945    /// Handle -1 names no request: the C layer refuses the wait without
946    /// calling MPI, which stands in for an `MPI_Wait` that returned an error.
947    #[test]
948    #[cfg_attr(miri, ignore = "calls the C layer")]
949    fn drop_after_a_failed_wait_leaks_the_buffers_and_the_op() {
950        let registration = Arc::new(OpRegistration { slot: 9 });
951        let mut send = vec![0u64; 4];
952        let mut recv = vec![0u64; 4];
953        let mut counts = vec![0i32; 4];
954        let (sp, rp, cp) = (send.as_mut_ptr(), recv.as_mut_ptr(), counts.as_mut_ptr());
955        let mut req = PersistentRequest::new_pair(-1, send, recv, counts);
956        req.op = Some(Arc::clone(&registration));
957        req.token = Some(active_token());
958        drop(req);
959        assert_eq!(Arc::strong_count(&registration), 2);
960        // The token stays counted, so `Mpi::drop` still skips `MPI_Finalize`.
961        // Other tests hold tokens too, so the sum is only bounded below.
962        assert!(rt::live_handle_tokens() >= 1);
963        // SAFETY: the failed wait made the drop forget the three `Vec`s (each
964        // of length and capacity four, as built above) and the op clone, so
965        // each allocation is live and has no other owner, and the leaked strong
966        // count is the one released here.
967        unsafe {
968            drop(Vec::from_raw_parts(sp, 4, 4));
969            drop(Vec::from_raw_parts(rp, 4, 4));
970            drop(Vec::from_raw_parts(cp, 4, 4));
971            Arc::decrement_strong_count(Arc::as_ptr(&registration));
972        }
973        assert_eq!(Arc::strong_count(&registration), 1);
974        rt::release_forgotten_handle_token();
975    }
