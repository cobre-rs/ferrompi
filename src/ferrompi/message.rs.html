9/// A message that a matched probe took out of the receive queue and that has
10/// not been received yet.
11///
12/// [`Communicator::mprobe`](crate::Communicator::mprobe) and
13/// [`Communicator::improbe`](crate::Communicator::improbe) return it with the
14/// message's [`Status`]. Only [`mrecv`](Self::mrecv) on this handle receives
15/// the message: no other probe or receive, on any thread, can take it. That
16/// makes probe-then-receive correct when several threads receive on one
17/// communicator, where [`probe`](crate::Communicator::probe) followed by
18/// [`recv`](crate::Communicator::recv) can receive a message other than the
19/// one that was probed.
20///
21/// # Example
22///
23/// ```no_run
24/// # use ferrompi::{Mpi, Source, Tag};
25/// # fn main() -> ferrompi::Result<()> {
26/// # let mpi = Mpi::init()?;
27/// # let world = mpi.world();
28/// let (message, status) = world.mprobe::<f64>(Source::Any, Tag::Any)?;
29/// let mut buf = vec![0.0f64; status.count.expect("whole number of f64")];
30/// message.mrecv(&mut buf)?;
31/// # Ok(())
32/// # }
33/// ```
34///
35/// # Dropping
36///
37/// MPI requires every matched message to be received (MPI-4.1 §3.8.2, p.98),
38/// so dropping a `Message` receives it into a scratch buffer and throws the
39/// data away. The scratch receive uses `MPI_BYTE` whatever type the message
40/// was sent with. That strictly breaks MPI's type-matching rule
41/// (MPI-4.1 §3.3.1, p.46), under which a send and its receive name the same
42/// type, but MPICH and Open MPI accept the receive.
43///
44/// - The drop is an MPI call. On a thread that may not call MPI at the provided
45///   thread level, it makes no MPI call and does not abort: the message stays
46///   matched and unreceived for the rest of the process.
47/// - A drop whose receive fails, whose scratch buffer cannot be allocated, or
48///   whose message has a length MPI could not report (for example more than
49///   `i32::MAX` bytes on a library without the large-count calls) is treated
50///   the same way: the message counts as unreceived for the rest of the
51///   process.
52/// - While a `Message` is unreceived, whatever thread holds it, dropping
53///   [`Mpi`](crate::Mpi) skips `MPI_Finalize` (see [`Mpi`](crate::Mpi)):
54///   finalizing with a matched message pending is erroneous (MPI-4.1 §11.2.2,
55///   p.489). The message can still be received or dropped afterwards.
56///
57/// # Compile-time guarantees
58///
59/// A message is received at most once. `mrecv` takes the handle by value:
60///
61/// ```compile_fail,E0382
62/// # fn f(world: &ferrompi::Communicator) -> ferrompi::Result<()> {
63/// let (message, _) = world.mprobe::<u64>(1, 9)?;
64/// let mut first = [0u64; 4];
65/// let mut second = [0u64; 4];
66/// message.mrecv(&mut first)?;
67/// message.mrecv(&mut second)?;
68/// # Ok(())
69/// # }
70/// ```
71///
72/// ```no_run
73/// # fn f(world: &ferrompi::Communicator) -> ferrompi::Result<()> {
74/// let (message, _) = world.mprobe::<u64>(1, 9)?;
75/// let mut first = [0u64; 4];
76/// message.mrecv(&mut first)?;
77/// # Ok(())
78/// # }
79/// ```
80///
81/// and the handle cannot be copied:
82///
83/// ```compile_fail,E0599
84/// # fn f(world: &ferrompi::Communicator) -> ferrompi::Result<()> {
85/// let (message, _) = world.mprobe::<u64>(1, 9)?;
86/// let copy = message.clone();
87/// # Ok(())
88/// # }
89/// ```
90///
91/// ```no_run
92/// # fn f(world: &ferrompi::Communicator) -> ferrompi::Result<()> {
93/// let (message, _) = world.mprobe::<u64>(1, 9)?;
94/// let moved = message;
95/// # Ok(())
96/// # }
97/// ```
98#[must_use = "a matched message that is not received is discarded when dropped"]
99pub struct Message {
100    /// The `MPI_Message`, as the C layer carries it.
101    handle: i64,
102    /// The message's length in bytes, for the discard; negative if MPI could
103    /// not report it.
104    bytes: i64,
105    /// Held from the match until MPI received the message. `Mpi::drop` skips
106    /// `MPI_Finalize` while it is counted.
107    token: Option<rt::HandleToken>,
108}
109
110impl Message {
111    pub(crate) fn new(handle: i64, bytes: i64, token: Option<rt::HandleToken>) -> Self {
112        Message {
113            handle,
114            bytes,
115            token,
116        }
117    }
118
119    /// Receive the matched message into `buf`.
120    ///
121    /// Returns the receive's [`Status`], as [`recv`](crate::Communicator::recv)
122    /// does: the matched source and tag, and the number of `T` elements
123    /// received. A message matched from [`Source::ProcNull`](crate::Source::ProcNull)
124    /// returns at once with `Source::ProcNull`, `Tag::Any` and `Some(0)`, and
125    /// leaves `buf` untouched.
126    ///
127    /// `buf` may be longer than the message. With an MPI library that has no
128    /// large-count calls, MPI is told of at most `i32::MAX` elements of room,
129    /// so a longer `buf` is accepted here, where every other call returns
130    /// class [`MpiErrorClass::Count`](crate::MpiErrorClass::Count) for a count
131    /// above `i32::MAX`.
132    ///
133    /// # Errors
134    ///
135    /// Returns [`Error::Mpi`] with class
136    /// [`MpiErrorClass::Truncate`](crate::MpiErrorClass::Truncate) if the
137    /// message does not fit in `buf`. A receive that MPI ran consumes the
138    /// message whatever it returned, so the message is then gone.
139    ///
140    /// A call rejected before reaching MPI ([`Error::ThreadLevelViolation`] on a
141    /// thread that may not call MPI, or, in a debug build at `Serialized`,
142    /// while another thread's call is in progress) receives nothing; the
143    /// message is then dropped as described under [Dropping](Self#dropping).
144    pub fn mrecv<T: MpiDatatype>(mut self, buf: &mut [T]) -> Result<Status> {
145        let (p, n, dt) = buf_mut(buf);
146        let mut status = ffi::FerrompiStatus::default();
147        // SAFETY: this blocking call returns only after MPI is done with the buffer.
148        // self.handle is a message that a matched probe returned and that no receive
149        // was issued for: the token is still held, and it is given up below as soon as
150        // MPI ran one.
151        let ret = unsafe { ffi::ferrompi_mrecv(p, n, dt, self.handle, &mut status) };
152        // A call the guard rejects before MPI sees it leaves the message matched,
153        // token included, and the drop of `self` deals with it. Otherwise MPI
154        // consumed the message whatever it returned, and the handle must never
155        // reach MPI again: MPICH leaves it set after a failed receive.
156        if ret != FERROMPI_ERR_THREAD_LEVEL && ret != FERROMPI_ERR_FINALIZED {
157            self.token = None;
158        }
159        Error::check_with_op(ret, "mrecv")?;
160        Ok(Status::from_ffi(status))
161    }
162}

164impl Drop for Message {
165    /// Receives an unreceived message into a scratch buffer, as `MPI_BYTE`.
166    ///
167    /// The token is released only once that receive succeeded. On every other
168    /// path it is forgotten, so the message keeps counting and `Mpi`'s drop
169    /// skips `MPI_Finalize`: a refused call (wrong thread, or `Mpi` already
170    /// finalized), a failed receive, an unknown length, or a scratch buffer
171    /// that could not be allocated.
172    fn drop(&mut self) {
173        let Some(token) = self.token.take() else {
174            return;
175        };
176        let mut scratch: Vec<u8> = Vec::new();
177        let discarded = match usize::try_from(self.bytes) {
178            Ok(bytes) if scratch.try_reserve_exact(bytes).is_ok() => {
179                let p = buf_ptr(scratch.as_mut_ptr(), bytes, true).cast_mut();
180                // SAFETY: `p` is valid for writes of `bytes` bytes: it is the
181                // reserved capacity of `scratch`, or a fixed address when `bytes`
182                // is zero, where MPI writes nothing. self.handle is a matched,
183                // unreceived message, because the token was still held. A null
184                // status asks for none.
185                let ret = unsafe {
186                    ffi::ferrompi_mrecv(
187                        p,
188                        self.bytes,
189                        DatatypeTag::Byte as i32,
190                        self.handle,
191                        std::ptr::null_mut(),
192                    )
193                };
194                ret == 0
195            }
196            _ => false,
197        };
198        if !discarded {
199            std::mem::forget(token);
200        }
201    }
202}
