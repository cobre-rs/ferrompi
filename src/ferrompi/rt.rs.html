1//! Process-wide FFI lifecycle guard:
2//! `Uninit -> Initializing -> Active(level) -> Finalized`.
3//!
4//! [`enter`] is called by every guarded extern wrapper in [`crate::ffi`] and
5//! rejects the call once state is `Finalized`, or once state is
6//! `Active(Single)`/`Active(Funneled)` and the caller is not the thread that
7//! called `activate`, without touching MPI. [`drop_guard`] is the equivalent
8//! check for a `Drop` impl, which cannot return `Err`: it skips the MPI call
9//! silently after finalize, and aborts the process on a wrong-thread drop
10//! below `Serialized`. `Initializing` is a transient state held only between
11//! [`begin_init`] and [`activate`]/[`abandon_init`]; both `enter` and
12//! `drop_guard` pass it through like `Uninit` since no `Mpi` handle exists
13//! yet to call either.
14
15use std::cell::Cell;
16use std::io::Write;
17use std::os::raw::c_int;
18#[cfg(debug_assertions)]
19use std::sync::atomic::AtomicBool;
20use std::sync::atomic::{AtomicU8, Ordering};
21
22use crate::error::{Error, Result, FERROMPI_ERR_FINALIZED, FERROMPI_ERR_THREAD_LEVEL};
23use crate::ThreadLevel;
24
25const UNINIT: u8 = 0;
26const ACTIVE_SINGLE: u8 = 1;
27const ACTIVE_FUNNELED: u8 = 2;
28const FINALIZED: u8 = 5;
29const INITIALIZING: u8 = 6;
30
31static STATE: AtomicU8 = AtomicU8::new(UNINIT);
32
33thread_local! {
34    /// Set by `activate` on the thread that calls it. `Mpi` is `!Send`, so
35    /// `STATE` and this flag are always written by the same single thread.
36    static ON_INIT_THREAD: Cell<bool> = const { Cell::new(false) };
37}
38
39/// Move `Uninit` to `Initializing`, serialising concurrent
40/// `Mpi::init`/`init_thread` calls. Returns `Err(Error::Finalized)` once
41/// state is `Finalized`, and `Err(Error::AlreadyInitialized)` for any other
42/// state (an `Mpi` is alive, or another thread is already initialising).
43/// Called once, at the top of `Mpi::init_thread`; every path out of that
44/// function afterward reaches either [`activate`] or [`abandon_init`].
45pub(crate) fn begin_init() -> Result<()> {
46    // Relaxed: a successful compare-exchange only ever transitions out of the
47    // initial zero state, which carries no prior writer to synchronize with;
48    // a losing thread returns without touching MPI, so it needs no ordering
49    // beyond observing the current value through this atomic's modification
50    // order, which the compare-exchange itself guarantees regardless of
51    // ordering annotation.
52    match STATE.compare_exchange(UNINIT, INITIALIZING, Ordering::Relaxed, Ordering::Relaxed) {
53        Ok(_) => Ok(()),
54        Err(FINALIZED) => Err(Error::Finalized),
55        Err(_) => Err(Error::AlreadyInitialized),
56    }
57}
58
59/// Move `Initializing` back to `Uninit`. Called when `Mpi::init_thread` does
60/// not complete after [`begin_init`] succeeded: MPI turned out already
61/// finalized, or `MPI_Init_thread` itself failed. Only the thread that won
62/// `begin_init`'s compare-exchange calls this, so a plain store is safe.
63pub(crate) fn abandon_init() {
64    // Relaxed: only the calling thread can observe `Initializing` (it is the
65    // sole holder, per this function's contract), so there is no concurrent
66    // writer to synchronize with.
67    STATE.store(UNINIT, Ordering::Relaxed);
68}
69
70/// Move `Initializing` to `Active(level)` and record the calling thread as
71/// the init thread. Called once, from `Mpi::init_thread` after
72/// `MPI_Init_thread` succeeds.
73pub(crate) fn activate(level: ThreadLevel) {
74    // Relaxed: `Mpi` is `!Send`/`!Sync`, so every later `enter()` load that
75    // must observe this store — whether on this thread or one that receives
76    // a `Communicator`/`Request` afterward — is already ordered by the
77    // synchronizing operation (thread spawn, channel, mutex, `Arc`) that
78    // thread used to obtain that handle; this store needs no ordering of
79    // its own to be visible through that chain.
80    STATE.store(1 + level as u8, Ordering::Relaxed);
81    ON_INIT_THREAD.with(|c| c.set(true));
82}
83
84/// Move an `Active` state to `Finalized`, returning `true`. Returns `false`
85/// and changes nothing in any other state, so a stub `Mpi` dropped without
86/// init (`Uninit`) — or a second drop after finalize — is a no-op.
87pub(crate) fn finalize() -> bool {
88    // Relaxed: `Mpi` is `!Send`, so this load only ever races with another
89    // thread's `enter()` load, never with a second write on this thread.
90    let prev = STATE.load(Ordering::Relaxed);
91    if (1..=4).contains(&prev) {
92        // Relaxed: no ordering on this store closes the window where a
93        // concurrent `enter()` on another thread reads `Active` just before
94        // this store and then calls MPI just after `ferrompi_finalize` runs
95        // below. No ordering on `STATE` can close that gap; until this crate
96        // tracks in-flight calls, it is instead a caller contract: `Mpi`
97        // must not be dropped while another thread is inside an MPI call
98        // through this crate.
99        STATE.store(FINALIZED, Ordering::Relaxed);
100        true
101    } else {
102        false
103    }
104}

106/// Returns `true` once state is `Finalized`. `Mpi::is_finalized()` uses this
107/// so it still reports `true` after a skipped `MPI_Finalize` (a live window
108/// kept `Mpi::drop` from calling it): `finalize()`
109/// already moved `STATE` to `Finalized` before that skip decision runs.
110pub(crate) fn is_finalized() -> bool {
111    // Relaxed: see `enter`'s comment — visibility of the `Finalized` state
112    // set on another thread is carried by that thread's own handle hand-off,
113    // not by this load's ordering.
114    STATE.load(Ordering::Relaxed) == FINALIZED
115}

117/// Called at the top of every guarded extern wrapper. Returns
118/// `FERROMPI_ERR_FINALIZED` once state is `Finalized`;
119/// `FERROMPI_ERR_THREAD_LEVEL` once state is `Active(Single)` or
120/// `Active(Funneled)` and the caller is not the init thread; `0` otherwise.
121#[inline(always)]
122pub(crate) fn enter() -> c_int {
123    // Relaxed: see `activate`'s comment — visibility of an `Active`/`Finalized`
124    // state set on another thread is carried by that thread's own handle
125    // hand-off, not by this load's ordering.
126    let state = STATE.load(Ordering::Relaxed);
127    if state == FINALIZED {
128        return FERROMPI_ERR_FINALIZED;
129    }
130    if (state == ACTIVE_SINGLE || state == ACTIVE_FUNNELED) && !ON_INIT_THREAD.with(Cell::get) {
131        return FERROMPI_ERR_THREAD_LEVEL;
132    }
133    0
134}

136/// Set for the duration of one guarded FFI call while `STATE` is
137/// `Active(Serialized)`, to detect two such calls overlapping. Debug-only:
138/// `Serialized` requires the caller to serialize its own MPI calls, so this
139/// is a diagnostic, not a correctness mechanism release builds must pay for.
140#[cfg(debug_assertions)]
141static SERIALIZED_IN_CALL: AtomicBool = AtomicBool::new(false);
142
143/// Attempts to take [`SERIALIZED_IN_CALL`]. Returns `true` on success.
144#[cfg(debug_assertions)]
145fn take_in_call_flag() -> bool {
146    // Acquire/Relaxed: a successful exchange must synchronize-with the
147    // matching `release_in_call_flag`'s Release store, so the guarded FFI
148    // call this thread is about to make cannot be reordered ahead of the
149    // previous holder's; a failed exchange holds nothing and needs no
150    // ordering.
151    SERIALIZED_IN_CALL
152        .compare_exchange(false, true, Ordering::Acquire, Ordering::Relaxed)
153        .is_ok()
154}
155
156/// Releases [`SERIALIZED_IN_CALL`].
157#[cfg(debug_assertions)]
158fn release_in_call_flag() {
159    // Release: pairs with `take_in_call_flag`'s Acquire, so this thread's
160    // guarded FFI call happens-before the next thread's that takes the flag.
161    SERIALIZED_IN_CALL.store(false, Ordering::Release);
162}

164/// Called at the top of a guarded extern wrapper, after [`enter`] passes. At
165/// `Active(Serialized)`, takes [`SERIALIZED_IN_CALL`] and returns `Ok(true)`,
166/// or `Err(FERROMPI_ERR_THREAD_LEVEL)` if another call already holds it. In
167/// every other state, returns `Ok(false)` without touching the flag.
168#[cfg(debug_assertions)]
169pub(crate) fn begin_call() -> std::result::Result<bool, c_int> {
170    // Relaxed: see `enter`'s comment — visibility of the `Active` state set
171    // on another thread is carried by that thread's own handle hand-off,
172    // not by this load's ordering.
173    if STATE.load(Ordering::Relaxed) == 1 + ThreadLevel::Serialized as u8 {
174        if take_in_call_flag() {
175            Ok(true)
176        } else {
177            Err(FERROMPI_ERR_THREAD_LEVEL)
178        }
179    } else {
180        Ok(false)
181    }
182}
183
184/// Releases the flag taken by [`begin_call`], when `held` is `true`.
185#[cfg(debug_assertions)]
186pub(crate) fn end_call(held: bool) {
187    if held {
188        release_in_call_flag();
189    }
190}

192/// Called at the top of a `Drop` impl's MPI-calling path, in place of
193/// [`enter`] (a `Drop` impl cannot propagate an `Err`). Returns `false` once
194/// state is `Finalized`, so the caller skips its MPI call silently. Once
195/// state is `Active(Single)`/`Active(Funneled)` and the caller is not the
196/// init thread, this never returns: see [`drop_abort`]. Otherwise returns
197/// `true`.
198pub(crate) fn drop_guard(type_name: &'static str) -> bool {
199    // Relaxed: see `enter`'s comment — visibility of an `Active`/`Finalized`
200    // state set on another thread is carried by that thread's own handle
201    // hand-off, not by this load's ordering.
202    let state = STATE.load(Ordering::Relaxed);
203    if state == FINALIZED {
204        return false;
205    }
206    if (state == ACTIVE_SINGLE || state == ACTIVE_FUNNELED) && !ON_INIT_THREAD.with(Cell::get) {
207        drop_abort(type_name);
208    }
209    true
210}
211
212/// Prints `ferrompi: <type_name> dropped on thread <label>` to stderr and
213/// aborts the process. Split out of [`drop_guard`] and marked `#[cold]` so
214/// the wrong-thread path does not bloat the common-case branch.
215#[cold]
216fn drop_abort(type_name: &'static str) -> ! {
217    let current = std::thread::current();
218    let label = match current.name() {
219        Some(name) => name.to_string(),
220        None => format!("{:?}", current.id()),
221    };
222    // eprintln! panics on a closed stderr, and a panic inside `Drop` during
223    // unwinding aborts without printing this message, so the write result
224    // is ignored instead.
225    let _ = writeln!(
226        std::io::stderr(),
227        "ferrompi: {type_name} dropped on thread {label}"
228    );
229    std::process::abort();
230}

232#[cfg(all(test, debug_assertions))]
233mod tests {
234    use super::{release_in_call_flag, take_in_call_flag};
235
236    #[test]
237    fn in_call_flag_rejects_a_second_thread() {
238        assert!(take_in_call_flag(), "first take must succeed");
239
240        let overlapping_took = std::thread::scope(|s| {
241            s.spawn(take_in_call_flag)
242                .join()
243                .expect("overlap thread panicked")
244        });
245        assert!(!overlapping_took, "overlapping take must be rejected");
246
247        release_in_call_flag();
248
249        let next_took = std::thread::scope(|s| {
250            s.spawn(|| {
251                let took = take_in_call_flag();
252                if took {
253                    release_in_call_flag();
254                }
255                took
256            })
257            .join()
258            .expect("next thread panicked")
259        });
260        assert!(next_took, "take after release must succeed");
261    }
