1//! Process-wide FFI lifecycle guard:
2//! `Uninit -> Initializing -> Active(level) -> Finalizing -> Finalized`.
3//!
4//! [`enter`] is called by every guarded extern wrapper in [`crate::ffi`] and
5//! rejects the call, without touching MPI: in `Uninit`; in `Initializing` on
6//! every thread but the one that called [`begin_init`]; once state is
7//! `Active(Single)`/`Active(Funneled)` and the caller is not that thread; and
8//! once state is `Finalizing` or `Finalized`. It returns an [`InFlight`]
9//! token that the wrapper holds for the whole call. At `Active(Serialized)`
10//! and `Active(Multiple)` the token counts the call in a per-thread shard of
11//! `IN_FLIGHT`; at `Single`/`Funneled` it is empty, and the check stays one
12//! relaxed load plus the init-thread test. [`drop_guard`] is the equivalent
13//! check for a `Drop` impl, which cannot return `Err`: it skips the MPI call
14//! silently once finalizing, and aborts the process on a wrong-thread drop
15//! below `Serialized`. `Initializing` is a transient state held only between
16//! [`begin_init`] and [`activate`]/[`abandon_init`]: `Mpi::init_thread` reads
17//! the world's rank and size in it, so `enter` admits the initializing thread
18//! there. `drop_guard` passes `Uninit` and `Initializing` through: a handle
19//! reaches a caller only after [`activate`], so none exists yet for a `Drop`
20//! impl to free.
21//!
22//! Finalizing is two steps. [`begin_finalize`] stores `Finalizing`, which
23//! refuses new calls, waits for the nonblocking scopes open on other threads
24//! to close, and at `Serialized`/`Multiple` waits ([`drain_shards`]) until
25//! every counted call has returned; [`end_finalize`] then stores `Finalized`,
26//! just before `MPI_Finalize` runs, and waits again for the completion calls
27//! admitted in between. A skipped `MPI_Finalize` leaves the state `Finalizing`
28//! for the rest of the process.
29//!
30//! [`enter_completion`] and [`check_completion`] are the entry for the calls
31//! that complete, test or cancel a request that is already pending, receive a
32//! message that is already matched, or close an RMA epoch that is already open
33//! (the closing fence, `MPI_Win_complete`, `MPI_Win_unlock`/`MPI_Win_unlock_all`,
34//! and `MPI_Win_wait`/`MPI_Win_test` of an exposure epoch). They differ from
35//! [`enter`] and [`check`] only in admitting `Finalizing`, under the thread rule
36//! of the granted level, including a counted call that read an active state
37//! just before the `Finalizing` store. The error-code lookup behind
38//! `Error::from_code` takes the same entry, so it never overlaps `MPI_Finalize`.
39//!
40//! A nonblocking scope holds a [`ScopeToken`] for its whole extent, at every
41//! thread level: [`open_scope`] refuses once `Finalizing` is stored, and
42//! [`scopes_on_this_thread`] lets `Mpi::drop` skip `MPI_Finalize` for a scope
43//! open on its own thread, whose requests the scope still completes. A scope
44//! opened on the init thread at `Single`/`Funneled` is tracked only by that
45//! thread's own count, which costs no atomic operation; every other scope is
46//! also counted in its thread's shard of `LIVE_SCOPES`, which [`begin_finalize`]
47//! waits on.
48//!
49//! An active persistent request and a matched, unreceived message each hold a
50//! [`HandleToken`], counted in a shard of `LIVE_HANDLES` at every thread level,
51//! whatever thread holds the handle. [`begin_finalize`] never waits for it,
52//! because the handle can be held by a thread that is itself waiting for the
53//! finalizer; [`live_handle_tokens`] lets `Mpi::drop` skip `MPI_Finalize`
54//! instead while the sum is nonzero.

84thread_local! {
85    /// Set by `begin_init` on the thread that wins it and cleared by
86    /// `abandon_init`, so at most one thread has it set. `Mpi` is `!Send`, so
87    /// that thread also makes every later write of `STATE`.
88    static ON_INIT_THREAD: Cell<bool> = const { Cell::new(false) };
89
90    /// This thread's index into `IN_FLIGHT`, `LIVE_SCOPES` and `LIVE_HANDLES`;
91    /// `usize::MAX` until its first counted use.
92    static SHARD: Cell<usize> = const { Cell::new(usize::MAX) };

94    /// How many nonblocking scopes are open on this thread.
95    static SCOPES_HERE: Cell<usize> = const { Cell::new(0) };
96
97    /// How many of those are also counted in a shard of `LIVE_SCOPES`.
98    static COUNTED_HERE: Cell<usize> = const { Cell::new(0) };
99}

101const SHARDS: usize = 64;

103/// One shard of `IN_FLIGHT`, `LIVE_SCOPES` or `LIVE_HANDLES`. 128-byte
104/// alignment keeps two shards off the same pair of cache lines, so threads
105/// counting on different shards do not contend.
106#[repr(align(128))]
107struct Shard(AtomicUsize);
108
109static IN_FLIGHT: [Shard; SHARDS] = [const { Shard(AtomicUsize::new(0)) }; SHARDS];
110
111static NEXT_SHARD: AtomicUsize = AtomicUsize::new(0);

113/// How many nonblocking scopes are open on any thread, except those opened on
114/// the init thread at `Single`/`Funneled`, which only `SCOPES_HERE` counts. A
115/// scope counts in the shard its thread's `SHARD` index names, and the sum of
116/// the shards is the count.
117static LIVE_SCOPES: [Shard; SHARDS] = [const { Shard(AtomicUsize::new(0)) }; SHARDS];
118
119/// How many persistent requests are active and matched messages are unreceived,
120/// on any thread and at any thread level. Each counts in the shard its opening
121/// thread's `SHARD` index names, and the token it holds subtracts from that
122/// same shard from whatever thread drops it; the sum of the shards is the
123/// count.
124static LIVE_HANDLES: [Shard; SHARDS] = [const { Shard(AtomicUsize::new(0)) }; SHARDS];

126/// Held by a guarded call or `Drop` path for as long as it may call MPI. At
127/// `Serialized`/`Multiple` it owns one count in a shard of `IN_FLIGHT`,
128/// which [`begin_finalize`] waits for; elsewhere it is empty.
129#[must_use]
130pub(crate) struct InFlight(Option<&'static Shard>);
131
132impl Drop for InFlight {
133    #[inline(always)]
134    fn drop(&mut self) {
135        if let Some(shard) = self.0 {
136            // Release: pairs with the SeqCst (hence acquiring) loads in
137            // `drain_shards`, so every MPI call made under this token
138            // happens-before the `MPI_Finalize` that follows the drain.
139            shard.0.fetch_sub(1, Ordering::Release);
140        }
141    }
142}

144/// Held by a nonblocking scope for its whole extent, at every thread level.
145/// [`begin_finalize`] waits for the counted tokens of scopes on other threads,
146/// and [`scopes_on_this_thread`] feeds the skip decision for the finalizing
147/// thread. A token is counted unless it was opened on the init thread at
148/// `Single`/`Funneled`; a scope opened before `Mpi::init` stays counted, in the
149/// shard of `LIVE_SCOPES` the token holds.
150#[must_use]
151pub(crate) struct ScopeToken(Option<&'static Shard>);
152
153impl Drop for ScopeToken {
154    #[inline]
155    fn drop(&mut self) {
156        SCOPES_HERE.with(|here| here.set(here.get() - 1));
157        if let Some(shard) = self.0 {
158            COUNTED_HERE.with(|here| here.set(here.get() - 1));
159            // Release: pairs with the SeqCst (hence acquiring) load in
160            // `wait_for_other_scopes`, so every MPI call the scope made
161            // happens-before the `MPI_Finalize` that follows the wait.
162            shard.0.fetch_sub(1, Ordering::Release);
163        }
164    }
165}

167/// Held by an active persistent request, at every thread level, from before its
168/// start until MPI completed it, and by a matched message from before its probe
169/// until MPI received it. Unlike a [`ScopeToken`] it is movable: it keeps
170/// no thread-local count and can be dropped on any thread, so it always counts in
171/// its shard and [`begin_finalize`] never waits for it. [`live_handle_tokens`]
172/// feeds the skip decision of `Mpi::drop`.
173#[must_use]
174pub(crate) struct HandleToken(&'static Shard);
175
176impl Drop for HandleToken {
177    #[inline]
178    fn drop(&mut self) {
179        // Release: pairs with the SeqCst (hence acquiring) loads in
180        // `live_handle_tokens`, so every MPI call that completed the request
181        // or received the message happens-before the `MPI_Finalize` that
182        // follows a sum of zero.
183        self.0 .0.fetch_sub(1, Ordering::Release);
184    }
185}

187/// `n` tokens counted by one increment, from [`open_handle_tokens`]. Each
188/// [`take`](Self::take) hands out one; what is not handed out by the time this
189/// drops is released with it.
190#[must_use]
191pub(crate) struct HandleTokens {
192    shard: &'static Shard,
193    left: Cell<usize>,
194}
195
196impl HandleTokens {
197    /// The next token, or `None` once all were handed out.
198    #[inline]
199    pub(crate) fn take(&self) -> Option<HandleToken> {
200        let left = self.left.get().checked_sub(1)?;
201        self.left.set(left);
202        Some(HandleToken(self.shard))
203    }
204}
205
206impl Drop for HandleTokens {
207    #[inline]
208    fn drop(&mut self) {
209        let left = self.left.get();
210        if left > 0 {
211            // Release: as in `HandleToken`'s drop.
212            self.shard.0.fetch_sub(left, Ordering::Release);
213        }
214    }
215}

217/// This thread's shard of `shards`, taken round-robin on first use and kept.
218/// `IN_FLIGHT`, `LIVE_SCOPES` and `LIVE_HANDLES` all go through it, so a thread
219/// uses one index in all three.
220#[inline]
221fn shard(shards: &'static [Shard; SHARDS]) -> &'static Shard {
222    let index = SHARD.with(Cell::get);
223    if index < SHARDS {
224        &shards[index]
225    } else {
226        assign_shard(shards)
227    }
228}
229
230#[cold]
231#[inline(never)]
232fn assign_shard(shards: &'static [Shard; SHARDS]) -> &'static Shard {
233    // Relaxed: the counter only spreads threads over shards; it orders
234    // nothing.
235    let index = NEXT_SHARD.fetch_add(1, Ordering::Relaxed) % SHARDS;
236    SHARD.with(|cell| cell.set(index));
237    &shards[index]
238}

240/// Move `Uninit` to `Initializing` and mark the calling thread as the init
241/// thread, serialising concurrent `Mpi::init`/`init_thread` calls. Returns
242/// `Err(Error::Finalized)` once state is `Finalizing` or `Finalized`, and
243/// `Err(Error::AlreadyInitialized)` for any other state (an `Mpi` is alive, or
244/// another thread is already initialising).
245/// Called once, at the top of `Mpi::init_thread`; every path out of that
246/// function afterward reaches either [`activate`] or [`abandon_init`].
247pub(crate) fn begin_init() -> Result<()> {
248    // Relaxed: the winner needs no order with an earlier writer of `Uninit`.
249    // That value is the initial one or `abandon_init`'s, and no later call
250    // rests on the calls an abandoned attempt made: `MPI_Finalized` may be
251    // called at any time and by several threads at once (MPI-4.1 §11.4.1),
252    // and after an `MPI_Init_thread` that failed a second one is erroneous
253    // whichever thread makes it (MPI-4.1 §11.2.1). A losing thread returns
254    // without calling MPI, whatever value its compare-exchange read.
255    match STATE.compare_exchange(UNINIT, INITIALIZING, Ordering::Relaxed, Ordering::Relaxed) {
256        Ok(_) => {
257            ON_INIT_THREAD.with(|c| c.set(true));
258            Ok(())
259        }
260        Err(FINALIZING | FINALIZED) => Err(Error::Finalized),
261        Err(_) => Err(Error::AlreadyInitialized),
262    }
263}

265/// Take the init-thread mark off the calling thread and move `Initializing`
266/// back to `Uninit`. Called when `Mpi::init_thread` does not complete after
267/// [`begin_init`] succeeded: MPI turned out already finalized, or
268/// `MPI_Init_thread` itself failed. Only the thread that won `begin_init`'s
269/// compare-exchange calls this, so a plain store is safe.
270pub(crate) fn abandon_init() {
271    // The mark first: once the state is given back, another thread can win
272    // `begin_init` and set its own.
273    ON_INIT_THREAD.with(|c| c.set(false));
274    // Relaxed: no reader needs the store ordered. The guard refuses a thread
275    // without the mark in `Uninit` as in `Initializing`, and the next
276    // `begin_init` needs no order with this thread (see there).
277    STATE.store(UNINIT, Ordering::Relaxed);
278}

280/// Move `Initializing` to `Active(level)`. Called once, from
281/// `Mpi::init_thread` after `MPI_Init_thread` succeeds, on the thread
282/// [`begin_init`] marked.
283pub(crate) fn activate(level: ThreadLevel) {
284    // Relaxed: stored before `STATE` on this thread, so a load that acquires
285    // the store below is ordered after this one too.
286    LEVEL.store(level as u8, Ordering::Relaxed);
287    // Release: a thread other than this one that reads `Active(Serialized)`,
288    // `Active(Multiple)` or a later state is admitted by the guard to an MPI
289    // call only through `enter_counted`. Its SeqCst load of `STATE` reads this
290    // store or one of the two later ones, both SeqCst and both made by this
291    // thread, so the admitted call is ordered after `MPI_Init_thread` and
292    // after the `LEVEL` store. That holds for a thread that was handed no
293    // handle too (a constructor that takes none), which nothing else orders
294    // after them.
295    STATE.store(1 + level as u8, Ordering::Release);
296}

298/// Opens a nonblocking scope: returns its [`ScopeToken`], or
299/// `Err(FERROMPI_ERR_FINALIZED)` once state is `Finalizing` or `Finalized`. A
300/// counted scope's count rises before its first request-creating call, so that
301/// call's per-call release cannot precede it; an uncounted scope relies on the
302/// skip path instead.
303#[inline]
304pub(crate) fn open_scope() -> std::result::Result<ScopeToken, c_int> {
305    // Relaxed: see `enter`'s comment.
306    let state = STATE.load(Ordering::Relaxed);
307    if state == FINALIZING || state == FINALIZED {
308        return Err(FERROMPI_ERR_FINALIZED);
309    }
310    let counted_in;
311    if state == ACTIVE_SINGLE || state == ACTIVE_FUNNELED {
312        if ON_INIT_THREAD.with(Cell::get) {
313            // Uncounted: at `Single`/`Funneled` only the init thread can create
314            // a request, and `Mpi` is `!Send`, so `Mpi::drop` runs on this very
315            // thread and skips `MPI_Finalize` while `scopes_on_this_thread()`
316            // is positive. No other thread's wait needs to see this scope.
317            SCOPES_HERE.with(|here| here.set(here.get() + 1));
318            return Ok(ScopeToken(None));
319        }
320        counted_in = shard(&LIVE_SCOPES);
321        // Relaxed: a scope on another thread holds no request at this level, so
322        // only its count matters, not its order.
323        counted_in.0.fetch_add(1, Ordering::Relaxed);
324    } else {
325        counted_in = shard(&LIVE_SCOPES);
326        // SeqCst: the increment of the pair with `begin_finalize`'s
327        // `Finalizing` store, as in `enter_counted`. Either the wait sees this
328        // count or the re-check below sees that store. Every other state takes
329        // this path, `Uninit` and `Initializing` included: a thread may open a
330        // scope before `Mpi` is initialized and post its first request after,
331        // and only this pair orders that count before the finalizer's wait.
332        counted_in.0.fetch_add(1, Ordering::SeqCst);
333        // SeqCst: reads `Finalizing` or `Finalized` if that store precedes the
334        // increment in the SeqCst order.
335        let state = STATE.load(Ordering::SeqCst);
336        if state == FINALIZING || state == FINALIZED {
337            counted_in.0.fetch_sub(1, Ordering::Release);
338            return Err(FERROMPI_ERR_FINALIZED);
339        }
340    }
341    SCOPES_HERE.with(|here| here.set(here.get() + 1));
342    COUNTED_HERE.with(|here| here.set(here.get() + 1));
343    Ok(ScopeToken(Some(counted_in)))
344}

346/// How many nonblocking scopes are open on the calling thread.
347pub(crate) fn scopes_on_this_thread() -> usize {
348    SCOPES_HERE.with(Cell::get)
349}

351/// Counts `n` persistent requests about to start, or one message about to be
352/// probed for, with one increment: returns the tokens, or
353/// `Err(FERROMPI_ERR_FINALIZED)` once state is `Finalizing` or
354/// `Finalized`, or `Err(FERROMPI_ERR_THREAD_LEVEL)` on a thread other than the
355/// init thread at `Single`/`Funneled`, counting nothing. The count rises before
356/// the guarded call that starts the requests or probes for the message, so that
357/// call's per-call release cannot precede it.
358///
359/// A token is counted on every thread and never takes [`open_scope`]'s uncounted
360/// branch: it can move to a thread whose own count `Mpi::drop` does not read.
361/// At `Single`/`Funneled` the increment is Relaxed, because every start,
362/// completion and release of a request runs on the init thread, where
363/// `Mpi::drop` reads the sum: another thread's start and completion are
364/// refused, and a drop there aborts while the request is active and, once
365/// `Finalizing`, leaks it with its token forgotten. A message's probe, receive
366/// and discard are refused there too, and a refused discard forgets the token.
367#[inline]
368pub(crate) fn open_handle_tokens(n: usize) -> std::result::Result<HandleTokens, c_int> {
369    // Relaxed: see `enter`'s comment.
370    let state = STATE.load(Ordering::Relaxed);
371    if state == FINALIZING || state == FINALIZED {
372        return Err(FERROMPI_ERR_FINALIZED);
373    }
374    let counted_in;
375    if state == ACTIVE_SINGLE || state == ACTIVE_FUNNELED {
376        if !ON_INIT_THREAD.with(Cell::get) {
377            return Err(FERROMPI_ERR_THREAD_LEVEL);
378        }
379        counted_in = shard(&LIVE_HANDLES);
380        // Relaxed: this thread is the only one that starts, completes or
381        // releases a request at this level (a drop elsewhere aborts in
382        // `Active` and leaks at `Finalizing`), and it reads the sum in
383        // `Mpi::drop`.
384        counted_in.0.fetch_add(n, Ordering::Relaxed);
385    } else {
386        counted_in = shard(&LIVE_HANDLES);
387        // SeqCst: the increment of the pair with `begin_finalize`'s
388        // `Finalizing` store, as in `open_scope`. Either the later sum sees this
389        // count or the re-check below sees that store. Every other state takes
390        // this path, `Uninit` and `Initializing` included.
391        counted_in.0.fetch_add(n, Ordering::SeqCst);
392        // SeqCst: reads `Finalizing` or `Finalized` if that store precedes the
393        // increment in the SeqCst order.
394        let state = STATE.load(Ordering::SeqCst);
395        if state == FINALIZING || state == FINALIZED {
396            counted_in.0.fetch_sub(n, Ordering::Release);
397            return Err(FERROMPI_ERR_FINALIZED);
398        }
399    }
400    Ok(HandleTokens {
401        shard: counted_in,
402        left: Cell::new(n),
403    })
404}

409#[cfg(test)]

414/// How many persistent requests are active and matched messages are unreceived,
415/// summed over the shards of `LIVE_HANDLES`. Read by `Mpi::drop` after
416/// [`begin_finalize`], which does not wait for them: a nonzero sum makes it
417/// skip `MPI_Finalize`.
418///
419/// The shards are read one after another, not as a snapshot, and a sum of zero
420/// is still exact:
421/// - A request adds one to a shard before its start call and takes one from that
422///   same shard once MPI completed it, so a shard never reads below its floor and
423///   the sum is zero only if every shard reads zero.
424/// - The `Finalizing` store precedes every load below. An open on the SeqCst path
425///   has its increment either before the load of its shard in the SeqCst order,
426///   so the load sees it unless it also sees the request's release, or after,
427///   and then its re-check reads `Finalizing`, the open fails and takes the count
428///   back. A back-out can only raise a reading, never hide a request: it can
429///   cause a skip that was not needed, never a missed one. So can a start or a
430///   probe the guarded call refuses after the open succeeded, and a probe that
431///   failed or matched nothing, each of which gives the count back after its
432///   call returned, and a `wait`, a `test`, an `mrecv` or the discard of a
433///   dropped message racing `Mpi::drop`, which releases its token after its own
434///   call returned.
435/// - Every write to a shard is a read-modify-write, so each Release decrement
436///   heads a release sequence holding all later writes, whichever thread made
437///   them. A load that sees a shard without a request's count read that
438///   request's release or a later write, and so acquires the `MPI_Wait`,
439///   `MPI_Test` or `MPI_Waitall` that completed it; for a message, the
440///   `MPI_Mrecv` that received it.
441pub(crate) fn live_handle_tokens() -> usize {
442    // SeqCst: the load of the pair with `open_handle_tokens`'s increment, and an
443    // acquire of the release sequence the value it reads belongs to.
444    LIVE_HANDLES
445        .iter()
446        .map(|shard| shard.0.load(Ordering::SeqCst))
447        .sum()
448}

450/// Waits until every counted scope open on another thread has closed. No
451/// timeout: a scope that never ends blocks this wait, as a blocked MPI call
452/// blocks [`drain_shards`].
453///
454/// It spins until the sum of the shards equals this thread's own counted
455/// scopes. Another thread's scope may be on any shard, this thread's own
456/// included, so the wait reads them all; the sum reaches that count only when
457/// every shard is at its floor. The shards are read one after another, not as a
458/// snapshot, and the sum is still exact:
459/// - A scope adds one to a shard and later takes one from that same shard. So
460///   by coherence a shard never reads below its floor, this thread's own
461///   scopes on it, the sum never reads below `here`, and it equals `here` only
462///   if no shard reads another thread's scope. A scope never moves between
463///   shards, so the reads need no common instant.
464/// - The `Finalizing` store precedes every load below. An open on the SeqCst
465///   path has its increment either before the load of its shard in the SeqCst
466///   order, so the load sees it unless it also sees the scope's decrement, or
467///   after, and then its re-check reads `Finalizing`, the open fails and takes
468///   the count back. A back-out only raises a reading for a moment: it can
469///   delay this wait, not end it early.
470/// - Every write to a shard is a read-modify-write, so each Release decrement
471///   heads a release sequence holding all later writes, whichever thread made
472///   them. A load that sees a shard without a scope's count read that scope's
473///   decrement or a later write, and so acquires every MPI call the scope made.
474/// - An open at `Single`/`Funneled` on another thread takes the Relaxed path
475///   with no re-check. It holds no request, so this wait need not see it.
476#[cold]
477fn wait_for_other_scopes() {
478    let here = COUNTED_HERE.with(Cell::get);
479    // SeqCst: the load of the pair with `open_scope`'s increment, and an
480    // acquire of the release sequence the value it reads belongs to.
481    while LIVE_SCOPES
482        .iter()
483        .map(|shard| shard.0.load(Ordering::SeqCst))
484        .sum::<usize>()
485        != here
486    {
487        for _ in 0..64 {
488            std::hint::spin_loop();
489        }
490        std::thread::yield_now();
491    }
492}

494/// Waits until every shard reads zero, that is, until every counted call and
495/// `Drop` path has returned. No timeout: a thread blocked forever inside an
496/// MPI call blocks this wait, as it would block `MPI_Finalize`.
497#[cold]
498fn drain_shards() {
499    for shard in &IN_FLIGHT {
500        // SeqCst: the shard load of the pair with `enter_counted`'s increment,
501        // and an acquire of the Release decrement it observes.
502        while shard.0.load(Ordering::SeqCst) != 0 {
503            for _ in 0..64 {
504                std::hint::spin_loop();
505            }
506            std::thread::yield_now();
507        }
508    }
509}

511/// Move an `Active` state to `Finalizing`, which refuses new calls and scope
512/// opens, then wait for the scopes open on other threads to close and, at
513/// `Serialized`/`Multiple`, for every counted call to return. Returns `true`.
514/// Returns `false` and changes nothing in any other state, so a stub `Mpi`
515/// dropped without init (`Uninit`) — or a second drop after finalize — is a
516/// no-op.
517pub(crate) fn begin_finalize() -> bool {
518    // Relaxed: `Mpi` is `!Send`, so this load only ever races with another
519    // thread's `enter()` load, never with a second write on this thread.
520    let prev = STATE.load(Ordering::Relaxed);
521    if !(ACTIVE_SINGLE..=ACTIVE_MULTIPLE).contains(&prev) {
522        return false;
523    }
524    // SeqCst: the store of the pair with `enter_counted`. A call whose
525    // increment precedes this store in the SeqCst order is seen by the drain
526    // below; a call whose increment follows it reads `Finalizing` and is
527    // refused. So no counted call can begin once `MPI_Finalize` does.
528    STATE.store(FINALIZING, Ordering::SeqCst);
529    // Scopes before calls: a scope's own calls are counted ones, so by the
530    // time the drain runs, no scope is still making them.
531    wait_for_other_scopes();
532    if prev >= ACTIVE_SERIALIZED {
533        drain_shards();
534    }
535    true
536}
537
538/// Move `Finalizing` to `Finalized`, then at `Serialized`/`Multiple` wait for
539/// the completion calls [`enter_completion`] admitted while `Finalizing`. Called
540/// by `Mpi::drop` after [`begin_finalize`] returned `true`, only on the path
541/// that goes on to call `MPI_Finalize`; a skipped `MPI_Finalize` never reaches
542/// it.
543pub(crate) fn end_finalize() {
544    // SeqCst: totally ordered after the `Finalizing` store and the first
545    // drain, and the store of the pair with `enter_counted`'s increment: a
546    // completion call whose increment precedes this store is seen by the
547    // drain below; one whose increment follows it reads `Finalized` and is
548    // refused.
549    STATE.store(FINALIZED, Ordering::SeqCst);
550    // Relaxed: this thread stored `LEVEL` in `activate`.
551    if LEVEL.load(Ordering::Relaxed) >= ThreadLevel::Serialized as u8 {
552        drain_shards();
553    }
554}

556/// Returns `true` once state is `Finalizing` or `Finalized`.
557/// `Mpi::is_finalized()` uses this so it reports `true` from the start of
558/// `Mpi::drop`, and still does after a skipped `MPI_Finalize` (a live window
559/// kept `Mpi::drop` from calling it): `begin_finalize()` already moved
560/// `STATE` to `Finalizing` before that skip decision runs, and it stays there.
561pub(crate) fn is_finalized() -> bool {
562    // Relaxed: no call is admitted on this answer. A `false` that is out of
563    // date sends `Mpi::is_finalized` on to `MPI_Finalized`, which MPI allows
564    // at any time (MPI-4.1 §11.4.1).
565    matches!(STATE.load(Ordering::Relaxed), FINALIZING | FINALIZED)
566}

568/// Called at the top of every guarded extern wrapper. Returns
569/// `Err(FERROMPI_ERR_FINALIZED)` once state is `Finalizing` or `Finalized`;
570/// `Err(FERROMPI_ERR_THREAD_LEVEL)` once state is `Active(Single)` or
571/// `Active(Funneled)` and the caller is not the init thread;
572/// `Err(FERROMPI_ERR_NOT_INITIALIZED)` in `Uninit`, and in `Initializing` on
573/// every thread but the init thread; otherwise the [`InFlight`] token the
574/// caller holds until its MPI call returns.
575#[inline(always)]
576pub(crate) fn enter() -> std::result::Result<InFlight, c_int> {
577    // Relaxed: this load only picks the path. The init thread reads its own
578    // stores. Any other thread is refused or goes to the counted path, whose
579    // SeqCst load is what orders its MPI call after `activate` (see there),
580    // so an old value admits nothing. A thread that holds a handle reads no
581    // value older than `activate`'s store: its loads are ordered after it by
582    // the hand-off of the handle, or by the counted call that made the handle.
583    let state = STATE.load(Ordering::Relaxed);
584    if state == ACTIVE_SINGLE || state == ACTIVE_FUNNELED {
585        if ON_INIT_THREAD.with(Cell::get) {
586            return Ok(InFlight(None));
587        }
588        return Err(FERROMPI_ERR_THREAD_LEVEL);
589    }
590    enter_other(state)
591}
592
593/// [`enter`] for every state but `Active(Single)`/`Active(Funneled)`, kept out
594/// of line so the `Funneled` path stays the single range test and init-thread
595/// check it was before calls were counted.
596#[inline(never)]
597fn enter_other(state: u8) -> std::result::Result<InFlight, c_int> {
598    if state == ACTIVE_SERIALIZED || state == ACTIVE_MULTIPLE {
599        return enter_counted(false);
600    }
601    if state == FINALIZING || state == FINALIZED {
602        return Err(FERROMPI_ERR_FINALIZED);
603    }
604    enter_before_active()
605}
606
607/// [`enter`] for `Uninit` and `Initializing`: admits the init thread, which
608/// reads the world's rank and size while `Initializing` (`Mpi::init_thread`),
609/// and refuses every other call. Cold and out of line: a process leaves these
610/// two states once, and [`enter_other`] is the path of every call at
611/// `Serialized`/`Multiple`.
612#[cold]
613#[inline(never)]
614fn enter_before_active() -> std::result::Result<InFlight, c_int> {
615    // `begin_init` sets the mark and `abandon_init` clears it, so in these two
616    // states it is set only on the thread that holds `Initializing`. A thread
617    // without it that read one of them late, after `activate`'s store, holds
618    // no handle that would order the two: it is refused, never admitted.
619    if ON_INIT_THREAD.with(Cell::get) {
620        return Ok(InFlight(None));
621    }
622    Err(FERROMPI_ERR_NOT_INITIALIZED)
623}

625/// [`enter`] for the calls that complete, test or cancel a request that is
626/// already pending, receive a message that is already matched, or close an RMA
627/// epoch that is already open (the closing fence, `MPI_Win_complete`,
628/// `MPI_Win_unlock`/`MPI_Win_unlock_all`, and `MPI_Win_wait`/`MPI_Win_test` of
629/// an exposure epoch). It differs only in `Finalizing`, which it admits under
630/// the thread rule of the granted level: the init-thread check at
631/// `Single`/`Funneled`, the counted path at `Serialized`/`Multiple`. That
632/// includes a call that read an active state just before the `Finalizing`
633/// store. `Finalized` is still refused. Also the entry of the error-code lookup.
634#[inline(always)]
635pub(crate) fn enter_completion() -> std::result::Result<InFlight, c_int> {
636    // Relaxed: see `enter`'s comment.
637    let state = STATE.load(Ordering::Relaxed);
638    if state == ACTIVE_SINGLE || state == ACTIVE_FUNNELED {
639        if ON_INIT_THREAD.with(Cell::get) {
640            return Ok(InFlight(None));
641        }
642        return Err(FERROMPI_ERR_THREAD_LEVEL);
643    }
644    enter_other_completion(state)
645}
646
647/// [`enter_completion`] for every state but `Active(Single)`/`Active(Funneled)`,
648/// out of line for the same reason as [`enter_other`]. Outside `Finalizing` it
649/// matches [`enter`], except that a counted call's re-check admits `Finalizing`:
650/// a call that read an active state just before `begin_finalize` stored
651/// `Finalizing` is not refused.
652#[inline(never)]
653fn enter_other_completion(state: u8) -> std::result::Result<InFlight, c_int> {
654    if state == ACTIVE_SERIALIZED || state == ACTIVE_MULTIPLE {
655        return enter_counted(true);
656    }
657    if state != FINALIZING {
658        return enter_other(state);
659    }
660    // Relaxed: `LEVEL` is stored once, by `activate`, before `STATE`. The
661    // init thread reads its own store, and a thread that holds a handle or
662    // that the counted path admitted before is ordered after it. Any other
663    // thread may still read `Single`, the value `LEVEL` starts with, and is
664    // then refused below; with the stored level it is refused below as well
665    // or goes to `enter_counted`, which orders its call.
666    if LEVEL.load(Ordering::Relaxed) >= ThreadLevel::Serialized as u8 {
667        return enter_counted(true);
668    }
669    if ON_INIT_THREAD.with(Cell::get) {
670        return Ok(InFlight(None));
671    }
672    Err(FERROMPI_ERR_THREAD_LEVEL)
673}

675/// The state check of [`enter`] without a token, for a call on a request:
676/// counts nothing, and refuses the states `enter` refuses from `Active` on. A
677/// request exists only from `Active` on, so `Uninit` and `Initializing` are not
678/// tested here. [`check_completion`], for the two `wait` pre-checks, builds
679/// on it; their guarded `ffi` call takes the token.
680#[inline(always)]
681fn check() -> c_int {
682    // Relaxed: see `enter`'s comment.
683    let state = STATE.load(Ordering::Relaxed);
684    if state == FINALIZING || state == FINALIZED {
685        return FERROMPI_ERR_FINALIZED;
686    }
687    if (state == ACTIVE_SINGLE || state == ACTIVE_FUNNELED) && !ON_INIT_THREAD.with(Cell::get) {
688        return FERROMPI_ERR_THREAD_LEVEL;
689    }
690    0
691}
692
693/// [`check`] for the calls that [`enter_completion`] admits: the same result
694/// in every state but `Finalizing`, which it admits under the thread rule of
695/// the granted level.
696#[inline(always)]
697pub(crate) fn check_completion() -> c_int {
698    let code = check();
699    if code == FERROMPI_ERR_FINALIZED {
700        return check_finalized(code);
701    }
702    code
703}
704
705/// [`check_completion`] after [`check`] refused the state as finalized: still
706/// refused at `Finalized`, admitted at `Finalizing`.
707#[cold]
708#[inline(never)]
709fn check_finalized(code: c_int) -> c_int {
710    // Relaxed: see `check`'s comment. `Finalizing` is left only for
711    // `Finalized`, so a second load never admits what `check` refused.
712    if STATE.load(Ordering::Relaxed) != FINALIZING {
713        return code;
714    }
715    // Relaxed: see `enter_other_completion`'s comment.
716    if LEVEL.load(Ordering::Relaxed) < ThreadLevel::Serialized as u8
717        && !ON_INIT_THREAD.with(Cell::get)
718    {
719        return FERROMPI_ERR_THREAD_LEVEL;
720    }
721    0
722}

724/// The counted path, for `Active(Serialized)` and `Active(Multiple)`: take
725/// this thread's shard, increment it, then re-check `STATE`. With `completion`
726/// the re-check refuses only `Finalized`; otherwise `Finalizing` too.
727#[inline]
728fn enter_counted(completion: bool) -> std::result::Result<InFlight, c_int> {
729    let shard = shard(&IN_FLIGHT);
730    // SeqCst: the increment of the pair with `begin_finalize`'s `Finalizing`
731    // store and `end_finalize`'s `Finalized` store. It is ordered before the
732    // re-check below, so either the drain sees this count or the re-check
733    // sees that store.
734    shard.0.fetch_add(1, Ordering::SeqCst);
735    let token = InFlight(Some(shard));
736    // SeqCst: reads `Finalizing` or `Finalized` if that store precedes the
737    // increment in the SeqCst order.
738    let state = STATE.load(Ordering::SeqCst);
739    if state == FINALIZED || (state == FINALIZING && !completion) {
740        return Err(FERROMPI_ERR_FINALIZED);
741    }
742    Ok(token)
743}

745/// Set for the duration of one guarded FFI call at the `Serialized` level,
746/// including while `Finalizing` or after a skipped `MPI_Finalize`, to detect
747/// two such calls overlapping. Debug-only:
748/// `Serialized` requires the caller to serialize its own MPI calls, so this
749/// is a diagnostic, not a correctness mechanism release builds must pay for.
750#[cfg(debug_assertions)]
751static SERIALIZED_IN_CALL: AtomicBool = AtomicBool::new(false);
752
753/// Attempts to take [`SERIALIZED_IN_CALL`]. Returns `true` on success.
754#[cfg(debug_assertions)]
755fn take_in_call_flag() -> bool {
756    // Acquire/Relaxed: a successful exchange must synchronize-with the
757    // matching `release_in_call_flag`'s Release store, so the guarded FFI
758    // call this thread is about to make cannot be reordered ahead of the
759    // previous holder's; a failed exchange holds nothing and needs no
760    // ordering.
761    SERIALIZED_IN_CALL
762        .compare_exchange(false, true, Ordering::Acquire, Ordering::Relaxed)
763        .is_ok()
764}
765
766/// Releases [`SERIALIZED_IN_CALL`].
767#[cfg(debug_assertions)]
768fn release_in_call_flag() {
769    // Release: pairs with `take_in_call_flag`'s Acquire, so this thread's
770    // guarded FFI call happens-before the next thread's that takes the flag.
771    SERIALIZED_IN_CALL.store(false, Ordering::Release);
772}

774/// The debug overlap check's hold on [`SERIALIZED_IN_CALL`], from
775/// [`begin_call`] until it drops. Dropping it releases the flag when it holds
776/// it, so a call that unwinds does not leave the flag set.
777#[cfg(debug_assertions)]
778#[must_use]
779pub(crate) struct CallFlag(bool);
780
781#[cfg(debug_assertions)]
782impl Drop for CallFlag {
783    fn drop(&mut self) {
784        if self.0 {
785            release_in_call_flag();
786        }
787    }
788}

790/// Called at the top of a guarded extern wrapper, after [`enter`] passes. At
791/// the `Serialized` level, in `Active` and also while `Finalizing`, takes
792/// [`SERIALIZED_IN_CALL`] and returns the [`CallFlag`] that holds it, or
793/// `Err(FERROMPI_ERR_THREAD_LEVEL)` if another call already holds it. At every
794/// other level, returns a [`CallFlag`] that holds nothing and leaves the flag
795/// alone. A caller binds the result after its [`InFlight`] token, so the flag
796/// is released before the token.
797#[cfg(debug_assertions)]
798pub(crate) fn begin_call() -> std::result::Result<CallFlag, c_int> {
799    // Relaxed: this runs after `enter` or `enter_completion` passed: on the
800    // init thread, which stores `LEVEL` (`Single` until then), or on a thread
801    // the counted path admitted, whose SeqCst load of `STATE` is ordered
802    // after that store.
803    if LEVEL.load(Ordering::Relaxed) == ThreadLevel::Serialized as u8 {
804        if take_in_call_flag() {
805            Ok(CallFlag(true))
806        } else {
807            Err(FERROMPI_ERR_THREAD_LEVEL)
808        }
809    } else {
810        Ok(CallFlag(false))
811    }
812}

814/// Called at the top of a `Drop` impl's MPI-calling path, in place of
815/// [`enter`] (a `Drop` impl cannot propagate an `Err`). Returns `None` once
816/// state is `Finalizing` or `Finalized`, so the caller skips its MPI call
817/// silently. Once state is `Active(Single)`/`Active(Funneled)` and the caller
818/// is not the init thread, this never returns: see [`drop_abort`]. Otherwise
819/// returns the [`InFlight`] token, which the caller binds for the whole
820/// `drop` body so it covers every MPI call there.
821pub(crate) fn drop_guard(type_name: &'static str) -> Option<InFlight> {
822    // Relaxed: see `enter`'s comment.
823    let state = STATE.load(Ordering::Relaxed);
824    if state == FINALIZING || state == FINALIZED {
825        return None;
826    }
827    if state == ACTIVE_SINGLE || state == ACTIVE_FUNNELED {
828        if !ON_INIT_THREAD.with(Cell::get) {
829            drop_abort(type_name);
830        }
831    } else if state == ACTIVE_SERIALIZED || state == ACTIVE_MULTIPLE {
832        return enter_counted(false).ok();
833    }
834    Some(InFlight(None))
835}

837/// Prints `ferrompi: <type_name> dropped on thread <label>` to stderr and
838/// aborts the process. Split out of [`drop_guard`] and marked `#[cold]` so
839/// the wrong-thread path does not bloat the common-case branch.
840#[cold]
841fn drop_abort(type_name: &'static str) -> ! {
842    let current = std::thread::current();
843    let label = match current.name() {
844        Some(name) => name.to_string(),
845        None => format!("{:?}", current.id()),
846    };
847    // eprintln! panics on a closed stderr, and a panic inside `Drop` during
848    // unwinding aborts without printing this message, so the write result
849    // is ignored instead.
850    let _ = writeln!(
851        std::io::stderr(),
852        "ferrompi: {type_name} dropped on thread {label}"
853    );
854    std::process::abort();
855}
